* Example Sentence Prompt
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'examples
  :name 'examples
  :label "Example"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Example sentence: " '("⚠ WARNING") nil nil nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "WARNING")))
                `(:properties (("EXAMPLE" . ,val))
                  :post-apply (,(apply-partially
                                 #'my/manifolding-atlas--apply-scoped-post
                                 'examples "EXAMPLE" val))))))
#+end_src
