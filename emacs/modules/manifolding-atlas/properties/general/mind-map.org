#+title:      Mind Map Placement Prompt
#+priority:   6


* Mind Map


Why: Place the new note in the mind map tree — parent, child, sibling, or unknown.

** Prompt
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-mind-map ()
  "Prompt for mind-map placement. Returns (:placement STR :target-id STR-or-nil)."
  (let ((choice (read-char-choice
                 "[p]arent  [c]hild  [s]ibling  [u]nknown: "
                 '(?p ?c ?s ?u ?\r ?\n)))
        target)
    (when (memq choice '(?\r ?\n)) (setq choice ?u))
    (unless (eq choice ?u)
      (pcase choice
        (?p (setq target (manifolding-atlas-select "Child to place under this note: "
                                         :require-match t)))
        (?c (setq target (manifolding-atlas-select "Parent for this note: "
                                         :require-match t)))
        (?s (setq target (manifolding-atlas-select "Place after this note: "
                                         :require-match t)))))
    (list :placement (pcase choice
                       (?p "parent") (?c "child") (?s "sibling") (?u "root"))
          :target-id (when target (manifolding-atlas-note-id target)))))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'mind-map
  :name 'mind-map
  :label "Mind Map"
  :prompt-fn #'my/manifolding-atlas-prompt-mind-map
  :contexts '(file task)
  :merge-strategy 'replace
  :to-plist (lambda (val _ctx)
               (when val
                 (let ((placement (plist-get val :placement))
                       (target-id (plist-get val :target-id))
                       result)
                   (if (string= placement "root")
                       (push (cons "MIND_MAP_PLACEMENT" "WARNING") result)
                     (let ((link (my/manifolding-atlas--ensure-ref-note 'mm-placement placement)))
                       (push (cons "MIND_MAP_PLACEMENT" (or link placement)) result)))
                   (list :properties (nreverse result)
                         :post-apply (list (apply-partially
                                            #'mm/--apply-placement
                                            placement target-id)))))))
#+end_src
