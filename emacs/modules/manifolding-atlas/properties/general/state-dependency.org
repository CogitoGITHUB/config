#+title:      State Dependency Prompt
#+priority:   5

* State Dependency
Why: Tracks whether the note is blocked on something or has been delegated.
Default is nil (no dependency state).

** Default (default nil)
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-state-dependency ()
  "Prompt for STATE_DEPENDENCY. Returns \"WARNING\" if skipped."
  (let ((val (completing-read
              "STATE_DEPENDENCY: "
              '("⚠ WARNING" "WAITING" "DELEGATED")
              nil t nil nil "⚠ WARNING")))
    (if (string-equal val "⚠ WARNING") "WARNING" val)))
#+end_src

** With initial value
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-state-dependency-initial (initial)
  "Prompt for STATE_DEPENDENCY with INITIAL pre-selected."
  (completing-read
   "STATE_DEPENDENCY: "
   '("WAITING" "DELEGATED")
   nil t nil nil initial))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'state-dependency
  :name 'state-dependency
  :label "State: Dependency"
  :prompt-fn #'my/manifolding-atlas-prompt-state-dependency
  :contexts '(file heading task)
  :read-current-fn (lambda () (org-entry-get nil "STATE_DEPENDENCY"))
  :merge-strategy 'replace
   :to-plist (lambda (val _ctx)
               (if (or (null val) (string-equal val "WARNING"))
                   `(:properties (("STATE_DEPENDENCY" . "WARNING"))
                     :post-apply (,(apply-partially
                                    #'my/manifolding-atlas--clear-state-transcribes
                                    "state-dependency")))
                 (let* ((link (my/manifolding-atlas--ensure-ref-note 'state-dependency val))
                        (ref-id (my/manifolding-atlas--id-from-link link)))
                   `(:post-apply (,(apply-partially
                                    #'my/manifolding-atlas--apply-state-post
                                    "state-dependency" val ref-id)))))))
#+end_src
