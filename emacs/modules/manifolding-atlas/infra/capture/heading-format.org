#+title: Manifolding Atlas — Heading-Format Note Creation
#+priority: 12
* Heading-Format Note Creation
:PROPERTIES:
:CATEGORY: editing
:END:

Creates new manifolding-atlas notes with the title as an Org heading (`* Title`) instead
of a file-level `#+title:` keyword. This ensures `org-back-to-heading`
succeeds, making the Properties Update Plugin work without file-level fallbacks.

Advice on `manifolding-atlas--format-note-content` transforms the standard file-level
output into heading-based format:
- `#+title: Note Title` → `* Note Title`
- `#+filetags: :tag1:tag2:` → inline tags on the heading line
- Properties drawer and body stay in place

#+begin_src emacs-lisp
(defun my/manifolding-atlas--headingify-note-content (orig-fn id title &optional head meta tags properties)
  "Wrap `manifolding-atlas--format-note-content' to produce heading-based notes.
Injects TODO state from `my/manifolding-atlas--creation-todo' into the heading line."
  (let (content title-re filetags-re todo)
    (setq content (funcall orig-fn id title head meta tags properties))
    (setq title-re "^#\\+title: \\(.*\\)$")
    (setq filetags-re "^#\\+filetags: :\\(.*\\):$")
    (setq todo my/manifolding-atlas--creation-todo)
    (setq my/manifolding-atlas--creation-todo nil)
    (with-temp-buffer
      (insert content)
      (goto-char (point-min))
      (if (re-search-forward title-re nil t)
          (let ((note-title (match-string 1)))
            (delete-region (match-beginning 0) (progn (forward-line 1) (point)))
            (let ((tags-str (when (re-search-forward filetags-re nil t)
                              (match-string 1))))
              (when tags-str
                (delete-region (match-beginning 0) (progn (forward-line 1) (point))))
              (goto-char (point-min))
              (insert (format "* %s%s%s\n"
                              (if todo (concat todo " ") "")
                              note-title
                              (if tags-str (concat "   :" tags-str ":") "")))
              (goto-char (point-min))
              (while (looking-at "^\\s-*$")
                (delete-region (point) (progn (forward-line 1) (point))))
              (buffer-string)))
        content))))

(advice-add 'manifolding-atlas--format-note-content :around #'my/manifolding-atlas--headingify-note-content)
#+end_src
