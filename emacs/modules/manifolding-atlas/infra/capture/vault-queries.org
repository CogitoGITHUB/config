** Relationship Queries
Parent/child/source lookups via SOURCE and PARENT properties.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-children-of (note)
  "Return list of manifolding-atlas notes whose :SOURCE: property is NOTE's ID."
  (let ((id (manifolding-atlas-note-id note)))
    (manifolding-atlas-db-query
     (lambda (n)
       (string= (alist-get "SOURCE" (manifolding-atlas-note-properties n)) id)))))

(defun my/manifolding-atlas-parent-of (note)
  "Return the manifolding-atlas note referenced by NOTE's :PARENT: property, or nil."
  (when-let ((parent-id (alist-get "PARENT" (manifolding-atlas-note-properties note))))
    (manifolding-atlas-db-get-by-id parent-id)))

(defun my/manifolding-atlas-source-of (note)
  "Return the manifolding-atlas note referenced by NOTE's :SOURCE: property, or nil."
  (when-let ((source-id (alist-get "SOURCE" (manifolding-atlas-note-properties note))))
    (manifolding-atlas-db-get-by-id source-id)))
#+end_src
** Vault Health
Reports dead links, orphan notes, isolated notes and title collisions
in a single *manifolding-atlas-health* buffer.

*** Section writers
#+begin_src emacs-lisp
(defun my/manifolding-atlas--health-section (label entries format-fn)
  "Insert a LABEL section rendering ENTRIES via FORMAT-FN."
  (insert (format "\n%s: %d\n" label (length entries)))
  (dolist (e entries)
    (insert (funcall format-fn e))))
#+end_src

*** Health command
#+begin_src emacs-lisp
(defun my/manifolding-atlas-check-vault ()
  "Report dead links, orphan notes, isolated notes and title collisions."
  (interactive)
  (let ((dead       (manifolding-atlas-db-query-dead-links))
        (orphans    (manifolding-atlas-db-query-orphan-notes))
        (isolated   (manifolding-atlas-db-query-isolated-notes))
        (collisions (manifolding-atlas-db-query-title-collisions)))
    (with-current-buffer (get-buffer-create "*manifolding-atlas-health*")
      (erase-buffer)
      (insert (format "Dead links: %d\n" (length dead)))
      (dolist (d dead)
        (insert (format "  %s -> missing %s\n"
                        (manifolding-atlas-note-title (car d)) (cdr d))))
      (my/manifolding-atlas--health-section
       "Orphan notes" orphans
       (lambda (o) (format "  %s\n" (manifolding-atlas-note-title o))))
      (my/manifolding-atlas--health-section
       "Isolated notes" isolated
       (lambda (i) (format "  %s\n" (manifolding-atlas-note-title i))))
      (my/manifolding-atlas--health-section
       "Title collisions" collisions
       (lambda (c) (format "  \"%s\" -- %d notes\n"
                           (car c) (length (cdr c)))))
      (display-buffer (current-buffer)))))
#+end_src
