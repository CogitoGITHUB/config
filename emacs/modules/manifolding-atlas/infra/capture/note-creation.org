** Fast mode level menu
Direct letter selection; press ? inside the prompt for a help
window describing every option. Always returns a number 0-3.
SELECTION (s) asks which properties to customize; everything
else gets WARNING.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--fast-level ()
  "Prompt for capture speed."
  (let* ((ans (read-multiple-choice
               "Capture speed"
               '((?a "NORMAL" "every prompt fires — the full interrogation")
                 (?o "SKIP GENERAL" "specialized prompts still ask; general ones take defaults")
                 (?e "SKIP ALL" "accept every default — zero questions")
                 (?u "QUICK" "defaults, and no body capture afterwards")
                 (?s "SELECTION" "select which properties to customize; everything else gets WARNING")
                 (?\r "NORMAL" "same as a"))))
          ;; rmc normally returns (KEY . DESCRIPTION); tolerate a bare
          ;; key or anything else by normalizing first.
          (key (if (consp ans) (car ans) ans)))
    (pcase key
      (?o (setq my/manifolding-atlas--fast-level 1)
          (message "Capture mode: SKIP GENERAL") 1)
      (?e (setq my/manifolding-atlas--fast-level 2)
          (message "Capture mode: SKIP ALL") 2)
      (?u (setq my/manifolding-atlas--fast-level 3)
          (message "Capture mode: QUICK") 3)
      (?s (setq my/manifolding-atlas--fast-level 4)
          (message "Capture mode: SELECTION") 4)
      (_ (setq my/manifolding-atlas--fast-level 0)
         (message "Capture mode: NORMAL") 0)))
#+end_src

** Slug spacing advice
#+begin_src emacs-lisp
(setq use-dialog-box nil)

(defun my/manifolding-atlas-title-to-slug-spaces (orig-fun title)
  "Replace underscores with spaces in the slug from ORIG-FUN."
  (replace-regexp-in-string "_" " " (funcall orig-fun title)))

(advice-add 'manifolding-atlas-title-to-slug :around #'my/manifolding-atlas-title-to-slug-spaces)
#+end_src
** Default Parameters
#+begin_src emacs-lisp
(defvar my/manifolding-atlas-default-subdir-root nil
  "When non-nil, use this as the default directory for the subdirectory prompt
instead of `my/manifolding-atlas-root-dir'.")

(with-eval-after-load 'manifolding-atlas
  (setq manifolding-atlas-create-default-function
        (lambda (title)
          (let* ((raw-name (read-string
                            (format "File name for \"%s\" (blank = slug): " title)))
                 (slug (if (string-empty-p raw-name)
                           (manifolding-atlas-title-to-slug title)
                         (file-name-sans-extension raw-name)))
                  (subdir (my/manifolding-atlas-choose-subdir (or my/manifolding-atlas-default-subdir-root (my/manifolding-atlas-root-dir))))
                 (fname (concat subdir slug))
                 (prompts (my/manifolding-atlas-collect-prompts-with-fast '() '(file))))
            (setq my/manifolding-atlas--pending-post-apply (plist-get prompts :post-apply))
            (when (fboundp 'my/manifolding-atlas-org-prompt--park-deferred)
              (my/manifolding-atlas-org-prompt--park-deferred))
            (list
             :file-name  fname
             :tags       (plist-get prompts :tags)
             :properties (plist-get prompts :properties)
             :body       (plist-get prompts :body))))))
#+end_src
