* Capture Entry Points
:PROPERTIES:
:CATEGORY: editing
:END:
In manifolding-atlas v2 there is no org-roam-capture. The pattern is:
=manifolding-atlas-create= makes the file and registers it in the DB; org-capture opens
that file so you fill in the body. Put your =doct= declarations in
=manifolding-atlas/templates/manifolding-atlas.el= (loaded automatically by the doct loader).

* Finalizers

Named functions so the async prompt session can call them later.
Dynamic-binding compilation means lambdas must not capture lexical
variables — =apply-partially= with named functions is the safe pattern.

** File capture tail
#+begin_src emacs-lisp
(defun my/manifolding-atlas--finish-file-capture (note fname fast)
  "Tail of file captures: open NOTE per FAST mode.
Level 3 skips the body editor and asks to open the file instead."
  (if (= fast 3)
      (when-let ((file (manifolding-atlas-note-path note)))
        (when (y-or-n-p (format "Open %s? " (file-name-nondirectory file)))
          (find-file file)))
    (my/manifolding-atlas--open-capture-for-note note)))
#+end_src

* New file capture

Prompts for title, subdir and tags; creates the note; opens capture for body.

** Entry point

*** Title/subdir prompts and creation
#+begin_src emacs-lisp
(defun my/manifolding-atlas-capture-new-file ()
  "Prompt for title, subdir and tags; create the note; open capture for body."
  (let* ((manifolding-atlas-create-default-function (lambda (_) '()))
         (title  (read-string "Note title: "))
         (raw-name (read-string (format "File name for \"%s\" (blank = slug): " title)))
         (slug (if (string-empty-p raw-name)
                   (manifolding-atlas-title-to-slug title)
                 (file-name-sans-extension raw-name)))
         (subdir (my/manifolding-atlas-choose-subdir))
         (fname (concat subdir slug))
         (fast (my/manifolding-atlas--fast-level))
         (prompts (my/manifolding-atlas-collect-prompts-with-fast '(note) '(file) fast))
          (note   (my/manifolding-atlas--create-file-wrapper
                   title
                   fname
                   (plist-get prompts :tags)
                   (plist-get prompts :properties)
                   (plist-get prompts :body))))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (my/manifolding-atlas-org-prompt--session-maybe-start
     note (apply-partially #'my/manifolding-atlas--finish-file-capture
                           note fname fast))))

** Create file wrapper
Shadows the default-function so create doesn't double-collect.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--create-file-wrapper (title fname tags properties body)
  (let ((manifolding-atlas-create-default-function (lambda (_) '())))
    (manifolding-atlas-create title fname
                   :tags tags
                   :properties properties
                   :body body)))
#+end_src

* Heading capture

Selects a parent note, creates a child heading under it, then opens via org-capture.

** Entry point

*** Parent selection helper
Offers the note at point first, then falls back to search.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-capture--select-parent ()
  "Pick a parent note: current buffer's note, or search."
  (let ((buffer-note (my/manifolding-atlas--buffer-note)))
    (if buffer-note
        (if (member (read-char-choice
                     (format "Use \"%s\" as parent? (RET/y=yes, n=search) "
                             (manifolding-atlas-note-title buffer-note))
                     '(?y ?n ?\r ?\n))
                    '(?y ?\r ?\n))
            buffer-note
          (manifolding-atlas-select "Parent note: " :require-match t))
      (manifolding-atlas-select "Parent note: " :require-match t))))
#+end_src

*** Heading capture command
#+begin_src emacs-lisp
(defun my/manifolding-atlas-capture-heading ()
  "Select a parent note, create a heading under it, then open via org-capture."
  (interactive)
  (let (manifolding-atlas-create-default-function parent title prompts note fast)
    (setq manifolding-atlas-create-default-function (lambda (_) '()))
    (setq fast (my/manifolding-atlas--fast-level))
    (setq parent (my/manifolding-atlas-capture--select-parent))
    (setq title (read-string "Heading title: "))
    (setq prompts (my/manifolding-atlas-collect-prompts-with-fast
                   '(heading) nil fast))
    (setq my/manifolding-atlas--pending-post-apply nil)
    (setq note (manifolding-atlas-create title nil
                              :parent parent
                              :tags   (plist-get prompts :tags)
                              :properties (plist-get prompts :properties)
                              :after  (plist-get prompts :after)))
    (unless (manifolding-atlas-note-p note)
      (error "manifolding-atlas-create returned %S (expected a manifolding-atlas-note)" note))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (my/manifolding-atlas-org-prompt--session-maybe-start
     note (apply-partially #'my/manifolding-atlas--finish-heading-capture
                           note fast))))
#+end_src

* Insert heading at point
Runs heading prompts for metadata, then opens a capture buffer
for body editing. The new heading becomes a child of the heading
at point, or a child of the file if point is at file level.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-insert-heading (title)
  "Insert a manifolding-atlas heading at point in the current buffer."
  (interactive (list (read-string "Heading title: ")))
  (let (id parent-note prompts note fast)
    (setq fast (my/manifolding-atlas--fast-level))
    (setq id (my/manifolding-atlas-links--current-note-id))
    (setq parent-note (manifolding-atlas-db-get-by-id id))
    (setq prompts (my/manifolding-atlas-collect-prompts-with-fast
                   '(heading) nil fast))
    (setq my/manifolding-atlas--pending-post-apply nil)
    (setq note (manifolding-atlas-create title nil
                               :parent parent-note
                              :tags (plist-get prompts :tags)
                              :properties (plist-get prompts :properties)
                              :after (plist-get prompts :after)))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (my/manifolding-atlas-org-prompt--session-maybe-start
     note (apply-partially #'my/manifolding-atlas--finish-heading-capture
                           note fast t))))
#+end_src
