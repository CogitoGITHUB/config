** Open Capture for Note
#+begin_src emacs-lisp
(defvar my/manifolding-atlas--chosen-capture nil
  "The capture template name chosen in the current capture flow.
Set by `my/manifolding-atlas--choose-capture', used by
`my/manifolding-atlas--open-capture-for-note' to store `:TEMPLATE:' on the note.")

(defun my/manifolding-atlas--capture-file-hash (name)
  "Return a SHA-256 hash of capture file NAME, or nil."
  (let ((file (my/manifolding-atlas--capture-path name)))
    (when (file-exists-p file)
      (with-temp-buffer
        (insert-file-contents file)
        (secure-hash 'sha256 (buffer-string))))))

(defun my/manifolding-atlas--choose-capture (&optional note)
  "Choose a capture template via completing-read with preview.
When NOTE is a manifolding-atlas-note, preview shows the note's file with the
template appended — exactly what will land on disk after capture."
  (let ((title (if note (manifolding-atlas-note-title note) nil))
        (note-file (if note (manifolding-atlas-note-path note) nil))
        names current preview-win chosen)
    (setq names (cons "none" (my/manifolding-atlas--capture-names)))
    (setq current "none")
    (delete-other-windows)
    (setq preview-win (split-window-below
                       (- (floor (* 0.85 (frame-height))))))
    (when-let ((buf (my/manifolding-atlas--capture-buffer current title note-file)))
      (set-window-buffer preview-win buf))
    ;; ── completing-read with live buffer swap ──────────────
    (setq chosen
          (minibuffer-with-setup-hook
              (lambda ()
                (add-hook
                 'post-command-hook
                 (lambda ()
                   (let ((candidate
                          (ignore-errors
                            (when (fboundp 'vertico--candidate)
                              (vertico--candidate)))))
                     (when (and (stringp candidate)
                                (member candidate names)
                                (not (equal candidate current))
                                (window-live-p preview-win))
                       (if (equal candidate "none")
                           (let ((buf (get-buffer-create " *capture-preview-empty*")))
                             (when note-file
                               (with-current-buffer buf
                                 (erase-buffer)
                                 (insert-file-contents note-file)
                                 (org-mode)
                                 (outline-show-all)
                                 (when (fboundp 'org-indent-mode)
                                   (org-indent-mode 1))))
                             (set-window-buffer preview-win buf))
                         (when-let ((buf (my/manifolding-atlas--capture-buffer candidate title note-file)))
                           (set-window-buffer preview-win buf)))
                       (setq current candidate))))
                 nil 'local))
            (completing-read "Body template: " names nil t nil nil "none")))
    ;; ── Cleanup preview ────────────────────────────────────────
    (my/manifolding-atlas--capture-cleanup names)
    (delete-window preview-win)
    (setq my/manifolding-atlas--chosen-capture
          (if (and chosen
                   (not (string-empty-p chosen))
                   (not (equal chosen "none")))
              chosen
            nil))
    (if my/manifolding-atlas--chosen-capture
        (with-temp-buffer
          (insert-file-contents
           (my/manifolding-atlas--capture-path my/manifolding-atlas--chosen-capture))
          (buffer-string))
      "%?")))

(defun my/manifolding-atlas--open-capture-for-note (note &optional template)
  "Open org-capture for NOTE with capture preview.
When a template is chosen, stores `:TEMPLATE:' and `:TEMPLATE_HASH:'
properties on NOTE for schema validation.
When TEMPLATE is \"\" (empty), skip template selection and use empty body."
  (let* ((file (manifolding-atlas-note-path note))
         (tmpl (cond
                ((stringp template) template)
                (my/manifolding-atlas--skip-capture "")
                (t (my/manifolding-atlas--choose-capture note))))
         (use-template (or (stringp template)
                           my/manifolding-atlas--chosen-capture)))
    (unless file
      (user-error "Note has no file path"))
    (when use-template
      (my/manifolding-atlas--property-put note "TEMPLATE" use-template)
      (my/manifolding-atlas--property-put note "TEMPLATE_HASH"
                                (or (my/manifolding-atlas--capture-file-hash use-template)
                                    "unknown")))
    (if (and my/manifolding-atlas--skip-capture (not (stringp template)))
        (find-file file)
      (let ((org-capture-templates
             `(("v" "Edit note body"
                 plain
                 (file+function
                  ,file
                  (lambda ()
                    (goto-char (point-max))
                    (skip-chars-backward " \t\n")))
                 ,tmpl
                 :kill-buffer t
                 :unnarrowed t)))
            (org-capture-use-indirect-buffer nil))
        (when-let ((buf (get-file-buffer file)))
          (when (eq buf (current-buffer))
            (switch-to-buffer (other-buffer))))
        (org-capture nil "v")))))

(defun my/manifolding-atlas--cleanup-after-capture ()
  "After org-capture finalizes, prompt to open the new note.
Runs via `org-capture-after-finalize-hook'."
  (when (and (bound-and-true-p my/manifolding-atlas--new-note)
             my/manifolding-atlas--new-note)
    (let ((file (manifolding-atlas-note-path my/manifolding-atlas--new-note)))
      (when (and file (not my/manifolding-atlas--skip-open-prompt))
        (when (y-or-n-p (format "Open %s? " (file-name-nondirectory file)))
          (find-file file)))))
  (setq my/manifolding-atlas--new-note nil)
  (setq my/manifolding-atlas--skip-open-prompt nil))

(add-hook 'org-capture-after-finalize-hook #'my/manifolding-atlas--cleanup-after-capture)
(add-hook 'org-capture-mode-hook (lambda () (modaled-set-state "insert")))

** Property Utilities
#+begin_src emacs-lisp
(defun my/manifolding-atlas-links--current-note-id ()
  "Return the ID of the manifolding-atlas note at point, or nil.
Checks heading-level ID first, then file-level ID."
  (or (org-id-get)
      (org-entry-get (point-min) "ID")))

(defun my/manifolding-atlas--property-put (note key value)
  "Set org property KEY to VALUE on NOTE using `org-entry-put'.
This stores the property inside NOTE's property drawer, making it
queryable via `manifolding-atlas-note-properties' and the manifolding-atlas SQL properties table."
  (let* ((file (manifolding-atlas-note-path note))
         (pos (manifolding-atlas-note-pos note))
         (level (manifolding-atlas-note-level note)))
    (with-current-buffer (find-file-noselect file)
      (save-excursion
        (cond
         ((and level (> level 0) pos)
          (goto-char pos)
          (org-back-to-heading t))
         (t
          (goto-char (point-min))))
        (org-entry-put (point) key value)
        (save-buffer)))))

** Capture Advice
#+begin_src emacs-lisp
(defun my/manifolding-atlas--open-in-capture (note)
  "After manifolding-atlas-create: record source relationship, set up capture."
  (when note
    (when-let* ((source-id (my/manifolding-atlas-links--current-note-id))
                ((not (string= source-id (manifolding-atlas-note-id note)))))
      (my/manifolding-atlas--property-put note "SOURCE" source-id))
    (setq my/manifolding-atlas--new-note note))
  note)

(defun my/manifolding-atlas--open-capture-new-note (note)
  "Finalizer for parked prompt sessions: open capture for NOTE."
  (my/manifolding-atlas--open-capture-for-note note))

(defun my/manifolding-atlas-find-advice (original-fn &rest args)
  "Wrap `manifolding-atlas-find' to open org-capture for newly created notes.
The `:filter-return' advice on `manifolding-atlas-create' `setq's the global
`my/manifolding-atlas--new-note' so `org-capture-after-finalize-hook' can
access it later for the \"Open file?\" prompt.
Parked deferred prompts from the default-function collection run as an
async session against the new note before capture opens."
  (setq my/manifolding-atlas--new-note nil)
  (let* ((result (apply original-fn args)))
    (when my/manifolding-atlas--new-note
      (if (fboundp 'my/manifolding-atlas-org-prompt--session-start-parked)
          (my/manifolding-atlas-org-prompt--session-start-parked
           my/manifolding-atlas--new-note
           (apply-partially #'my/manifolding-atlas--open-capture-new-note
                            my/manifolding-atlas--new-note))
        (my/manifolding-atlas--open-capture-for-note my/manifolding-atlas--new-note)))
    result))

(with-eval-after-load 'manifolding-atlas
  (advice-add 'manifolding-atlas-create :filter-return #'my/manifolding-atlas--open-in-capture)
  (advice-add 'manifolding-atlas-find :around #'my/manifolding-atlas-find-advice)
  (advice-add 'manifolding-atlas-insert :around #'my/manifolding-atlas-find-advice))
#+end_src
