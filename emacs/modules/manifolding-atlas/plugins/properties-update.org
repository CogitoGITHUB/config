*** Note Contexts
#+begin_src emacs-lisp
(defun my/manifolding-atlas--note-contexts (note)
  "Return contexts applicable to NOTE.
If the note carries a tag matching a specialized registered context
(contact, dictionary, rule), return only those specialized contexts so
the menu adapts to the note's kind.  Otherwise return the base
'file/'heading context derived from the note's level."
  (let* ((level (manifolding-atlas-note-level note))
         (base (if (and level (> level 0)) 'heading 'file))
         (tags (manifolding-atlas-note-tags note))
         (registered (delete-dups
                      (seq-mapcat (lambda (s) (plist-get s :contexts))
                                  my/manifolding-atlas-prompt-registry)))
         (specialized (cl-remove-if-not (lambda (c) (member (symbol-name c) tags))
                                        (cl-remove 'file (cl-remove 'heading registered)))))
    (if specialized specialized (list base))))

*** Prompt Lookup
#+begin_src emacs-lisp
(defun my/manifolding-atlas--update-prompt-spec (key)
  "Find prompt spec with KEY matching the current note context."
  (seq-find (lambda (s)
              (and (eq (plist-get s :key) key)
                   (seq-intersection
                    my/manifolding-atlas--current-contexts
                    (plist-get s :contexts) #'eq)))
            my/manifolding-atlas-prompt-registry))

(defun my/manifolding-atlas--clean-display-val (val)
  "Return clean display for property value VAL.
Strips Org link brackets, keeps WARNING visible."
  (cond
   ((null val) "—")
   ((string-empty-p val) "—")
   ((string-match "\\]\\[\\([^]]+\\)\\]" val)
    (match-string-no-properties 1 val))
   (t val)))

(defun my/manifolding-atlas--update-format-prompt (key)
  "Return description string for prompt KEY: toggle state, label, current value."
  (let* ((spec (my/manifolding-atlas--update-prompt-spec key))
         (label (plist-get spec :label))
         (short (if (string-match "^[^:]+: \\(.*\\)" label)
                    (match-string 1 label)
                  label))
         (current (when-let ((fn (plist-get spec :read-current-fn)))
                    (condition-case nil (funcall fn) (error nil))))
         (display (my/manifolding-atlas--clean-display-val
                   (pcase current
                     ((pred listp) (string-join current ", "))
                     ((pred stringp) current)
                     (_ nil))))
         (mark (if (memq key my/manifolding-atlas--selected-keys) "✓" " ")))
    (format "[%s] %-15s %s" mark short display)))

*** Dynamic Suffixes
#+begin_src emacs-lisp
(defvar my/manifolding-atlas--update-key-map nil
  "Alist mapping transient key chars to prompt :key symbols.")

(defun my/manifolding-atlas--toggle-prompt ()
  "Toggle the prompt identified by the key that invoked this command."
  (interactive)
  (condition-case err
      (let* ((key (event-basic-type last-command-event))
             (cell (assq key my/manifolding-atlas--update-key-map))
             (prompt-key (cdr cell)))
        (when prompt-key
          (if (memq prompt-key my/manifolding-atlas--selected-keys)
              (setq my/manifolding-atlas--selected-keys (delq prompt-key my/manifolding-atlas--selected-keys))
            (push prompt-key my/manifolding-atlas--selected-keys))
          (transient-setup 'my/manifolding-atlas--update-properties-menu)))
    (error (message "Toggle error: %s" err))))

(defun my/manifolding-atlas--update-children (&optional _)
  "Build suffix entries from the prompt registry for the current context.
Uses `transient-parse-suffixes' to convert shorthand (KEY DESC CMD)
suffix specs to the internal format expected by `transient--init-suffix.'
DESC is a pre-computed string (computed fresh each time the menu opens)."
  (condition-case err
      (let* ((prompts (seq-filter
                       (lambda (s)
                         (let ((k (plist-get s :key)))
                           (and (not (memq k '(position)))
                                (plist-get s :read-current-fn)
                                (plist-get s :prompt-fn)
                                (seq-intersection
                                 my/manifolding-atlas--current-contexts
                                 (plist-get s :contexts) #'eq))))
                       my/manifolding-atlas-prompt-registry))
             (keys "abcdefghijklmnopqrstuvwxyz")
             (idx 0)
             children)
        (setq my/manifolding-atlas--update-key-map nil)
        (dolist (spec (reverse prompts))
          (when (< idx (length keys))
            (let ((k (aref keys idx))
                  (prompt-key (plist-get spec :key)))
              (push (cons k prompt-key) my/manifolding-atlas--update-key-map)
              (push (list (char-to-string k)
                          (my/manifolding-atlas--update-format-prompt prompt-key)
                          'my/manifolding-atlas--toggle-prompt)
                    children)
              (setq idx (1+ idx)))))
        (transient-parse-suffixes 'my/manifolding-atlas--update-properties-menu
                                  (nreverse children)))
    (error
     (message "Update children error: %s" err)
     nil)))

*** Confirm / Cancel
#+begin_src emacs-lisp
(transient-define-suffix my/manifolding-atlas--update-confirm ()
  "Run all selected prompts and apply changes."
  (interactive)
  (let ((keys my/manifolding-atlas--selected-keys))
    (setq my/manifolding-atlas--selected-keys nil)
    (dolist (key keys)
      (let (spec val frag)
        (setq spec (my/manifolding-atlas--update-prompt-spec key))
        (setq val (condition-case nil
                     (funcall (plist-get spec :prompt-fn))
                   (quit nil)))
        ;; skip the WARNING sentinel — user didn't change anything
        (when (and (stringp val) (string-equal val "WARNING"))
          (setq val nil))
        (setq frag (funcall (plist-get spec :to-plist) val
                            my/manifolding-atlas--current-context))
        (when frag
          (my/manifolding-atlas--apply-fragment frag spec))))
    (when (fboundp 'my/manifolding-atlas--add-live-transcludes)
      (my/manifolding-atlas--add-live-transcludes my/manifolding-atlas--current-note))
    (save-buffer)
    (message "Updated properties on %s" (manifolding-atlas-note-title my/manifolding-atlas--current-note))
    (when (called-interactively-p 'any)
      (my/manifolding-atlas--update-properties-menu))))

(transient-define-suffix my/manifolding-atlas--update-cancel ()
  "Cancel the update."
  (interactive)
  (setq my/manifolding-atlas--selected-keys nil)
  (message "Update cancelled."))
#+end_src
*** Buffer Note Helper
#+begin_src emacs-lisp
(defun my/manifolding-atlas--note-at-point ()
  "Return the manifolding-atlas note for the heading under (or enclosing) point.
Walks up to the enclosing heading if the cursor is in the body/drawer,
so it targets exactly the note you are on."
  (let (id)
    (unless (org-at-heading-p)
      (ignore-errors (org-back-to-heading t)))
    (setq id (org-entry-get nil "ID"))
    (and id (manifolding-atlas-db-get-by-id id))))

(defun my/manifolding-atlas--buffer-note ()
  "Return the manifolding-atlas note for the current buffer, or nil.
Prefers the note under the cursor, falls back to the buffer's file."
  (or (my/manifolding-atlas--note-at-point)
      (when-let ((file (buffer-file-name))
                 (notes (manifolding-atlas-db-query-by-file-path (file-truename file))))
        (let ((pos-point (point)))
          (or (let ((best nil))
                (dolist (n notes best)
                  (let ((npos (manifolding-atlas-note-pos n)))
                    (when (and npos (<= npos pos-point)
                               (or (null best) (> npos (manifolding-atlas-note-pos best))))
                      (setq best n)))))
              (car notes))))))

*** Property Drawer Fallback
Schema validation reads from the description list (meta), but we store
state values in the property drawer.  This advice makes validation fall
back to the property drawer when no description list value exists.

#+begin_src emacs-lisp
(defun my/manifolding-atlas--meta-get-fallback (orig-fun note prop &optional type)
  "Fall back to the property drawer when the description list has no value for PROP.
Skips non-string types to avoid interfering with note/link resolution."
  (or (funcall orig-fun note prop type)
      (when (memq type '(nil string))
        (when-let* ((file (manifolding-atlas-note-path note))
                    (key (if (symbolp prop) (symbol-name prop) prop))
                    (val (if (and (buffer-file-name)
                                 (equal (file-truename (buffer-file-name))
                                        (file-truename file)))
                             ;; current buffer is already the note's file
                             (save-excursion
                               (when-let ((pos (manifolding-atlas-note-pos note)))
                                 (ignore-errors (goto-char pos)))
                               (org-entry-get nil key))
                           ;; visit the note's file
                           (with-current-buffer (find-file-noselect file)
                             (save-excursion
                               (when-let ((pos (manifolding-atlas-note-pos note)))
                                 (ignore-errors (goto-char pos)))
                               (org-entry-get nil key))))))
          (unless (string-empty-p val)
            (list val))))))

(advice-add 'manifolding-atlas-note-meta-get-list :around #'my/manifolding-atlas--meta-get-fallback)

*** Fix Violations Suffix

#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--fix-violations ()
  "Prompt for each missing or WARNING-valued schema field.
Writes the value to the property drawer only; the
`manifolding-atlas-note-meta-get-list' advice makes schema validation fall back
to the property drawer when no description-list value exists."
  (interactive)
  (let* ((prompted 0)
         (note (my/manifolding-atlas--buffer-note))
         all-fields schema-keys field-lookup)
    (unless note
      (user-error "Not in a manifolding-atlas note buffer"))
    ;; Collect all schema fields and build a key→field lookup
    (setq all-fields
          (cl-loop for name in (manifolding-atlas-schema-applicable note)
                   append (manifolding-atlas-schema-fields (manifolding-atlas-schema-get name))))
    (setq field-lookup (make-hash-table :test 'equal))
    (dolist (f all-fields)
      (puthash (plist-get f :key) f field-lookup))
    (setq schema-keys (hash-table-keys field-lookup))
    ;; Fix schema fields (missing or WARNING)
    (dolist (key schema-keys)
      (let* ((val (org-entry-get nil key))
             (field (gethash key field-lookup)))
        (when (or (null val) (and (stringp val) (string-match-p "WARNING" val)))
          (condition-case nil
              (let* ((one-of (plist-get field :one-of))
                     (new-val (if one-of
                                  (completing-read
                                   (format "%s: " key) one-of nil t nil nil)
                                (read-string (format "%s: " key)))))
                (unless (string-empty-p new-val)
                  (org-entry-put nil key new-val)
                  (setq prompted (1+ prompted))))
            (quit nil)))))
    ;; Fix any remaining property-drawer entries with WARNING
    ;; (non-schema properties left from the old sentinel system)
    (dolist (pair (org-entry-properties nil 'standard))
      (let* ((key (car pair))
             (val (cdr pair)))
        (when (and (stringp val) (string-match-p "WARNING" val)
                   (not (member key schema-keys))
                   (not (member key '("ID"))))
          (condition-case nil
              (let ((new-val (read-string (format "%s: " key))))
                (unless (string-empty-p new-val)
                  (org-entry-put nil key new-val)
                  (setq prompted (1+ prompted))))
            (quit nil)))))
    (save-buffer)
    (message "Updated %d properties in drawer." prompted)))
;; Clear stale transient property from old transient-define-suffix;
;; this symbol is called from a hydra, not transient's own machinery.
(put 'my/manifolding-atlas--fix-violations 'transient--suffix nil)

#+begin_src emacs-lisp
(defun my/manifolding-atlas--set-file-property (key value strategy)
  "Set KEY to VALUE in the file-level property drawer using STRATEGY."
  (save-excursion
    (goto-char (point-min))
    (if (re-search-forward "^:PROPERTIES:$" nil t)
        (let ((end (save-excursion
                     (or (re-search-forward "^:END:$" nil t) (point-max)))))
          (goto-char (match-end 0))
          (if (re-search-forward (format "^:%s: ?" (regexp-quote key)) end t)
              (if (string-empty-p value)
                  (delete-region (line-beginning-position) (1+ (line-end-position)))
                (when (eq strategy 'replace)
                  (replace-match (format ":%s: %s" key value) t t)))
            (unless (string-empty-p value)
              (re-search-forward "^:END:$" nil t)
              (beginning-of-line)
              (insert (format ":%s: %s\n" key value)))))
      (unless (string-empty-p value)
        (insert ":PROPERTIES:\n")
        (insert (format ":%s: %s\n" key value))
        (insert ":END:\n")))))

(defun my/manifolding-atlas--apply-fragment (frag spec)
  "Apply FRAG plist from SPEC to the current buffer."
  (let ((strategy (or (plist-get spec :merge-strategy) 'replace))
        (at-heading (org-at-heading-p)))
    (unless at-heading
      (setq at-heading (ignore-errors (org-back-to-heading t) t))
      (unless at-heading
        (setq at-heading (ignore-errors (org-next-visible-heading 1) t))))
    (when (plist-member frag :todo)
      (let ((todo (plist-get frag :todo)))
        (if (and (stringp todo) (not (string-empty-p todo)))
            (progn
              (org-todo todo)
              (org-entry-delete nil "TODO"))
          (save-excursion
            (org-back-to-heading t)
            (org-entry-delete nil "TODO")
            (let ((case-fold-search nil))
              (looking-at org-complex-heading-regexp)
              (replace-match
               (concat (match-string 1) " " (match-string 4)
                       (when (match-string 5)
                         (concat " " (match-string 5))))
               t t))))))
    (when-let ((tags (plist-get frag :tags)))
      (if at-heading
          (pcase strategy
            ('merge (org-set-tags (cl-union (org-get-tags nil t) tags :test #'string=)))
            ('replace (org-set-tags tags)))
        (user-error "Tags on file-level notes not supported yet")))
    (when-let ((props (plist-get frag :properties)))
      (dolist (pair props)
        (if at-heading
            (let ((v (cdr pair)))
              (if (string-empty-p v)
                  (org-entry-delete nil (car pair))
                (pcase strategy
                  ('merge (unless (org-entry-get nil (car pair))
                            (org-entry-put nil (car pair) v)))
                  ('replace (org-entry-put nil (car pair) v)))))
          (my/manifolding-atlas--set-file-property (car pair) (cdr pair) strategy))))
    (when-let ((body (plist-get frag :body)))
      (save-excursion
        (goto-char (point-max))
        (insert "\n" body)))
    (when-let ((fns (plist-get frag :post-apply)))
      (let ((file (buffer-file-name)))
        (dolist (fn fns)
          (funcall fn my/manifolding-atlas--current-note))
        (unless (and file (eq (current-buffer) (get-file-buffer file)))
          (when file (find-file file)))))))

#+begin_src emacs-lisp
(defun my/manifolding-atlas--context-file ()
  "Return the note file for the current context, or nil.
Works from a note buffer and from dired (file under the cursor)."
  (or (buffer-file-name)
      (and (derived-mode-p 'dired-mode)
           (ignore-errors (dired-get-file-for-visit)))))

(defun my/manifolding-atlas--index-ensure-note ()
  "Return the note for the current buffer, indexing the file on demand.
Prefers the note under the cursor; if the DB has no row for the file
(sync lag), calls `manifolding-atlas-db-update-file' and re-queries so the menu
always opens on the note you are actually on.  From dired, resolves
the file under the cursor first."
  (or (my/manifolding-atlas--note-at-point)
      (when-let ((file (my/manifolding-atlas--context-file)))
        (condition-case nil
            (progn
              (manifolding-atlas-db-update-file (file-truename file))
              (or (my/manifolding-atlas--note-at-point)
                  (car (manifolding-atlas-db-query-by-file-path (file-truename file)))))
          (error nil)))))

(defun my/manifolding-atlas-update-properties ()
  "Update properties on an existing manifolding-atlas note.
When called interactively, check for schema violations first and offer
to open the update menu.  When called from MISSING PROMPTS advice,
skip the prompt and open the menu directly."
  (interactive)
  (let (buffer-note note file pos level)
    (setq buffer-note (my/manifolding-atlas--index-ensure-note))
    (setq note (or buffer-note
                   (when (manifolding-atlas-db-query)
                     (manifolding-atlas-select "Update properties for: "))))
    (unless note
      (user-error "No manifolding-atlas note here%s"
                  (if-let ((file (my/manifolding-atlas--context-file)))
                      (format " (%s not indexed)" file)
                    "")))
    (setq file (manifolding-atlas-note-path note))
    (setq pos (manifolding-atlas-note-pos note))
    (setq level (manifolding-atlas-note-level note))
    (unless file
      (user-error "Note has no file path"))
    (setq my/manifolding-atlas--current-note note
          my/manifolding-atlas--current-context (if (and level (> level 0)) 'heading 'file)
          my/manifolding-atlas--current-contexts (my/manifolding-atlas--note-contexts note)
          my/manifolding-atlas--selected-keys nil)
    (find-file file)
    (goto-char (or (and pos (> pos 0) pos) (point-min)))
    (unless (org-at-heading-p)
      (ignore-errors (org-back-to-heading t))
      (unless (org-at-heading-p)
        (ignore-errors (org-next-visible-heading 1))))
    (my/manifolding-atlas--update-properties-menu)))
