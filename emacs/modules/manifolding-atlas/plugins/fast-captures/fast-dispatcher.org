#+title:      Fast Captures Dispatcher
#+priority:   15

Dynamic dispatcher that scans `fast-captures/` for `fast-*.org` files
and presents a hydra menu.  New capture types are picked up automatically.

* Core
#+begin_src emacs-lisp
(defun my/manifolding-atlas-quick-captures ()
  "Choose a fast capture type via hydra menu."
  (interactive)
  (let* ((dir (expand-file-name "modules/manifolding-atlas/plugins/fast-captures/"
                                 user-emacs-directory))
         (files (directory-files dir nil "^fast-.+\\.org\\'" t))
         (bindings (delq nil (mapcar (lambda (f)
                                      (let* ((name (substring f 5 -4))
                                             (fn (intern (concat "my/manifolding-atlas-quick-" name))))
                                        (when (fboundp fn)
                                          (let ((key (char-to-string (aref name 0)))
                                                (label (capitalize name)))
                                            `(,key ,fn ,label :exit t)))))
                                    files))))
    (eval (macroexpand
           `(defhydra my/fast-captures-hydra
              (:title "Fast Captures" :quit-key "q" :separator "─")
               "Actions"
               ,@bindings)))
    (my/fast-captures-hydra/body)))
#+end_src
