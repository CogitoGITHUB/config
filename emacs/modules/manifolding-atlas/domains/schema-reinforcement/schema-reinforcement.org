#+title: Manifolding Atlas Schema Reinforcement
#+priority: 15

Keeps schema enforcement active across the whole vault. Violations land
in the existing =admin/MISSING PROMPTS= file (rebuilt from
=manifolding-atlas-schema-collection-health=) and index-time warnings are
emitted by the vendored validation filter. This plugin adds the missing
triggers: full sweeps after sync/save activity, one spike-safe boot
sweep, instant per-save feedback, and a prompt/template registry audit.

*** Require
#+begin_src emacs-lisp
(require 'manifolding-atlas)
(require 'manifolding-atlas-db)
(require 'manifolding-atlas-schema)
#+end_src

*** State
#+begin_src emacs-lisp
(defvar my/manifolding-atlas-reinforce--timer nil
  "Debounced idle timer handle for scheduled sweeps.")

(defvar my/manifolding-atlas-reinforce--last-sweep nil
  "Float time of the last completed sweep.")

(defvar my/manifolding-atlas-reinforce-idle-delay 15
  "Idle seconds after vault activity before a debounced sweep runs.")

(defvar my/manifolding-atlas-reinforce-boot-delay 30
  "Idle seconds after boot before the one-shot boot sweep.")

(defvar my/manifolding-atlas-reinforce-audit-allowlist
  '("FILE_NAME" "SUBDIR" "POSITION" "TAG" "ALIAS" "THINKING_TYPES"
    "MASTERING" "MIND_MAP" "POS" "ETYMOLOGY" "EXAMPLE" "FREQUENCY"
    "PRONUNCIATION" "REGISTER" "SYNONYMS" "TEMPLATE")
  "Property keys managed outside schemas; excluded from the audit.")
#+end_src

*** Sweep
#+begin_src emacs-lisp
(defun my/manifolding-atlas-reinforce-sweep ()
  "Run a full-vault schema sweep and rebuild MISSING PROMPTS."
  (interactive)
  ;; Sweeps can fire at arbitrary ambient stack depth (idle inside
  ;; recursive-edits etc). Carry an independent deep budget so ambient
  ;; depth is irrelevant.
  (let ((max-lisp-eval-depth 12000)
        (max-specpdl-size 12000))
    (condition-case err
        (progn
          (my/manifolding-atlas-missing--rebuild-from-health)
          (setq my/manifolding-atlas-reinforce--last-sweep (float-time))
          (my/manifolding-atlas-reinforce--audit))
      (error (message "Schema sweep failed: %s" err)))))

(defun my/manifolding-atlas-reinforce--schedule ()
  "Debounce: re-arm the idle sweep timer."
  (when my/manifolding-atlas-reinforce--timer
    (cancel-timer my/manifolding-atlas-reinforce--timer))
  (setq my/manifolding-atlas-reinforce--timer
        (run-with-idle-timer my/manifolding-atlas-reinforce-idle-delay
                             nil
                             #'my/manifolding-atlas-reinforce--sweep-maybe)))

(defun my/manifolding-atlas-reinforce--sweep-maybe ()
  "Run a sweep only when the DB is ready; never reschedules itself."
  (setq my/manifolding-atlas-reinforce--timer nil)
  (let ((ready (condition-case nil
                   (caar (emacsql (manifolding-atlas-db)
                                  [:select [id] :from notes :limit 1]))
                 (error nil))))
    (when ready
      (my/manifolding-atlas-reinforce-sweep))))
#+end_src

*** Save Guard
#+begin_src emacs-lisp
(defvar my/manifolding-atlas-reinforce--guard-timer nil)

(defun my/manifolding-atlas-reinforce--validate-buffer-note ()
  "Validate the current buffer's note from freshly synced DB data.
Violations are intentionally silent here — they surface in
admin/MISSING PROMPTS via the reinforcement sweeps."
  (setq my/manifolding-atlas-reinforce--guard-timer nil)
  (when-let ((id (my/manifolding-atlas-links--current-note-id))
             (note (manifolding-atlas-db-get-by-id id)))
    (manifolding-atlas-schema-note-violations note)))

(defun my/manifolding-atlas-reinforce--save-guard ()
  "Debounce a validation of the just-saved note.
Runs after the idle delay so fswatch has synced the new state into
the DB and we grade fresh data, not pre-save state."
  (when my/manifolding-atlas-reinforce--guard-timer
    (cancel-timer my/manifolding-atlas-reinforce--guard-timer))
  (setq my/manifolding-atlas-reinforce--guard-timer
        (run-with-idle-timer 3 nil
                             #'my/manifolding-atlas-reinforce--validate-buffer-note)))

(defun my/manifolding-atlas-reinforce--on-worker-done (&rest _)
  "React to sync worker completion by scheduling a sweep."
  (my/manifolding-atlas-reinforce--schedule))

(with-eval-after-load 'manifolding-atlas
  (add-hook 'org-capture-after-finalize-hook
            #'my/manifolding-atlas-reinforce--schedule t)
  (add-hook 'manifolding-atlas-db-worker-done-functions
            #'my/manifolding-atlas-reinforce--on-worker-done))
#+end_src

*** Vault Buffer Hook
#+begin_src emacs-lisp
(defun my/manifolding-atlas-reinforce--enable-in-vault ()
  "Enable save guard and debounced sweeps in vault note buffers."
  (when (and buffer-file-name
             (string-prefix-p (expand-file-name
                               (my/manifolding-atlas-root-dir))
                              buffer-file-name))
    (add-hook 'after-save-hook
              #'my/manifolding-atlas-reinforce--save-guard nil t)
    (add-hook 'after-save-hook
              #'my/manifolding-atlas-reinforce--schedule nil t)))

(add-hook 'org-mode-hook #'my/manifolding-atlas-reinforce--enable-in-vault)
#+end_src

*** Registry Audit
#+begin_src emacs-lisp
(defun my/manifolding-atlas-reinforce--schema-field-keys ()
  "Return all property keys covered by registered schemas."
  (let ((keys nil))
    (dolist (name (manifolding-atlas-schema-list))
      (dolist (field (manifolding-atlas-schema-fields
                      (manifolding-atlas-schema-get name)))
        (push (plist-get field :key) keys)))
    (delete-dups keys)))

(defun my/manifolding-atlas-reinforce--audit ()
  "Warn about prompts not covered by any schema and templates whose
generated schema is missing or drifted. Silent when clean."
  (let ((covered (append my/manifolding-atlas-reinforce-audit-allowlist
                         (my/manifolding-atlas-reinforce--schema-field-keys)))
        (uncovered nil)
        (template-issues nil))
    (dolist (spec my/manifolding-atlas-prompt-registry)
      (let* ((name (plist-get spec :name))
             (files (and name
                         (directory-files-recursively
                          (my/manifolding-atlas-schema-dir)
                          (format "^%s\\.org$" name))))
             (file (car files))
             (key (condition-case nil
                      (my/manifolding-atlas-schema--extract-prop-key name)
                    (error nil))))
        (unless (or (memq name '(position subdir file-name tags todo
                                        schedule deadline properties))
                    (and file (string-match-p "/prompts/" file))
                    (null key)
                    (member key covered))
          (push key uncovered))))
    (dolist (name (mapcar #'car (my/manifolding-atlas-schema--template-props)))
      (let* ((schema-name (intern (concat "template-" name)))
             (schema (manifolding-atlas-schema-get schema-name)))
        (if (null schema)
            (push (format "%s: no template schema" name) template-issues)
          (let* ((field-keys (mapcar (lambda (f) (plist-get f :key))
                                     (manifolding-atlas-schema-fields schema)))
                 (expected (cons "TEMPLATE_HASH"
                                 (cdr (assoc name
                                             (my/manifolding-atlas-schema--template-props))))))
            (dolist (want expected)
              (unless (member want field-keys)
                (push (format "%s: schema missing field %s" name want)
                      template-issues)))))))
    (when uncovered
      (lwarn 'manifolding-atlas-reinforce :warning
             "Prompts without schema coverage: %s"
             (string-join uncovered ", ")))
    (when template-issues
      (lwarn 'manifolding-atlas-reinforce :warning
             "Template schema issues: %s"
             (string-join (nreverse template-issues) "; ")))))
#+end_src
