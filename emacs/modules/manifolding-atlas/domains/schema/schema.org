#+title: Schema Validation
#+priority: 15

Defines schemas that mirror the state property prompts so
`manifolding-atlas-schema-collection-health` detects skipped (WARNING-valued) fields
as `missing-required` violations, and `manifolding-atlas-schema-flymake` lints them
during editing.

Fields that allow `"WARNING"` as a value (meaning "prompt skipped at creation
time") use a `:validate` function that flags `"WARNING"` as a violation.
This lets `manifolding-atlas-schema-collection-health` detect skipped prompts while
still matching the property's actual stored value.

*** Dynamic Schema from Prompts & Templates

Schemas are built from the prompt files in `modules/manifolding-atlas/schema/` and
the capture template files in `modules/manifolding-atlas/templates/files/`.  Adding a new
prompt or template file automatically keeps the schema in sync.

#+begin_src emacs-lisp
(defun my/manifolding-atlas-schema--prompt-dir ()
  "Return the manifolding-atlas/schema directory."
  (expand-file-name "manifolding-atlas/schema"
                    (expand-file-name "modules" user-emacs-directory)))

(defun my/manifolding-atlas-schema--template-dir ()
  "Return the capture templates directory."
  (expand-file-name "templates/files"
                    (expand-file-name "manifolding-atlas"
                                      (expand-file-name "modules" user-emacs-directory))))

(defun my/manifolding-atlas-schema--extract-choices (prompt-name)
  "Extract completing-read choices (excl. WARNING sentinel) from prompt file PROMPT-NAME."
  (let* ((dir (my/manifolding-atlas-schema--prompt-dir))
         (files (directory-files-recursively dir (format "^%s\\.org$" prompt-name))))
    (when files
      (with-temp-buffer
        (insert-file-contents (car files))
        (goto-char (point-min))
        (when (re-search-forward "completing-read" nil t)
          (condition-case nil
              (when (re-search-forward "'(\"" nil t)
                (let ((items (save-excursion
                               (backward-char 2)
                               (read (current-buffer)))))
                  (cl-remove-if (lambda (s) (string-match-p "WARNING" s)) items)))
            (error nil)))))))

(defun my/manifolding-atlas-schema--extract-prop-key (prompt-name)
  "Extract the property drawer key from prompt file PROMPT-NAME."
  (let* ((dir (my/manifolding-atlas-schema--prompt-dir))
         (files (directory-files-recursively dir (format "^%s\\.org$" prompt-name))))
    (when files
      (with-temp-buffer
        (insert-file-contents (car files))
        (when (re-search-forward "org-entry-get nil \"\\([A-Z_]+?\\)\"" nil t)
          (match-string 1))))))

(defun my/manifolding-atlas-schema--validate (choices val _note)
  "Validate a field value against CHOICES (dynamic-binding compat).
Extracts label from Org link format if present."
  (if (equal val "WARNING") "missing-required"
    (let ((label (if (string-match "\\[\\[id:\\([^]]+\\)\\]\\[\\([^]]+\\)\\]\\]" val)
                     (match-string 2 val)
                   val)))
      (if (and choices (not (member label choices)))
          (format "invalid-value: %s" label)
        t))))

(defun my/manifolding-atlas-schema--field-from-prompt (prompt-name &optional required)
  "Generate a manifolding-atlas-schema field plist from prompt file PROMPT-NAME.
When REQUIRED is non-nil, the field is mandatory."
  (let* ((key (or (my/manifolding-atlas-schema--extract-prop-key prompt-name)
                  (upcase (replace-regexp-in-string
                           "-" "_" (symbol-name prompt-name)))))
         (choices (my/manifolding-atlas-schema--extract-choices prompt-name)))
    `(:key ,key
      :type string
      ,@(when choices `(:one-of ,choices))
      ,@(when required '(:required t))
      :validate ,(apply-partially #'my/manifolding-atlas-schema--validate choices))))

(defun my/manifolding-atlas-schema--template-prop-keys ()
  "Return list of property keys defined in capture template files."
  (let ((template-dir (my/manifolding-atlas-schema--template-dir))
        keys)
    (when (file-directory-p template-dir)
      (dolist (file (directory-files template-dir t "\\.org\\'"))
        (with-temp-buffer
          (insert-file-contents file)
          (goto-char (point-min))
          (when (re-search-forward "^:PROPERTIES:$" nil t)
            (forward-line)
            (while (not (looking-at "^:END:$"))
              (when (looking-at "^:\\([A-Z_]+?\\):[ \t]*$")
                (push (match-string 1) keys))
              (forward-line))))))
    (delete-dups keys)))

(defun my/manifolding-atlas-schema--template-props ()
  "Return alist of (template-name . (prop-key ...)) from capture template files."
  (let ((template-dir (my/manifolding-atlas-schema--template-dir))
        result)
    (when (file-directory-p template-dir)
      (dolist (file (directory-files template-dir t "\\.org\\'"))
        (let* ((name (file-name-sans-extension (file-name-nondirectory file)))
               keys)
          (with-temp-buffer
            (insert-file-contents file)
            (goto-char (point-min))
            (when (re-search-forward "^:PROPERTIES:$" nil t)
              (forward-line)
              (while (not (looking-at "^:END:$"))
                (when (looking-at "^:\\([A-Z_]+?\\):[ \t]*$")
                  (push (match-string 1) keys))
                (forward-line))))
          (when keys
            (push (cons name (nreverse keys)) result)))))
    (nreverse result)))
#+end_src

*** Base Schema

Applies to every note in the vault.  Fields are derived from all
`state-*` prompts in `modules/manifolding-atlas/schema/general/`.

#+begin_src emacs-lisp
(with-eval-after-load 'manifolding-atlas
  (manifolding-atlas-schema-define
   'base
   :predicate (lambda (_note) t)
   :fields
   (list (my/manifolding-atlas-schema--field-from-prompt 'state-work t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-general t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-intelligence t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-operational t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-validation t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-source t)
         (my/manifolding-atlas-schema--field-from-prompt 'state-dependency)
         (my/manifolding-atlas-schema--field-from-prompt 'state-checklist))))
#+end_src

*** Template Schemas

Each capture template that defines a property drawer (e.g.
`:ROAM_REFS:` in `reference.org`) gets its own schema.
The schema applies only to notes whose `:TEMPLATE:` property matches
the template name.  `TEMPLATE_HASH` is also validated against the
current template file so outdated notes are flagged when the template
changes.

#+begin_src emacs-lisp
(with-eval-after-load 'manifolding-atlas
  (cl-loop for (template-name . keys) in (my/manifolding-atlas-schema--template-props)
           for schema-name = (intern (concat "template-" template-name))
           do (manifolding-atlas-schema-define
               schema-name
               :predicate
               `(lambda (note)
                  (or (member ,template-name
                              (manifolding-atlas-note-meta-get-list note 'TEMPLATE 'string))
                      (equal ,template-name
                             (cdr (assoc "TEMPLATE"
                                         (manifolding-atlas-note-properties note))))))
               :include 'base
               :fields
               `((:key "TEMPLATE_HASH"
                  :type string
                  :required t
                  :validate
                  ,(lambda (val note)
                     (let* ((name (or (car (manifolding-atlas-note-meta-get-list
                                            note 'TEMPLATE 'string))
                                      (cdr (assoc "TEMPLATE"
                                                  (manifolding-atlas-note-properties note)))))
                            (current (if name
                                          (my/manifolding-atlas--capture-file-hash name)
                                        nil)))
                       (cond ((and val current (not (string-equal val current)))
                              "template-outdated")
                             ((null val) "missing-required")
                             (t t)))))
                 ,@(cl-loop for key in keys
                            collect
                            `(:key ,key :type string
                              :required t
                              :validate
                              ,(lambda (val _note)
                                 (cond ((equal val "WARNING") "missing-required")
                                       ((null val) "missing-required")
                                       (t t))))))))))
#+end_src

*** Task Schema

Applies to notes tagged `task`.  Inherits all base fields and adds
TODO and PRIORITY from their prompt files.

#+begin_src emacs-lisp
(with-eval-after-load 'manifolding-atlas
  (manifolding-atlas-schema-define
   'task
   :predicate (lambda (note) (member "task" (manifolding-atlas-note-tags note)))
   :include 'base
   :fields
   (list (my/manifolding-atlas-schema--field-from-prompt 'todo)
         (my/manifolding-atlas-schema--field-from-prompt 'priority))))
#+end_src

*** Index-time Validation Action

Warning — indexed notes that violate an applicable schema emit a
non-blocking lwarn warning naming each violation. Violations are also
tracked via MISSING PROMPTS (`my/manifolding-atlas-missing--rebuild-from-health').

#+begin_src emacs-lisp
(with-eval-after-load 'manifolding-atlas
  (setq manifolding-atlas-db-schema-validation-action 'warning))

*** Migration from WARNING Sentinel

One-time migration: removes all `"WARNING"` property values from every
note in the vault.  Under the old prompt system, skipping a prompt stored
`"WARNING"` as the property value.  The new schema validation detects
these as `missing-required` violations instead, so the sentinel value is
no longer needed — simply having the property with any value satisfies
the schema's requirement to *set* the field.  Deleting the property
leaves the note truly missing that metadata.

#+begin_src emacs-lisp
(defun my/manifolding-atlas-schema-migrate-warnings ()
  "Delete all properties with value \"WARNING\" throughout the vault.

Under the old prompt system, skipping a prompt stored \"WARNING\" as
the property value to track it in MISSING PROMPTS.  Now schema
validation flags notes that are *missing* the metadata entirely as
`missing-required` violations, so the sentinel is no longer needed.
Deleting the WARNING sentinel lets the description-list metadata
(eg - STATE_WORK :: ACTIVE) become the canonical value.

Idempotent and safe to run multiple times.  Prompts for confirmation
before making any changes."
  (interactive)
  (unless (yes-or-no-p
           "Delete all \"WARNING\" property values throughout the vault? ")
    (user-error "Aborted"))
  (let ((count 0)
        (files (make-hash-table :test #'equal)))
    ;; Group notes by file so we only visit each file once
    (dolist (note (manifolding-atlas-db-query))
      (let ((file (manifolding-atlas-note-path note)))
        (when (and file (file-exists-p file))
          (push (list :level (manifolding-atlas-note-level note)
                      :pos (manifolding-atlas-note-pos note))
                (gethash file files)))))
    (maphash
     (lambda (file notes)
       (with-current-buffer (find-file-noselect file)
         (dolist (info notes)
           (let ((level (plist-get info :level))
                 (pos (plist-get info :pos)))
             (when (and level (> level 0) pos)
               (goto-char pos)
               (org-back-to-heading t))
             (dolist (prop '("STATE_WORK" "STATE_GENERAL" "STATE_INTELLIGENCE"
                             "STATE_OPERATIONAL" "STATE_VALIDATION" "STATE_SOURCE"
                             "STATE_DEPENDENCY" "STATE_CHECKLIST" "PRIORITY" "TAG"))
               (let ((val (org-entry-get nil prop)))
                 (when (and val (string-match-p "WARNING" val))
                   (org-entry-delete nil prop)
                   (setq count (1+ count)))))
             (when (and (org-at-heading-p)
                        (equal (org-get-todo-state) "WARNING"))
               (org-todo " ")
               (setq count (1+ count)))))
         (when (buffer-modified-p)
           (save-buffer))
         (kill-buffer)))
     files)
    (message "Migration complete: deleted %d WARNING property values (existing metadata preserved)." count)))
#+end_src
