* Email Prompt
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'contact-email
  :name 'contact-email
  :label "Email"
  :contexts '(contact)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Email: " '("⚠ WARNING") nil nil nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :read-current-fn (lambda () (org-entry-get nil "EMAIL"))
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("EMAIL" . ,val))))))
#+end_src
