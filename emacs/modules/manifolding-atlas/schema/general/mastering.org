#+title:      Mastering Review Prompt
#+priority:   5

* Mastering
Why: Controls whether the note participates in the mastering spaced-repetition review pool. Default is WARNING (not in pool).

** Default (default WARNING)
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-mastering ()
  "Prompt for MASTERING. Returns \"WARNING\" if skipped."
  (let ((val (completing-read
              "MASTERING: "
              '("⚠ WARNING" "REVIEW" "SUSPEND")
              nil t nil nil "⚠ WARNING")))
    (if (string-equal val "⚠ WARNING") "WARNING" val)))
#+end_src

** With initial value
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-mastering-initial (initial)
  "Prompt for MASTERING with INITIAL pre-selected."
  (completing-read
   "MASTERING: "
   '("REVIEW" "SUSPEND")
   nil t nil nil initial))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'mastering
  :name 'mastering
  :label "Mastering: Review"
  :prompt-fn #'my/manifolding-atlas-prompt-mastering
  :contexts '(file heading task)
  :read-current-fn (lambda () (org-entry-get nil "MASTERING"))
  :merge-strategy 'replace
  :to-plist (lambda (val _ctx)
              (cond
               ((or (null val) (string-equal val "WARNING"))
                `(:properties (("MASTERING" . "WARNING"))))
               ((string-equal val "SUSPEND")
                `(:properties (("MASTERING" . "SUSPEND"))
                  :post-apply (,(apply-partially
                                 #'my/manifolding-atlas-mastering--init-schedule t))))
               (t
                `(:properties (("MASTERING" . "REVIEW"))
                  :post-apply (,(apply-partially
                                 #'my/manifolding-atlas-mastering--init-schedule nil)))))))
#+end_src
