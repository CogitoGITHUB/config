#+title:      Priority Prompt
#+priority:   1

* Priority
Why: Highlight important notes in agendas and searches. A = highest, C = lowest.
Blank = no priority cookie.

** Variants
*** Default (blank = none)
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-priority ()
  "Prompt for priority. Returns \"WARNING\" if skipped."
  (let ((p (completing-read "Priority: "
                            '("⚠ WARNING" "A" "B" "C")
                            nil t nil nil "⚠ WARNING")))
    (if (string-equal p "⚠ WARNING") "WARNING" (upcase p))))
#+end_src

*** With fallback (e.g. for tasks)
#+begin_src emacs-lisp
(defun my/manifolding-atlas-prompt-priority-fallback (fallback)
  "Prompt for priority, defaulting to FALLBACK if blank."
  (let ((p (completing-read (format "Priority (default %s): " fallback)
                            '("⚠ WARNING" "A" "B" "C")
                            nil t nil nil fallback)))
    (if (string-equal p "⚠ WARNING") "WARNING" (upcase p))))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/manifolding-atlas-register-prompt
  :key 'priority
  :name 'priority
  :label "Priority"
  :prompt-fn #'my/manifolding-atlas-prompt-priority
  :contexts '(file heading)
  :read-current-fn (lambda () (org-entry-get nil "PRIORITY"))
  :merge-strategy 'replace
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("PRIORITY" . ,val))
                  :post-apply (,(apply-partially #'my/manifolding-atlas--post-sync "PRIORITY" val))))))

(my/manifolding-atlas-register-prompt
  :key 'priority
  :name 'priority
  :label "Priority"
  :prompt-fn (lambda () (my/manifolding-atlas-prompt-priority-fallback "B"))
  :contexts '(task)
  :read-current-fn (lambda () (org-entry-get nil "PRIORITY"))
  :merge-strategy 'replace
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("PRIORITY" . ,val))
                  :post-apply (,(apply-partially #'my/manifolding-atlas--post-sync "PRIORITY" val))))))
#+end_src
