;; Defined in manifolding-atlas.el, which requires this file; declared here to
;; avoid a load-time circular dependency.  Reached at runtime via the
;; `manifolding-atlas-utils-process-notes' macro used by the batch operations.
(declare-function manifolding-atlas-visit "manifolding-atlas")

(defun manifolding-atlas-meta (note-or-id)
  "Get metadata for NOTE-OR-ID.

Return plist (:file :buffer :pl :bound)

For file-level notes (level = 0), metadata is the first
description list in the file.

For heading-level notes (level > 0), metadata is the first
description list within that heading's subtree.

Metadata is defined by a description list, e.g.:

- key1 :: value1
- key2 :: value21
- key2 :: value22
- key3 :: value3

In most cases, it's better to use either `manifolding-atlas-meta-get' to
retrieve a single value for a given key or
`manifolding-atlas-meta-get-list' to retrieve all values for a given
key.

In case you are doing multiple calls to meta API, it's better to
get metadata using this function and use bang version of
functions, e.g. `manifolding-atlas-buffer-meta-get!'."
  (let* ((note (if (stringp note-or-id)
                   (manifolding-atlas-db-get-by-id note-or-id)
                 note-or-id))
         (file (when note (manifolding-atlas-note-path note)))
         (level (when note (manifolding-atlas-note-level note)))
         (pos (when note (manifolding-atlas-note-pos note))))
    (when file
      (manifolding-atlas-utils-with-file file
        (if (and level (> level 0) pos)
            ;; Heading-level note - navigate to position and scope to heading
            (progn
              (goto-char pos)
              (manifolding-atlas-buffer-meta 'heading))
          ;; File-level note - use buffer scope
          (manifolding-atlas-buffer-meta))))))

(defun manifolding-atlas-meta-get-list (note-or-id prop &optional type)
  "Get all values of PROP from NOTE-OR-ID.

Each element value depends on TYPE:

- raw - org element object
- string (default) - an interpreted object (without trailing
  newline)
- number - an interpreted number
- link - path of the link (either ID of the linked note or raw link)
- note - linked `manifolding-atlas-note'
- symbol - an interned symbol."
  (manifolding-atlas-buffer-meta-get-list! (manifolding-atlas-meta note-or-id) prop type))

(defun manifolding-atlas-meta-get (note-or-id prop &optional type)
  "Get value of PROP for NOTE-OR-ID.

Result depends on TYPE:

- raw - org element object
- string (default) - an interpreted object (without trailing
  newline)
- number - an interpreted number
- link - path of the link (either ID of the linked note or raw link)
- note - linked `manifolding-atlas-note'
- symbol - an interned symbol.

If the note contains multiple values for a given PROP, the first
one is returned. In case all values are required, use
`manifolding-atlas-meta-get-list'."
  (manifolding-atlas-buffer-meta-get! (manifolding-atlas-meta note-or-id) prop type))

(defun manifolding-atlas-meta-set (note-or-id prop value &optional append)
  "Set VALUE of PROP for NOTE-OR-ID.

If the VALUE is a list, then each element is inserted
separately.

Please note that all occurrences of PROP are replaced by VALUE.

When PROP is not yet set, VALUE is inserted at the beginning of
the meta, unless the optional argument APPEND is non-nil, in
which case VALUE is added at the end of the meta.

For heading-level notes (level > 0), operates within that
heading's subtree."
  (let* ((note (if (stringp note-or-id)
                   (manifolding-atlas-db-get-by-id note-or-id)
                 note-or-id))
         (file (when note (manifolding-atlas-note-path note)))
         (level (when note (manifolding-atlas-note-level note)))
         (pos (when note (manifolding-atlas-note-pos note))))
    (when file
      (manifolding-atlas-utils-with-file file
        (if (and level (> level 0) pos)
            ;; Heading-level note - navigate to position and scope to heading
            (progn
              (goto-char pos)
              (manifolding-atlas-buffer-meta-set prop value append 'heading))
          ;; File-level note - use buffer scope
          (manifolding-atlas-buffer-meta-set prop value append))))))

(defun manifolding-atlas-meta-set-batch (note-or-id props-alist)
  "Set multiple meta properties for NOTE-OR-ID efficiently.

PROPS-ALIST is an alist where each element is (PROP . VALUE).
VALUE can be a single value or a list of values.

This function is much more efficient than calling `manifolding-atlas-meta-set'
multiple times, as it only parses the file once.

For heading-level notes (level > 0), operates within that
heading's subtree.

Example:
  (manifolding-atlas-meta-set-batch note
    \\='((\"status\" . \"active\")
      (\"priority\" . 1)
      (\"tags\" . (\"a\" \"b\" \"c\"))))"
  (let* ((note (if (stringp note-or-id)
                   (manifolding-atlas-db-get-by-id note-or-id)
                 note-or-id))
         (file (when note (manifolding-atlas-note-path note)))
         (level (when note (manifolding-atlas-note-level note)))
         (pos (when note (manifolding-atlas-note-pos note))))
    (when file
      (manifolding-atlas-utils-with-file file
        (if (and level (> level 0) pos)
            ;; Heading-level note - navigate to position and scope to heading
            (progn
              (goto-char pos)
              (manifolding-atlas-buffer-meta-set-batch props-alist 'heading))
          ;; File-level note - use buffer scope
          (manifolding-atlas-buffer-meta-set-batch props-alist))))))

(defun manifolding-atlas-meta--read-value (type)
  "Read value of TYPE."
  (pcase type
    (`"string" (read-string "String: "))
    (`"number" (read-number "Number: "))
    (`"link" (read-string "URL: "))
    (`"note" (manifolding-atlas-select "Note"))))

(defun manifolding-atlas-meta-add ()
  "Interactive version of `manifolding-atlas-meta-set' for note at point.

When point is in a heading with an ID, operates on that heading.
When point is before the first heading, operates on the file-level note."
  (interactive)
  (if-let* ((id (org-id-get))
            (note (manifolding-atlas-db-get-by-id id)))
      (when-let* ((prop (read-string "Property: "))
                 (value-type (completing-read
                              "Value type: "
                              '(string number link note)
                              nil 'require-match))
                 (value (manifolding-atlas-meta--read-value value-type)))
        (manifolding-atlas-meta-set note prop value 'append))
    (user-error "No note at point")))

(defun manifolding-atlas-meta-add-list ()
  "Interactive version of `manifolding-atlas-meta-set' for note at point.

When point is in a heading with an ID, operates on that heading.
When point is before the first heading, operates on the file-level note."
  (interactive)
  (if-let* ((id (org-id-get))
            (note (manifolding-atlas-db-get-by-id id)))
      (when-let* ((prop (read-string "Property: "))
                 (value-type (completing-read
                              "Value type: "
                              '(string number link note)
                              nil 'require-match))
                 (values
                  (manifolding-atlas-utils-collect-while
                   #'manifolding-atlas-meta--read-value
                   nil
                   value-type)))
        (manifolding-atlas-meta-set note prop values 'append))
    (user-error "No note at point")))

(defun manifolding-atlas-meta-remove (&optional note-or-id prop)
  "Delete values of PROP for NOTE-OR-ID.

For heading-level notes (level > 0), operates within that
heading's subtree.

When called interactively, operates on the note at point:
- If point is in a heading with an ID, operates on that heading
- If point is before the first heading, operates on the file-level note"
  (interactive)
  ;; handle interactive call, e.g. guess the note and read a prop
  (when (called-interactively-p 'any)
    (let ((id (org-id-get)))
      (setq note-or-id (manifolding-atlas-db-get-by-id id))
      (unless note-or-id
        (user-error "No note at point"))
      (when-let*
          ((meta (manifolding-atlas-meta note-or-id))
           (pl (plist-get meta :pl))
           (props
            (seq-map
             (lambda (item)
               (substring-no-properties
                (org-element-interpret-data
                 (org-element-contents
                  (org-element-property :tag item)))))
             (org-element-map pl 'item #'identity))))
        (setq prop (completing-read
                    "Property: "
                    (seq-uniq props))))))

  ;; do the dirty work
  (let* ((note (if (stringp note-or-id)
                   (manifolding-atlas-db-get-by-id note-or-id)
                 note-or-id))
         (file (when note (manifolding-atlas-note-path note)))
         (level (when note (manifolding-atlas-note-level note)))
         (pos (when note (manifolding-atlas-note-pos note))))
    (when file
      (manifolding-atlas-utils-with-file file
        (if (and level (> level 0) pos)
            ;; Heading-level note - navigate to position and scope to heading
            (progn
              (goto-char pos)
              (manifolding-atlas-buffer-meta-remove prop 'heading))
          ;; File-level note - use buffer scope
          (manifolding-atlas-buffer-meta-remove prop))))))

;;; Batch meta operations

(defun manifolding-atlas-meta-batch-set (notes prop value)
  "Set VALUE of PROP for all NOTES.

Uses `manifolding-atlas-utils-process-notes' for efficient batch processing, which
places point at each note.  For a heading-level note the write is scoped
to that heading's subtree, mirroring `manifolding-atlas-meta-set'.

Returns the count of notes processed."
  (let ((count 0))
    (manifolding-atlas-utils-process-notes notes
      (manifolding-atlas-buffer-meta-set
       prop value nil
       (when (and (manifolding-atlas-note-level it) (> (manifolding-atlas-note-level it) 0)) 'heading))
      (setq count (1+ count)))
    count))

(defun manifolding-atlas-meta-batch-remove (notes prop)
  "Remove PROP from all NOTES.

Uses `manifolding-atlas-utils-process-notes' for efficient batch processing, which
places point at each note.  For a heading-level note the removal is
scoped to that heading's subtree, mirroring `manifolding-atlas-meta-remove'.

Returns the count of notes processed."
  (let ((count 0))
    (manifolding-atlas-utils-process-notes notes
      (manifolding-atlas-buffer-meta-remove
       prop
       (when (and (manifolding-atlas-note-level it) (> (manifolding-atlas-note-level it) 0)) 'heading))
      (setq count (1+ count)))
    count))
