;;; Commentary:
;;
;; Surface schema violations as notes are indexed.
;;
;; Files are the source of truth in manifolding-atlas, so a violating note cannot be
;; blocked from being written to disk.  Instead this module reacts when a
;; note enters the database: it validates the note against every
;; applicable schema and, depending on `manifolding-atlas-db-schema-validation-action',
;; either stays silent, emits a non-blocking warning, or keeps the
;; violating note out of the database.
;;
;; It plugs into the indexing path through
;; `manifolding-atlas-db-note-index-filter-functions' and is a no-op unless schemas
;; are registered and one applies to the note being indexed.
;;
;;; Code:

(require 'manifolding-atlas-note)
(require 'manifolding-atlas-schema)
(require 'manifolding-atlas-db-extract)

(defcustom manifolding-atlas-db-schema-validation-action 'warning
  "How to react when an indexed note violates an applicable schema.

The check runs while a note is written to the database (on save,
external sync, or a full scan), and only when schemas are registered and
one applies to the note - otherwise it does nothing.

- `silent'  index the note, say nothing.
- `warning' index the note, emit a non-blocking warning naming the
            violations (the default).
- `error'   keep the violating note OUT of the database (skip indexing
            it) and emit a warning.  The file on disk is untouched, so
            the note simply stays absent from the database until it is
            fixed and re-synced.

Because files are the source of truth, none of these touch the file;
they only control whether the database mirrors a violating note."
  :type '(choice (const :tag "Index silently" silent)
                 (const :tag "Index and warn" warning)
                 (const :tag "Skip and warn" error))
  :group 'manifolding-atlas)

(defun manifolding-atlas-db-schema-validation--format (note violations)
  "Format VIOLATIONS of NOTE into a human-readable warning string."
  (format "Note %S (%s) violates its schema:\n%s"
          (manifolding-atlas-note-title note)
          (manifolding-atlas-note-id note)
          (mapconcat
           (lambda (v)
             (format "  - [%s] %s"
                     (manifolding-atlas-violation-schema v)
                     (manifolding-atlas-violation-message v)))
           violations "\n")))

(defun manifolding-atlas-db-schema-validation--warn (note violations level)
  "Emit a warning at LEVEL describing VIOLATIONS of NOTE."
  (lwarn 'manifolding-atlas-schema level "%s"
         (manifolding-atlas-db-schema-validation--format note violations)))

(defun manifolding-atlas-db-schema-validation--filter (note)
  "Validate NOTE for `manifolding-atlas-db-note-index-filter-functions'.

Return non-nil to allow indexing NOTE and nil to skip it, dispatching on
`manifolding-atlas-db-schema-validation-action'.  Fast-exits when validation is
`silent' or no schemas are registered, and never affects a note that
conforms or that no schema applies to."
  (if (or (eq manifolding-atlas-db-schema-validation-action 'silent)
          (null (manifolding-atlas-schema-list)))
      t
    (let ((violations (manifolding-atlas-schema-note-violations note)))
      (cond
       ((null violations) t)
       ((eq manifolding-atlas-db-schema-validation-action 'error)
        (manifolding-atlas-db-schema-validation--warn note violations :error)
        nil)
       (t
        (manifolding-atlas-db-schema-validation--warn note violations :warning)
        t)))))

(add-hook 'manifolding-atlas-db-note-index-filter-functions
          #'manifolding-atlas-db-schema-validation--filter)

(provide 'manifolding-atlas-db-schema-validation)
