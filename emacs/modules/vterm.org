** Display
#+begin_src emacs-lisp
;; Show vterm as a bottom side window, 30% frame height
(setq vterm-toggle-fullscreen-p nil)

(add-to-list 'display-buffer-alist
             '((lambda (buf _)
                 (let ((b (if (bufferp buf) buf (get-buffer buf))))
                   (and b (with-current-buffer b
                            (or (derived-mode-p 'vterm-mode)
                                (string-prefix-p vterm-buffer-name (buffer-name b)))))))
               (display-buffer-reuse-window display-buffer-in-side-window)
               (side . bottom)
               (reusable-frames . visible)
               (window-height . 0.5)))

** Toggle Command
#+begin_src emacs-lisp
(defun my/vterm-toggle ()
  "Toggle the vterm popup window.
If a vterm window is visible, hide it.  Otherwise show or create one."
  (interactive)
  (let* ((vterm-wins
          (seq-filter
           (lambda (w)
             (with-current-buffer (window-buffer w)
               (derived-mode-p 'vterm-mode)))
           (window-list)))
         (vterm-visible (car vterm-wins)))
    (if vterm-visible
        ;; Already visible — hide it
        (delete-window vterm-visible)
      ;; Not visible — get-or-create a buffer and display it
      (let ((buf (or (seq-find (lambda (b)
                                 (with-current-buffer b
                                   (derived-mode-p 'vterm-mode)))
                               (buffer-list))
                     (generate-new-buffer vterm-buffer-name))))
        (with-current-buffer buf
          (unless (derived-mode-p 'vterm-mode)
            (vterm-mode)))
        (let ((win (display-buffer buf)))
          (when win (select-window win)))))))
#+end_src
