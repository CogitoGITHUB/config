(defun my/git-timemachine-restore ()
  "Restore the file on disk to the currently viewed git-timemachine revision."
  (interactive)
  (let* ((hash (car git-timemachine-revision))
         (file (buffer-file-name)))
    (when (yes-or-no-p (format "Restore file to revision %s? " (substring hash 0 8)))
      (git-timemachine-quit)
      (shell-command (format "git checkout %s -- %s" hash (shell-quote-argument file)))
      (revert-buffer t t)
      (message "Restored to revision %s." (substring hash 0 8)))))

(defun my/save-all ()
  "Save all modified buffers."
  (interactive)
  (let ((modified (seq-some #'buffer-modified-p (buffer-list))))
    (save-some-buffers t)
    (message (if modified "Saved." "Nothing to save."))))
(defun my/open-dashboard ()
  (interactive)
  (switch-to-buffer "*dashboard*"))
(defun my/smart-escape ()
  "Return to the appropriate modal state based on the current buffer."
  (interactive)
  (cond
   ((derived-mode-p 'org-mode)   (modaled-set-state "org"))
   ((derived-mode-p 'vterm-mode) (modaled-set-state "vterm"))
   (t                            (modaled-set-state "normal"))))
(defun my/treemacs-toggle ()
  "Toggle treemacs. If focused, close it. If open, focus it. If closed, open it."
  (interactive)
  (if (treemacs-is-treemacs-window? (selected-window))
      (treemacs-quit)
    (treemacs-select-window)))
(defun my/avy-zap-to-char-inclusive ()
  "Zap from point up to and including an avy-selected character."
  (interactive)
  (let ((start (point)))
    (avy-goto-char-timer)
    (forward-char 1)
    (kill-region start (point))))

      (lambda ()
        (cond
         ((minibufferp) nil)
         ((derived-mode-p 'org-mode)    "org")
         ((derived-mode-p 'dired-mode)  "dired")
         ((derived-mode-p 'vterm-mode)  "vterm")

          ((derived-mode-p 'magit-mode)  nil)
          ((derived-mode-p 'vundo-mode)  nil)

         (t "normal"))))
