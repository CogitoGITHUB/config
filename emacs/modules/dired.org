(defun my/dired-here ()
  "Open dired at the current buffer's directory."
  (interactive)
  (dired (expand-file-name
          (or (and buffer-file-name
                   (file-name-directory buffer-file-name))
              default-directory))))
(defun my/dired-find-file ()
  "Open file at point, guarding against nil (header lines etc.)."
  (interactive)
  (let ((file (ignore-errors (dired-get-filename nil t))))
    (if file
        (find-file file)
      (message "No file on this line"))))
(defun my/dired-avy-open-file ()
  "Select a dired line with avy and open the file directly."
  (interactive)
  (avy-with my/dired-avy-open-file
    (let ((r (avy--line (eq current-prefix-arg 4))))
      (when (and (not (memq r '(t nil))) (eq avy-action #'identity))
        (goto-char r)
        (my/dired-find-file)))))
