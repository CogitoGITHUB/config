** Clock Widget

#+begin_src emacs-lisp
(defvar manifold/clock-timer nil)

(defun manifold/clock-string ()
  (let* ((time (format-time-string "%A %d %B %Y    %H:%M:%S"))
         (width (or (and (get-buffer-window "*dashboard*")
                         (window-width (get-buffer-window "*dashboard*")))
                    80))
         (pad (max 0 (/ (- width (length time)) 2))))
    (concat (make-string pad ?\s) time)))

(defun dashboard-insert-manifold-clock (_list-size)
  (insert "\n")
  (insert (manifold/clock-string))
  (insert "\n\n"))

(add-to-list 'dashboard-item-generators
             '(manifold-clock . dashboard-insert-manifold-clock))

;; Live update every second — only the clock line, no full refresh
(defun manifold/clock-start ()
  (when (get-buffer "*dashboard*")
    (setq manifold/clock-timer
          (run-at-time t 1
            (lambda ()
              (when (get-buffer-window "*dashboard*")
                (with-current-buffer "*dashboard*"
                  (let ((inhibit-read-only t))
                    (save-excursion
                      (goto-char (point-min))
                      (when (re-search-forward
                             (rx (or "Mon" "Tue" "Wed" "Thu" "Fri" "Sat" "Sun")) nil t)
                        (beginning-of-line)
                        (delete-region (point) (line-end-position))
                        (insert (manifold/clock-string))))))))))))

(defun manifold/clock-stop ()
  (when manifold/clock-timer
    (cancel-timer manifold/clock-timer)
    (setq manifold/clock-timer nil)))

(add-hook 'dashboard-mode-hook #'manifold/clock-start)
(add-hook 'kill-buffer-hook
          (lambda ()
            (when (string= (buffer-name) "*dashboard*")
              (manifold/clock-stop))))
#+end_src

(defun manifold/pure-red--dashboard-banner ()
  (with-eval-after-load 'dashboard
    (dolist (face '(dashboard-banner-logo-title dashboard-text-banner))
      (when (facep face)
        (set-face-attribute face nil
                            :foreground "#FFFFFF" :background 'unspecified :weight 'bold :inherit nil)))))
