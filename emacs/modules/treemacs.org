** Core Config
#+begin_src emacs-lisp
(with-eval-after-load 'treemacs
  (setq treemacs-width                        30
        treemacs-show-hidden-files            t
        treemacs-silent-refresh               t
        treemacs-silent-filewatch             t
        treemacs-follow-after-init            t
        treemacs-is-never-other-window        nil
        treemacs-display-in-side-window       t
        treemacs-recenter-after-file-follow   t
        treemacs-recenter-after-tag-follow    t
        treemacs-sorting                      'alphabetic-asc
        treemacs-persist-file                 (expand-file-name ".cache/treemacs-persist"
                                                                user-emacs-directory))
  (treemacs-load-theme "nerd-icons")
  (set-face-attribute 'treemacs-fringe-indicator-face nil
                      :foreground (face-foreground 'default))
  (treemacs-follow-mode t)
  (treemacs-filewatch-mode t)
  (treemacs-fringe-indicator-mode 'always))
#+end_src

** Config
#+begin_src emacs-lisp
;; treemacs-faces.el defines treemacs-fringe-indicator-face with
;; :foreground from (face-background 'cursor nil t).  Ensure cursor
;; has an explicit background so that ever resolves to a valid color.
(unless (face-background 'cursor nil t)
  (set-face-attribute 'cursor nil :background "#808080"))
#+end_src
