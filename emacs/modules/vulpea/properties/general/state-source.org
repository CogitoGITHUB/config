#+title:      State Source Prompt
#+priority:   5

* State Source
Why: Tracks the Epistemic Legitimacy of the content — whether claims are grounded.
Default is UNSOURCED.

** Default (default UNSOURCED)
#+begin_src emacs-lisp
(defun my/vulpea-prompt-state-source ()
  "Prompt for STATE_SOURCE. Returns \"WARNING\" if skipped."
  (let ((val (completing-read
              "STATE_SOURCE: "
              '("⚠ WARNING" "UNSOURCED" "PARTIAL" "SOURCED")
              nil t nil nil "⚠ WARNING")))
    (if (string-equal val "⚠ WARNING") "WARNING" val)))
#+end_src

** With initial value
#+begin_src emacs-lisp
(defun my/vulpea-prompt-state-source-initial (initial)
  "Prompt for STATE_SOURCE with INITIAL pre-selected."
  (completing-read
   "STATE_SOURCE: "
   '("UNSOURCED" "PARTIAL" "SOURCED")
   nil t nil nil initial))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'state-source
  :name 'state-source
  :label "State: Source"
  :prompt-fn #'my/vulpea-prompt-state-source
  :contexts '(file heading task)
  :read-current-fn (lambda () (org-entry-get nil "STATE_SOURCE"))
  :merge-strategy 'replace
   :to-plist (lambda (val _ctx)
               (when val
                 (let ((link (my/vulpea--ensure-ref-note 'state-source val)))
                   `(:properties (("STATE_SOURCE" . ,(or link val)))
                     :post-apply (,(apply-partially #'my/vulpea--post-sync "STATE_SOURCE" (or link val)))))))))
#+end_src
