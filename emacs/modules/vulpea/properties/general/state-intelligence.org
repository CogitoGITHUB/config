#+title:      State Intelligence Prompt
#+priority:   5

* State Intelligence
Why: Tracks the epistemic credibility of the information. Default is UNVERIFIED.

** Default (default UNVERIFIED)
#+begin_src emacs-lisp
(defun my/vulpea-prompt-state-intelligence ()
  "Prompt for STATE_INTELLIGENCE. Returns \"WARNING\" if skipped."
  (let ((val (completing-read
              "STATE_INTELLIGENCE: "
              '("⚠ WARNING" "UNVERIFIED" "CONFIRMED" "CONTESTED" "OBSOLETE")
              nil t nil nil "⚠ WARNING")))
    (if (string-equal val "⚠ WARNING") "WARNING" val)))
#+end_src

** With initial value
#+begin_src emacs-lisp
(defun my/vulpea-prompt-state-intelligence-initial (initial)
  "Prompt for STATE_INTELLIGENCE with INITIAL pre-selected."
  (completing-read
   "STATE_INTELLIGENCE: "
   '("UNVERIFIED" "CONFIRMED" "CONTESTED" "OBSOLETE")
   nil t nil nil initial))
#+end_src

** Registration
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'state-intelligence
  :name 'state-intelligence
  :label "State: Intelligence"
  :prompt-fn #'my/vulpea-prompt-state-intelligence
  :contexts '(file heading task)
  :read-current-fn (lambda () (org-entry-get nil "STATE_INTELLIGENCE"))
  :merge-strategy 'replace
   :to-plist (lambda (val _ctx)
               (when val
                 (let ((link (my/vulpea--ensure-ref-note 'state-intelligence val)))
                   `(:properties (("STATE_INTELLIGENCE" . ,(or link val)))
                     :post-apply (,(apply-partially #'my/vulpea--post-sync "STATE_INTELLIGENCE" (or link val)))))))))
#+end_src
