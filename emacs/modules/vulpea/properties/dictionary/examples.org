* Example Sentence Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'examples
  :name 'examples
  :label "Example"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (read-string "Example sentence: "))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "")))
                `(:properties (("EXAMPLE" . ,val))))))
#+end_src
