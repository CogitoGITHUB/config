* Etymology Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'etymology
  :name 'etymology
  :label "Etymology"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Etymology: "
                                          '("⚠ WARNING" "Latin" "Greek" "Old English" "French"
                                            "German" "Arabic" "Hindi" "Japanese" "Other")
                                          nil t nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("ETYMOLOGY" . ,val))))))
#+end_src
