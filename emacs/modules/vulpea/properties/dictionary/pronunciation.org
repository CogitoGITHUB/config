* Pronunciation Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'pronunciation
  :name 'pronunciation
  :label "Pronunciation"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (read-string "Pronunciation (IPA): "))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "")))
                `(:properties (("PRONUNCIATION" . ,val))))))
#+end_src
