* Pronunciation Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'pronunciation
  :name 'pronunciation
  :label "Pronunciation"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Pronunciation (IPA): " '("⚠ WARNING") nil nil nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("PRONUNCIATION" . ,val))))))
#+end_src
