* Pronunciation Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'pronunciation
  :name 'pronunciation
  :label "Pronunciation"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Pronunciation (IPA): " '("⚠ WARNING") nil nil nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "WARNING")))
                (let ((label val))
                  `(:properties (("PRONUNCIATION" . ,val))
                    :post-apply (,(lambda (note)
                                    (let ((link (my/vulpea--create-scoped-note
                                                  'pronunciation label
                                                  (vulpea-note-id note))))
                                      (when link
                                        (my/vulpea--property-put note "PRONUNCIATION" link))))))))))
#+end_src
