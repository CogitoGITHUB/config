* Frequency Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'frequency
  :name 'frequency
  :label "Frequency"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Frequency: "
                                          '("⚠ WARNING" "common" "frequent" "rare" "obscure")
                                          nil t nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when val
                (let ((link (my/vulpea--ensure-ref-note 'frequency val)))
                  `(:properties (("FREQUENCY" . ,(or link val))))))))
#+end_src
