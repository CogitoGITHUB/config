* Synonyms Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'synonyms
  :name 'synonyms
  :label "Synonyms"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (read-string "Synonyms (comma-separated): "))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "")))
                `(:properties (("SYNONYMS" . ,val))))))
#+end_src
