* Synonyms Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'synonyms
  :name 'synonyms
  :label "Synonyms"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (let ((val (completing-read "Synonyms (comma-separated): " '("⚠ WARNING") nil nil nil nil "⚠ WARNING")))
                 (if (string-equal val "⚠ WARNING") "WARNING" val)))
  :to-plist (lambda (val _ctx)
              (when val
                `(:properties (("SYNONYMS" . ,val))))))
#+end_src
