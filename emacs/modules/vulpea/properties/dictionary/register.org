* Register Prompt
#+begin_src emacs-lisp
(my/vulpea-register-prompt
  :key 'register
  :name 'register
  :label "Register"
  :contexts '(dictionary)
  :prompt-fn (lambda ()
               (completing-read "Register: "
                                '("⚠ WARNING" "formal" "informal" "slang" "technical"
                                  "archaic" "literary" "colloquial")
                                nil t nil nil "⚠ WARNING"))
  :to-plist (lambda (val _ctx)
              (when (and val (not (string-equal val "⚠ WARNING")))
                `(:properties (("REGISTER" . ,val))))))
#+end_src
