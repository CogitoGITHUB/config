#+title:      Subdirectory Prompt
#+priority:   1

* Subdirectory
Why: Organize notes into a folder hierarchy instead of dumping everything flat.
The active directory persists across creations so you can batch-create notes in
one folder, then switch.  Uses Dired-style ~read-directory-name~ to browse.

** Active Directory State
#+begin_src emacs-lisp
(defvar my/vulpea--active-dir nil
  "Persistent active subdirectory, set by `my/vulpea-set-active-dir'.
When non-nil, `my/vulpea-prompt-subdir' uses this as the default instead of
deriving it from the current buffer's file path.  The value is a directory
name string like \"test\" or \"projects/foo\" (no trailing slash).")
#+end_src

** Set Active Directory Command
#+begin_src emacs-lisp
(defun my/vulpea-set-active-dir (dir)
  "Set the persistent active directory for note creation.
Interactively, use Dired-style browsing.
With a prefix argument, clear it."
  (interactive
   (if current-prefix-arg
       (list nil)
     (let* ((root (my/notes-directory))
            (default (if my/vulpea--active-dir
                         (expand-file-name my/vulpea--active-dir root)
                       root))
            (choice (read-directory-name "Active directory: " root default)))
       (list (unless (string-equal choice (file-name-as-directory root))
               (directory-file-name (file-relative-name choice root)))))))
  (if dir
      (setq my/vulpea--active-dir (directory-file-name dir))
    (setq my/vulpea--active-dir nil))
  (message "Active dir: %s" (or my/vulpea--active-dir "(none)")))
#+end_src

** Subdirectory Prompt (Dired)
#+begin_src emacs-lisp
(defun my/vulpea-prompt-subdir ()
  "Prompt for a subdirectory inside `my/notes-directory' using Dired-style browsing.
Defaults to current buffer's location.  Never auto-selects last-used dir."
  (let* ((root (my/notes-directory))
         (default-dir
          (let ((fname (buffer-file-name)))
            (when fname
              (let ((rel (file-relative-name fname root)))
                (when (and (not (file-name-absolute-p rel))
                           (string-match "\\(.+\\)/[^/]+$" rel))
                  (match-string 1 rel))))))
         (default (if default-dir
                      (expand-file-name default-dir root)
                    root))
         (raw (read-directory-name "Subdirectory: " default default))
         (dir (file-name-as-directory (directory-file-name raw))))
    (if (string-equal dir (file-name-as-directory root))
        ""
      (let ((rel (directory-file-name
                  (file-relative-name dir root))))
        (make-directory dir t)
        (file-name-as-directory rel)))))
#+end_src
