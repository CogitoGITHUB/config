#+title: Vulpea Protocols Plugin
#+priority: 15
:PROPERTIES:
:header-args:emacs-lisp: :tangle ../vulpea-protocols.el
:END:

Turns any heading into a protocol step. Steps are assembled into protocol
documents via the BUILD command, which uses avy for multi-select and
writes live transclusion pointers into topic-indexed protocol files.

Protocol steps share a heading's RULE status (dual-mark) — a heading can
be both a rule and a protocol step.

*** Database Table
#+begin_src emacs-lisp
(defun my/vulpea-protocols--ensure-table ()
  "Create the protocols bookkeeping table if it does not exist."
  (emacsql (vulpea-db)
    [:create :table :if :not :exists protocols
     ([(note-id :text :primary-key :not-null)
       (topics :text :not-null)
       (primary-topic :text)
       (protocol-order :integer :not-null)
       (protocol-level :text :not-null)
       (routed-files :text :not-null)]
      (:foreign-key [note-id] :references notes [id]
       :on-delete :cascade))]))
#+end_src

*** DB Extractor
#+begin_src emacs-lisp
(defun my/vulpea-protocols--extract (_ctx note-data)
  "Record protocol bookkeeping for NOTE-DATA if it is a protocol step."
  (let* ((note-id (plist-get note-data :id))
         (props   (plist-get note-data :properties))
         (is-proto (cdr (assoc "PROTOCOL" props))))
    (emacsql (vulpea-db)
      [:delete :from protocols :where (= note-id $s1)]
      note-id)
    (when (equal is-proto "t")
      (let* ((topics-str (or (cdr (assoc "PROTOCOL_TOPICS" props)) ""))
             (topics     (split-string topics-str " " t))
             (primary    (car topics))
             (order      (string-to-number (or (cdr (assoc "PROTOCOL_ORDER" props)) "0")))
             (level      (or (cdr (assoc "PROTOCOL_LEVEL" props)) "L1")))
        (emacsql (vulpea-db)
          [:insert :into protocols :values $v1]
          (vector note-id topics-str primary order level ""))))
    note-data))

(with-eval-after-load 'vulpea
  (my/vulpea-protocols--ensure-table)
  (vulpea-db-register-extractor
   (make-vulpea-extractor
    :name       'protocols
    :version    1
    :priority   89
    :extract-fn #'my/vulpea-protocols--extract)))
#+end_src

*** Routing Engine
**** Resolve destination file for a topic
#+begin_src emacs-lisp
(defun my/vulpea-protocols--default-file-for-topic (topic)
  "Default filename for TOPIC: protocols-<topic>."
  (expand-file-name (format "protocols-%s" topic)
                    my/vulpea-protocols--protocols-directory))

(defun my/vulpea-protocols--resolve-file (topic &optional override)
  "Resolve the destination file for TOPIC.
Priority: OVERRIDE > topic-file-map > default."
  (cond
   (override (expand-file-name override my/vulpea-protocols--protocols-directory))
   ((cdr (assoc topic my/vulpea-protocols-topic-file-map))
    (expand-file-name (cdr (assoc topic my/vulpea-protocols-topic-file-map))
                      my/vulpea-protocols--protocols-directory))
   (t (my/vulpea-protocols--default-file-for-topic topic))))
#+end_src

**** Write a pointer in a target file
#+begin_src emacs-lisp
(defun my/vulpea-protocols--pointer-exists-p (file note-id)
  "Return t if FILE already contains a transclusion pointer to NOTE-ID."
  (and (file-exists-p file)
       (with-temp-buffer
         (insert-file-contents file)
         (goto-char (point-min))
         (re-search-forward (format "\\[\\[id:%s\\]\\[" (regexp-quote note-id)) nil t))))

(defun my/vulpea-protocols--write-pointer (file note-id heading-title level-str)
  "Append a #+transclude: directive for NOTE-ID at LEVEL-STR into FILE, unless present."
  (unless (my/vulpea-protocols--pointer-exists-p file note-id)
    (make-directory (file-name-directory file) t)
    (with-current-buffer (find-file-noselect file)
      (goto-char (point-max))
      (unless (bolp) (insert "\n"))
      (insert (format "#+transclude: [[id:%s][%s]] :level %s\n"
                      note-id heading-title (substring level-str 1)))
      (save-buffer))))
#+end_src

**** Route a single protocol step
#+begin_src emacs-lisp
(defun my/vulpea-protocols--route-note (note)
  "Route NOTE (a vulpea-note already marked PROTOCOL: t) to its protocol file."
  (let* ((props    (vulpea-note-properties note))
         (topics   (split-string (or (cdr (assoc "PROTOCOL_TOPICS" props)) "") " " t))
         (level    (or (cdr (assoc "PROTOCOL_LEVEL" props)) "L1"))
         (title    (vulpea-note-title note))
         (note-id  (vulpea-note-id note))
         (routed   nil))
    (if (null topics)
        (let ((file (expand-file-name "protocols-universal"
                                      my/vulpea-protocols--protocols-directory)))
          (my/vulpea-protocols--write-pointer file note-id title level)
          (push file routed))
      (dolist (topic topics)
        (let ((file (my/vulpea-protocols--resolve-file topic)))
          (my/vulpea-protocols--write-pointer file note-id title level)
          (push file routed))))
    (emacsql (vulpea-db)
      [:update protocols :set [(= routed-files $s1)] :where (= note-id $s2)]
      (mapconcat #'identity routed ",")
      note-id)))
#+end_src

**** Mark current heading as a protocol step
#+begin_src emacs-lisp
(defun my/vulpea-protocols-mark-as-step ()
  "Mark the heading at point as a protocol step.
Preserves existing :RULE: t if set (dual-mark)."
  (interactive)
  (unless (org-at-heading-p) (org-back-to-heading t))
  (org-id-get-create)
  (let* ((already-proto (equal (org-entry-get nil "PROTOCOL") "t"))
         (current-topics (org-entry-get nil "PROTOCOL_TOPICS"))
         (topics-str (my/vulpea-protocols--prompt-topics current-topics))
         (topics (split-string topics-str " " t))
         (order (read-number "Order: " 1))
         (level (completing-read "Level: " '("L1" "L2" "L3" "L4" "L5"
                                             "L6" "L7" "L8" "L9" "L10")
                                 nil t "L1")))
    (org-entry-put nil "PROTOCOL" "t")
    (if (string-empty-p topics-str)
        (org-entry-delete nil "PROTOCOL_TOPICS")
      (org-entry-put nil "PROTOCOL_TOPICS" topics-str))
    (org-entry-put nil "PROTOCOL_ORDER" (number-to-string order))
    (org-entry-put nil "PROTOCOL_LEVEL" level)
    (save-buffer)
    (when-let ((note (my/vulpea--buffer-note)))
      (my/vulpea-protocols--route-note note))
    (message "%s as protocol step." (if already-proto "Updated" "Marked"))))
#+end_src

#+begin_src emacs-lisp
(defun my/vulpea-protocols-unmark-step ()
  "Remove PROTOCOL status from the heading at point.
Only strips protocol-specific properties; preserves :RULE: t if set."
  (interactive)
  (unless (org-at-heading-p) (org-back-to-heading t))
  (let* ((note-id (org-id-get))
         (routed (caar (emacsql (vulpea-db)
                        [:select [routed-files] :from protocols
                         :where (= note-id $s1)]
                        note-id))))
    (dolist (key '("PROTOCOL" "PROTOCOL_TOPICS" "PROTOCOL_ORDER" "PROTOCOL_LEVEL"))
      (org-entry-delete nil key))
    (save-buffer)
    (when routed
      (message "Unmarked. Pointers in %s are orphaned — see broken-links."
               routed))))
#+end_src

#+begin_src emacs-lisp
(defun my/vulpea-protocols-goto-routed-file ()
  "Open the protocol file the current heading was routed to."
  (interactive)
  (unless (org-at-heading-p) (org-back-to-heading t))
  (let* ((note-id (org-id-get))
         (routed (caar (emacsql (vulpea-db)
                        [:select [routed-files] :from protocols
                         :where (= note-id $s1)]
                        note-id))))
    (unless routed
      (user-error "Not a protocol step heading"))
    (let ((files (split-string routed "," t)))
      (if (cdr files)
          (find-file (completing-read "Open routed file: " files nil t))
        (find-file (car files))))))

(defun my/vulpea-find-protocols ()
  "Browse and open vulpea notes that are marked as protocol steps."
  (interactive)
  (let ((ids (mapcar #'car (emacsql (vulpea-db)
                         [:select [note-id] :from protocols]))))
    (vulpea-find :filter-fn (lambda (note)
                              (member (vulpea-note-id note) ids)))))
#+end_src

*** Build — assemble a protocol from multiple headings
#+begin_src emacs-lisp
(defun my/vulpea-protocols--collect-headings-avy ()
  "Select headings via avy-goto-line. C-g to finish."
  (let ((collected nil))
    (while (condition-case nil
               (progn
                 (avy-goto-line)
                 (unless (org-at-heading-p)
                   (user-error "Not a heading"))
                 (org-id-get-create)
                 (push (list (org-id-get)
                             (org-get-heading t t t t))
                       collected)
                 (message "Added: %s.  C-g to finish" (caar collected))
                 t)
             (quit nil)))
    (nreverse collected)))

(defun my/vulpea-protocols-build ()
  "Assemble a new protocol document from multiple source headings.
Creates a parent heading in the protocol file and writes transclusion
pointers for each selected step."
  (interactive)
  (let* ((title (read-string "Protocol title: "))
         (topics-str (my/vulpea-protocols--prompt-topics))
         (topics (and (not (string-empty-p topics-str))
                      (split-string topics-str " " t)))
         (primary (car topics))
         (file (if primary
                   (my/vulpea-protocols--resolve-file primary)
                 (expand-file-name "protocols-universal"
                                   my/vulpea-protocols--protocols-directory))))
    (make-directory (file-name-directory file) t)
    (message "Pick headings to add to protocol.  C-g when done.")
    (let* ((collected (my/vulpea-protocols--collect-headings-avy))
           (count (length collected))
           (order-counter 1))
      (unless (> count 0)
        (user-error "No headings selected"))
      (let ((parent-id nil))
        ;; Create parent heading in protocol file
        (with-current-buffer (find-file-noselect file)
          (goto-char (point-max))
          (unless (bolp) (insert "\n"))
          (insert "* TODO " title "\n")
          (org-set-property "ID" (org-id-uuid))
          (org-set-property "PROTOCOL" "t")
          (org-set-property "PROTOCOL_ORDER" "0")
          (org-set-property "PROTOCOL_LEVEL" "L1")
          (when topics-str
            (org-set-property "PROTOCOL_TOPICS" topics-str))
          (setq parent-id (org-id-get))
          (save-buffer))
        ;; Process each collected heading
        (dolist (step collected)
          (let* ((id (nth 0 step))
                 (heading-title (nth 1 step))
                 (order order-counter)
                 (level (completing-read
                         (format "Level for \"%s\" (L1-L10): " heading-title)
                         '("L1" "L2" "L3" "L4" "L5" "L6" "L7" "L8" "L9" "L10")
                         nil t "L1")))
            ;; Add properties to source heading
            (when-let ((marker (org-id-find id t)))
              (with-current-buffer (marker-buffer marker)
                (goto-char marker)
                (org-entry-put nil "PROTOCOL" "t")
                (org-entry-put nil "PROTOCOL_ORDER" (number-to-string order))
                (org-entry-put nil "PROTOCOL_LEVEL" level)
                (when topics-str
                  (org-entry-put nil "PROTOCOL_TOPICS" topics-str))
                (save-buffer)))
            ;; Write transclusion pointer under parent heading in protocol file
            (with-current-buffer (find-file-noselect file)
              (goto-char (point-min))
              (when (re-search-forward
                     (format ":ID:\\s-+%s" (regexp-quote parent-id))
                     nil t)
                (org-end-of-subtree t t)
                (unless (bolp) (insert "\n"))
                (insert (format "#+transclude: [[id:%s][%s]] :level %s\n"
                                id heading-title (substring level 1)))
                (save-buffer)))
            (setq order-counter (1+ order-counter))))
        (message "Built protocol \"%s\" with %d step(s)." title count)))))
#+end_src

*** Reset — batch TODO flip
#+begin_src emacs-lisp
(defun my/vulpea-protocols-reset ()
  "Set all protocol steps to TODO."
  (interactive)
  (let ((ids (mapcar #'car (emacsql (vulpea-db)
                        [:select [note-id] :from protocols])))
        (count 0))
    (dolist (id ids)
      (when-let ((marker (org-id-find id t)))
        (with-current-buffer (marker-buffer marker)
          (goto-char marker)
          (org-todo "TODO"))
        (set-marker marker nil)
        (setq count (1+ count))))
    (message "Reset %d protocol steps to TODO." count)))
#+end_src
