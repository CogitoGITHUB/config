#+title:      Vulpea — Fast Journal
#+priority:   15

Fast journal entry capture. Creates a timestamped heading in Journal.org.

* Core
#+begin_src emacs-lisp
(defun my/vulpea-quick-journal--date ()
  (format-time-string "%Y-%m-%d" (org-read-date nil t nil "Journal date")))

(defun my/vulpea-quick-journal--path (date)
  (let* ((parsed (date-to-time date))
         (year (format-time-string "%Y" parsed))
         (quarter (format "Q%d" (1+ (/ (1- (string-to-number (substring date 5 7))) 3))))
         (month (format-time-string "%Y-%m" parsed))
         (week (format-time-string "W%V" parsed))
         (dir (expand-file-name
               (format "Journal/%s/%s/%s/%s/" year quarter month week)
               (my/notes-directory)))
         (file (expand-file-name date dir)))
    `(:dir ,dir :file ,file :year ,year
           :quarter ,quarter :month ,month :week ,week :date ,date)))

(defun my/vulpea-quick-journal ()
  (interactive)
  (let* ((date (my/vulpea-quick-journal--date))
         (info (my/vulpea-quick-journal--path date))
         (dir (plist-get info :dir))
         (file (plist-get info :file))
         (ts (format-time-string "[%Y-%m-%d %a %H:%M]"))
         day-note)
    (make-directory dir t)
    (unless (file-exists-p file)
      (let* ((props `(("YEAR" . ,(plist-get info :year))
                      ("QUARTER" . ,(plist-get info :quarter))
                      ("MONTH" . ,(plist-get info :month))
                      ("WEEK" . ,(plist-get info :week))
                      ("DATE" . ,date)))
             (vulpea-create-default-function (lambda (_) '())))
        (let ((my/vulpea--skip-capture t))
          (vulpea-create date file
                         :properties props
                         :body ""))))
    (setq day-note (car (vulpea-db-query-by-file-path file)))
    (unless day-note
      (vulpea-db-update-file file)
      (setq day-note (car (vulpea-db-query-by-file-path file))))
    (unless day-note
      (error "Journal note not found for %s" date))
    (vulpea-create ts nil :parent day-note)
    (setq my/vulpea--skip-open-prompt t)
    (let ((org-capture-templates
           (append org-capture-templates
                   (list (list "j" "Journal" 'plain
                               `(file+function ,file
                                 (lambda ()
                                   (with-current-buffer
                                       (find-file-noselect ,file)
                                     (goto-char (point-max)))))
                               "*** %?" :kill-buffer t :unnarrowed t)))))
      (org-capture nil "j")
      (set-buffer-modified-p nil)
      (modaled-set-state "insert"))))
#+end_src
