#+title: Manifold Mind Map Plugin
#+priority: 15

Controller layer for the vault's visual mind map.

Model:
- Files → parent-child tree (the skeleton)
- Headings → siblings within their file (the flesh on each bone)
- Group → one file + all its heading notes = one visual card

Edges (three distinct layers):
- structural — file parent-child (drives layout, deterministic)
- semantic — id: links (intellectual connections, overlay)
- presentational — transclusion links (content embedding, overlay)

Ordering: ORDER_KEY real float per node → O(1) insert-between, zero cascade.
=mm/repack= normalises floats → clean integers when gaps < mm/epsilon.

Scale: All queries hit =mm_nodes= via indexed SQL. No Elisp-side sorting.
ON DELETE CASCADE auto-removes map entries when notes are deleted.

Status (=MM_STATUS= property): =active= | =deprecated= | =archived=.

JSON export writes =manifold-map.json= — API contract for a future UI.

*** Schema & Extractor
Creates the =mm_nodes= table in the vulpea DB and registers an extractor
that auto-adds each synced note as a node on first insert.

#+begin_src emacs-lisp
(defconst mm/epsilon 1e-10
  "Minimum ORDER_KEY gap before mm/repack is triggered automatically.")

(defun mm/ensure-table ()
  "Create mm_nodes table and indexes in vulpea DB if they do not exist."
  (emacsql (vulpea-db)
           [:create :table :if :not :exists mm-nodes
            ([(note-id :text :primary-key :not-null)
              (parent-id :text)
              (order-key :real :not-null)
              (node-type :text :not-null)
              (depth :integer :not-null)]
             (:foreign-key [note-id] :references notes [id]
              :on-delete :cascade))])
  (emacsql (vulpea-db)
           [:create :index :if :not :exists mm-idx-parent
            :on mm-nodes [parent-id order-key]])
  (emacsql (vulpea-db)
           [:create :index :if :not :exists mm-idx-depth
            :on mm-nodes [depth]]))

(defun mm/ensure-and-register ()
  "Ensure schema exists and register mm extractor with vulpea."
  (mm/ensure-table)
  (vulpea-db-register-extractor
   (make-vulpea-extractor
    :name       'mm-nodes
    :version    1
    :priority   70
    :extract-fn #'mm/extract)))

(with-eval-after-load 'vulpea
  (mm/ensure-and-register))

(defun mm/extract (_ctx note-data)
  "Vulpea extractor: register note in mm_nodes on DB sync.
File notes (level 0): parent-id nil — user assigns position explicitly.
Heading notes (level > 0): parent-id auto-set to the file's root note.
Never overwrites existing rows — user-set positions are preserved."
  (let* ((note-id   (plist-get note-data :id))
         (level     (or (plist-get note-data :level) 0))
         (file-path (plist-get note-data :path))
         (node-type (if (> level 0) "heading" "file"))
         (existing  (caar (emacsql (vulpea-db)
                                   [:select [note-id] :from mm-nodes
                                    :where (= note-id $s1)]
                                   note-id))))
    (unless existing
      (let* ((parent-id
              (when (> level 0)
                (caar (emacsql (vulpea-db)
                               [:select [id] :from notes
                                :where (and (= path $s1)
                                            (= level 0))]
                               file-path))))
             (depth
              (if parent-id
                  (1+ (or (caar (emacsql (vulpea-db)
                                         [:select [depth] :from mm-nodes
                                          :where (= note-id $s1)]
                                         parent-id))
                          0))
                0))
             (order-key
              (if (> level 0)
                  (float (or (plist-get note-data :pos) 0))
                0.0)))
        (emacsql (vulpea-db)
                 [:insert :or :ignore :into mm-nodes
                  :values $v1]
                 (vector note-id parent-id order-key node-type depth))))
    note-data))
#+end_src

*** Core Queries
All return raw EMACSQL rows. Callers lift into =vulpea-note= objects as needed.

#+begin_src emacs-lisp
(defun mm/node-at (note-id)
  "Return mm row (note-id parent-id order-key node-type depth) or nil."
  (car (emacsql (vulpea-db)
                [:select [note-id parent-id order-key node-type depth]
                 :from mm-nodes
                 :where (= note-id $s1)]
                note-id)))

(defun mm/roots ()
  "Return list of (note-id order-key) for all root file nodes."
  (emacsql (vulpea-db)
           [:select [note-id order-key]
            :from mm-nodes
            :where (and (is parent-id nil)
                        (= node-type "file"))
            :order-by [(asc order-key)]]))

(defun mm/children-of (note-id)
  "Return list of (note-id order-key) for direct file children of NOTE-ID."
  (emacsql (vulpea-db)
           [:select [note-id order-key]
            :from mm-nodes
            :where (and (= parent-id $s1)
                        (= node-type "file"))
            :order-by [(asc order-key)]]
           note-id))

(defun mm/headings-of (note-id)
  "Return list of (note-id order-key) for heading siblings in file NOTE-ID."
  (emacsql (vulpea-db)
           [:select [note-id order-key]
            :from mm-nodes
            :where (and (= parent-id $s1)
                        (= node-type "heading"))
            :order-by [(asc order-key)]]
           note-id))

(defun mm/siblings-of (note-id)
  "Return ordered (note-id order-key) list of siblings of NOTE-ID.
Excludes NOTE-ID itself."
  (when-let* ((row       (mm/node-at note-id))
              (node-type (nth 3 row)))
    (let ((parent-id (nth 1 row)))
      (if parent-id
          (emacsql (vulpea-db)
                   [:select [note-id order-key]
                    :from mm-nodes
                    :where (and (= parent-id $s1)
                                (= node-type $s2)
                                (!= note-id $s3))
                    :order-by [(asc order-key)]]
                   parent-id node-type note-id)
        (emacsql (vulpea-db)
                 [:select [note-id order-key]
                  :from mm-nodes
                  :where (and (is parent-id nil)
                              (= node-type $s1)
                              (!= note-id $s2))
                  :order-by [(asc order-key)]]
                 node-type note-id)))))

(defun mm/parent-of (note-id)
  "Return parent note-id of NOTE-ID, or nil if root."
  (nth 1 (mm/node-at note-id)))

(defun mm/at-depth (depth)
  "Return list of (note-id) for all file nodes at DEPTH."
  (emacsql (vulpea-db)
           [:select [note-id]
            :from mm-nodes
            :where (and (= depth $s1)
                        (= node-type "file"))
            :order-by [(asc order-key)]]
           depth))

(defun mm/subtree-of (note-id)
  "Return all descendant file note-ids under NOTE-ID. BFS — no recursion limit."
  (let (result (queue (list note-id)))
    (while queue
      (let* ((cur      (pop queue))
             (children (mapcar #'car (mm/children-of cur))))
        (setq result (append result children)
              queue  (append queue children))))
    result))

(defun mm/group-of (note-id)
  "Return Group plist for NOTE-ID: (:file vulpea-note :headings list-of-notes)."
  (list :file     (vulpea-db-get-by-id note-id)
        :headings (mapcar (lambda (row) (vulpea-db-get-by-id (car row)))
                          (mm/headings-of note-id))))
#+end_src

*** Edge Queries
Three distinct connection layers queried independently.

#+begin_src emacs-lisp
(defun mm/semantic-links-from (note-id)
  "Return list of dest note-ids linked from NOTE-ID via id: links."
  (mapcar (lambda (link) (plist-get link :dest))
          (vulpea-db-query-links-from note-id)))

(defun mm/presentational-links-from (note-id)
  "Return list of dest note-ids transcluded from NOTE-ID."
  (condition-case nil
      (mapcar #'car
              (emacsql (vulpea-db)
                       [:select [dest] :from transclude-links
                        :where (= source $s1)]
                       note-id))
    (error nil)))

(defun mm/all-edges ()
  "Return all three edge types as a plist:
:structural, :semantic, :presentational."
  (let* ((structural
          (mapcar (lambda (row) (cons (car row) (cadr row)))
                  (emacsql (vulpea-db)
                           [:select [note-id parent-id]
                            :from mm-nodes
                            :where (and (is-not parent-id nil)
                                        (= node-type "file"))])))
         (semantic
          (let (edges)
            (dolist (row (emacsql (vulpea-db)
                                  [:select [note-id] :from mm-nodes]))
              (dolist (dest (mm/semantic-links-from (car row)))
                (push (cons (car row) dest) edges)))
            (nreverse edges)))
         (presentational
          (condition-case nil
              (mapcar (lambda (row) (cons (car row) (cadr row)))
                      (emacsql (vulpea-db)
                               [:select [source dest]
                                :from transclude-links]))
            (error nil))))
    (list :structural    structural
          :semantic      semantic
          :presentational presentational)))
#+end_src

*** Position Management
Reparenting and order-key manipulation.

#+begin_src emacs-lisp
(defun mm/set-parent (note-id parent-id)
  "Set parent of NOTE-ID to PARENT-ID and recompute depth for subtree."
  (let* ((parent-row   (mm/node-at parent-id))
         (parent-depth (if parent-row (nth 4 parent-row) 0)))
    (emacsql (vulpea-db)
             [:update mm-nodes
              :set [(= parent-id $s1) (= depth $s2)]
              :where (= note-id $s3)]
             parent-id (1+ parent-depth) note-id)
    (mm/recompute-depths note-id (+ 2 parent-depth))))

(defun mm/make-root (note-id)
  "Remove parent of NOTE-ID, making it a root file node."
  (emacsql (vulpea-db)
           [:update mm-nodes
            :set [(= parent-id nil) (= depth 0)]
            :where (= note-id $s1)]
           note-id)
  (mm/recompute-depths note-id 1))

(defun mm/recompute-depths (note-id child-depth)
  "Recursively recompute depths for all file children of NOTE-ID."
  (dolist (child-id (mapcar #'car (mm/children-of note-id)))
    (emacsql (vulpea-db)
             [:update mm-nodes
              :set [(= depth $s1)]
              :where (= note-id $s2)]
             child-depth child-id)
    (mm/recompute-depths child-id (1+ child-depth))))

(defun mm/set-order-key (note-id key)
  "Set ORDER_KEY of NOTE-ID to KEY (a float)."
  (emacsql (vulpea-db)
           [:update mm-nodes
            :set [(= order-key $s1)]
            :where (= note-id $s2)]
           key note-id))

(defun mm/order-key-after (note-id)
  "Return an ORDER_KEY placing a new sibling immediately after NOTE-ID.
Auto-triggers mm/repack if the gap to the next sibling < mm/epsilon."
  (let* ((row       (mm/node-at note-id))
         (parent-id (nth 1 row))
         (node-type (nth 3 row))
         (cur-key   (nth 2 row))
         (next-key  (caar (emacsql (vulpea-db)
                                   [:select [(funcall min order-key)]
                                    :from mm-nodes
                                    :where (and (= parent-id $s1)
                                                (= node-type $s2)
                                                (> order-key $s3))]
                                   parent-id node-type cur-key))))
    (if (null next-key)
        (+ cur-key 1.0)
      (let ((gap (- next-key cur-key)))
        (when (< gap mm/epsilon)
          (mm/repack parent-id node-type)
          (setq cur-key  (nth 2 (mm/node-at note-id))
                next-key (caar (emacsql (vulpea-db)
                                        [:select [(funcall min order-key)]
                                         :from mm-nodes
                                         :where (and (= parent-id $s1)
                                                     (= node-type $s2)
                                                     (> order-key $s3))]
                                        parent-id node-type cur-key))))
        (/ (+ cur-key next-key) 2.0)))))

(defun mm/order-key-between (note-id-a note-id-b)
  "Return an ORDER_KEY placing a new node between NOTE-ID-A and NOTE-ID-B.
Auto-triggers mm/repack if gap < mm/epsilon."
  (let* ((key-a (nth 2 (mm/node-at note-id-a)))
         (key-b (nth 2 (mm/node-at note-id-b)))
         (lo    (min key-a key-b))
         (hi    (max key-a key-b)))
    (when (< (- hi lo) mm/epsilon)
      (let* ((row-a     (mm/node-at note-id-a))
             (parent-id (nth 1 row-a))
             (node-type (nth 3 row-a)))
        (mm/repack parent-id node-type)
        (setq key-a (nth 2 (mm/node-at note-id-a))
              key-b (nth 2 (mm/node-at note-id-b))
              lo    (min key-a key-b)
              hi    (max key-a key-b))))
    (/ (+ lo hi) 2.0)))

(defun mm/reparent (note-id new-parent-id &optional after-id)
  "Move NOTE-ID to be a child of NEW-PARENT-ID.
If AFTER-ID is given, place immediately after that sibling.
Otherwise append at end with order-key = max + 1.0."
  (let ((order-key
         (if after-id
             (mm/order-key-after after-id)
           (let ((max-key (caar (emacsql (vulpea-db)
                                         [:select [(funcall max order-key)]
                                          :from mm-nodes
                                          :where (= parent-id $s1)]
                                         new-parent-id))))
             (+ (or max-key 0.0) 1.0)))))
    (mm/set-parent note-id new-parent-id)
    (mm/set-order-key note-id order-key)))
#+end_src

*** Repack
Normalises floats to clean integers when gaps approach =mm/epsilon=.

#+begin_src emacs-lisp
(defun mm/repack (parent-id node-type)
  "Normalize ORDER_KEYs for siblings under PARENT-ID of NODE-TYPE.
Assigns 1.0, 2.0, 3.0, ... in current sorted order."
  (let ((siblings (emacsql (vulpea-db)
                           [:select [note-id]
                            :from mm-nodes
                            :where (and (= parent-id $s1)
                                        (= node-type $s2))
                            :order-by [(asc order-key)]]
                           parent-id node-type))
        (i 1.0))
    (dolist (row siblings)
      (emacsql (vulpea-db)
               [:update mm-nodes
                :set [(= order-key $s1)]
                :where (= note-id $s2)]
               i (car row))
      (setq i (+ i 1.0)))
    (message "mm: repacked %d siblings" (round (1- i)))))

(defun mm/repack-all ()
  "Repack all sibling groups in the vault."
  (interactive)
  (let ((groups (emacsql (vulpea-db)
                         [:select :distinct [parent-id node-type]
                          :from mm-nodes
                          :where (is-not parent-id nil)])))
    (dolist (row groups)
      (mm/repack (car row) (cadr row))))
  (let ((roots (emacsql (vulpea-db)
                        [:select [note-id]
                         :from mm-nodes
                         :where (and (is parent-id nil)
                                     (= node-type "file"))
                         :order-by [(asc order-key)]]))
        (i 1.0))
    (dolist (row roots)
      (emacsql (vulpea-db)
               [:update mm-nodes
                :set [(= order-key $s1)]
                :where (= note-id $s2)]
               i (car row))
      (setq i (+ i 1.0))))
  (message "mm: repack-all complete"))
#+end_src

*** Interactive Assignment
Commands to position notes in the tree interactively.

#+begin_src emacs-lisp
(defun mm/assign-parent ()
  "Set the parent of the current note in the mind map."
  (interactive)
  (let* ((note      (my/vulpea--buffer-note))
         (note-id   (vulpea-note-id note))
         (parent    (vulpea-select "Parent: " :require-match t))
         (parent-id (vulpea-note-id parent)))
    (when (string= note-id parent-id)
      (user-error "A note cannot be its own parent"))
    (mm/reparent note-id parent-id)
    (message "✓ %s → parent: %s"
             (vulpea-note-title note)
             (vulpea-note-title parent))))

(defun mm/assign-after ()
  "Place the current note immediately after a chosen sibling."
  (interactive)
  (let* ((note      (my/vulpea--buffer-note))
         (note-id   (vulpea-note-id note))
         (sibling   (vulpea-select "Place after: " :require-match t))
         (sib-id    (vulpea-note-id sibling))
         (parent-id (mm/parent-of sib-id))
         (new-key   (mm/order-key-after sib-id)))
    (if parent-id
        (mm/set-parent note-id parent-id)
      (mm/make-root note-id))
    (mm/set-order-key note-id new-key)
    (message "✓ Placed after: %s" (vulpea-note-title sibling))))

(defun mm/assign-between ()
  "Place the current note between two chosen siblings."
  (interactive)
  (let* ((note    (my/vulpea--buffer-note))
         (note-id (vulpea-note-id note))
         (before  (vulpea-select "After this: "  :require-match t))
         (after   (vulpea-select "Before this: " :require-match t))
         (new-key (mm/order-key-between
                   (vulpea-note-id before)
                   (vulpea-note-id after)))
         (parent-id (mm/parent-of (vulpea-note-id before))))
    (if parent-id
        (mm/set-parent note-id parent-id)
      (mm/make-root note-id))
    (mm/set-order-key note-id new-key)
    (message "✓ Placed between: %s / %s"
             (vulpea-note-title before)
             (vulpea-note-title after))))

(defun mm/make-root-interactive ()
  "Make the current note a root node (remove its parent)."
  (interactive)
  (let ((note (my/vulpea--buffer-note)))
    (mm/make-root (vulpea-note-id note))
    (message "✓ %s is now a root." (vulpea-note-title note))))

*** Navigation Commands
Quick movement between related nodes.

#+begin_src emacs-lisp
(defun mm/goto-parent ()
  "Visit the parent of the current note in the map."
  (interactive)
  (let* ((note      (my/vulpea--buffer-note))
         (parent-id (mm/parent-of (vulpea-note-id note))))
    (if parent-id
        (vulpea-visit (vulpea-db-get-by-id parent-id))
      (message "Root node — no parent."))))

(defun mm/goto-child ()
  "Select and visit a direct child of the current note."
  (interactive)
  (let* ((note     (my/vulpea--buffer-note))
         (children (mm/children-of (vulpea-note-id note))))
    (unless children (user-error "No children."))
    (let* ((cands (mapcar (lambda (row)
                            (let ((n (vulpea-db-get-by-id (car row))))
                              (cons (vulpea-note-title n) n)))
                          children))
           (target (cdr (assoc (completing-read "Child: "
                                                (mapcar #'car cands)
                                                nil t)
                               cands))))
      (when target (vulpea-visit target)))))

(defun mm/goto-sibling ()
  "Select and visit a sibling of the current note."
  (interactive)
  (let* ((note     (my/vulpea--buffer-note))
         (siblings (mm/siblings-of (vulpea-note-id note))))
    (unless siblings (user-error "No siblings."))
    (let* ((cands (mapcar (lambda (row)
                            (let ((n (vulpea-db-get-by-id (car row))))
                              (cons (vulpea-note-title n) n)))
                          siblings))
           (target (cdr (assoc (completing-read "Sibling: "
                                                (mapcar #'car cands)
                                                nil t)
                               cands))))
      (when target (vulpea-visit target)))))

(defun mm/goto-root ()
  "Visit the root ancestor of the current note."
  (interactive)
  (let* ((note (my/vulpea--buffer-note))
         (id   (vulpea-note-id note)))
    (while (mm/parent-of id)
      (setq id (mm/parent-of id)))
    (if (string= id (vulpea-note-id note))
        (message "Already at root.")
      (vulpea-visit (vulpea-db-get-by-id id)))))
#+end_src

*** Map Display
Renders the full tree as an Org buffer or a side-buffer group view.

#+begin_src emacs-lisp
(defun mm/map-display ()
  "Display the full mind map as an org tree in *manifold-map*."
  (interactive)
  (let ((roots (mm/roots)))
    (if (not roots)
        (message "No nodes in map. Use mm/assign-parent to build the tree.")
      (with-current-buffer (get-buffer-create "*manifold-map*")
        (erase-buffer)
        (org-mode)
        (setq-local org-return-follows-link t)
        (insert "#+TITLE: Manifold Mind Map\n\n")
        (mm/map--insert-level roots 1)
        (goto-char (point-min))
        (display-buffer (current-buffer))))))

(defun mm/map--insert-level (rows level)
  "Insert ROWS as org headings at LEVEL, recursing into children."
  (dolist (row rows)
    (let* ((note-id  (car row))
           (note     (vulpea-db-get-by-id note-id))
           (status   (mm/status-of note-id))
           (stars    (make-string level ?*))
           (tag      (unless (string= status "active")
                       (format " [%s]" (upcase status))))
           (headings (mm/headings-of note-id))
           (children (mm/children-of note-id)))
      (when note
        (insert (format "%s %s%s  [[id:%s]]\n"
                        stars
                        (vulpea-note-title note)
                        (or tag "")
                        note-id))
        (dolist (h-row headings)
          (when-let ((h-note (vulpea-db-get-by-id (car h-row))))
            (insert (format "%s %s  [[id:%s]]\n"
                            (make-string (1+ level) ?*)
                            (vulpea-note-title h-note)
                            (car h-row)))))
        (when children
          (mm/map--insert-level children (+ level 1 (if headings 1 0))))))))

(defun mm/group-display ()
  "Display the Group of the current buffer note in a side buffer."
  (interactive)
  (let* ((note  (my/vulpea--buffer-note))
         (group (mm/group-of (vulpea-note-id note))))
    (with-current-buffer (get-buffer-create "*mm-group*")
      (erase-buffer)
      (org-mode)
      (insert (format "#+TITLE: Group — %s\n\n"
                      (vulpea-note-title (plist-get group :file))))
      (insert "* File\n")
      (insert (format "  [[id:%s][%s]]\n"
                      (vulpea-note-id (plist-get group :file))
                      (vulpea-note-title (plist-get group :file))))
      (when (plist-get group :headings)
        (insert "* Headings\n")
        (dolist (h (plist-get group :headings))
          (insert (format "  [[id:%s][%s]]\n"
                          (vulpea-note-id h)
                          (vulpea-note-title h)))))
      (display-buffer (current-buffer)))))

*** JSON Export

#+begin_src emacs-lisp
(defun mm/export-json (&optional output-path)
  "Export the full mind map as JSON to OUTPUT-PATH.
Default: <notes-dir>/manifold-map.json"
  (interactive)
  (let* ((path  (or output-path
                    (expand-file-name "manifold-map.json"
                                      (my/notes-directory))))
         (edges (mm/all-edges))
         (data  `((nodes . ,(vconcat (mm/json--build-nodes)))
                  (edges . ((structural     .
                             ,(vconcat (mapcar (lambda (e)
                                                `((source . ,(car e))
                                                  (target . ,(cdr e))))
                                               (plist-get edges :structural))))
                            (semantic       .
                             ,(vconcat (mapcar (lambda (e)
                                                `((source . ,(car e))
                                                  (target . ,(cdr e))))
                                               (plist-get edges :semantic))))
                            (presentational .
                             ,(vconcat (mapcar (lambda (e)
                                                `((source . ,(car e))
                                                  (target . ,(cdr e))))
                                               (plist-get edges :presentational)))))))))
    (with-temp-file path
      (insert (json-encode data)))
    (message "✓ Manifold map → %s" path)))

(defun mm/json--build-nodes ()
  "Build list of JSON node alists for all file nodes."
  (let ((rows (emacsql (vulpea-db)
                       [:select [note-id parent-id order-key depth]
                        :from mm-nodes
                        :where (= node-type "file")
                        :order-by [(asc depth) (asc order-key)]]))
        nodes)
    (dolist (row rows)
      (let* ((note-id   (nth 0 row))
             (parent-id (nth 1 row))
             (order-key (nth 2 row))
             (depth     (nth 3 row))
             (note      (vulpea-db-get-by-id note-id))
             (status    (mm/status-of note-id))
             (headings  (mm/json--build-headings note-id)))
        (when note
          (push `((id       . ,note-id)
                  (title    . ,(vulpea-note-title note))
                  (type     . "file")
                  (parentId . ,parent-id)
                  (orderKey . ,order-key)
                  (depth    . ,depth)
                  (status   . ,status)
                  (headings . ,(vconcat headings)))
                nodes))))
    (nreverse nodes)))

(defun mm/json--build-headings (file-note-id)
  "Build list of heading alists for FILE-NOTE-ID's group."
  (let (headings)
    (dolist (row (mm/headings-of file-note-id))
      (when-let ((h-note (vulpea-db-get-by-id (car row))))
        (push `((id       . ,(car row))
                (title    . ,(vulpea-note-title h-note))
                (type     . "heading")
                (orderKey . ,(cadr row))
                (status   . ,(mm/status-of (car row))))
              headings)))
    (nreverse headings)))

(defun mm/init ()
  "Bootstrap mm_nodes from all existing vault notes.
Headings auto-parent to their file's root note.
File notes get parent-id nil — build the tree via mm/assign-parent.
Safe to run repeatedly — never overwrites existing entries."
  (interactive)
  (message "mm/init: scanning vault...")

    (dolist (note (vulpea-db-query))
      (mm/extract nil
                  (list :id    (vulpea-note-id note)
                        :level (vulpea-note-level note)

                        :pos   (vulpea-note-pos note)))

             count)))

*** Auto-Prompt on Create
Prompts to assign the new note in the mind map after every file-level
note creation — mandatory choice between parent, child, sibling, or
unknown. Headings auto-parent to their file, no prompt for those.

#+begin_src emacs-lisp
(defun mm/--prompt-placement (note)
  "Prompt to place NOTE in the mind map — parent/child/sibling/unknown."
  (let ((note-id (vulpea-note-id note))
        (file    (vulpea-note-path note)))
    (when (and file (file-exists-p file))
      (with-current-buffer (find-file-noselect file)
        (let ((choice (read-char-choice
                       "[p]arent  [c]hild  [s]ibling  [u]nknown: "
                       '(?p ?c ?s ?u))))
          (pcase choice
            (?p                                 ; parent — pick a child
             (condition-case nil
                 (let ((child (vulpea-select "Child to place under this note: "
                                             :require-match t))
                       (child-id (vulpea-note-id child)))
                   (unless (string= note-id child-id)
                     (mm/reparent child-id note-id)
                     (message "✓ %s is now a child of %s"
                              (vulpea-note-title child)
                              (vulpea-note-title note))))
               (quit (message "Placement cancelled."))))
            (?c                                 ; child — pick a parent
             (condition-case nil
                 (let ((parent (vulpea-select "Parent for this note: "
                                              :require-match t))
                       (parent-id (vulpea-note-id parent)))
                   (unless (string= note-id parent-id)
                     (mm/reparent note-id parent-id)
                     (message "✓ %s → child of %s"
                              (vulpea-note-title note)
                              (vulpea-note-title parent))))
               (quit (message "Placement cancelled."))))
            (?s                                 ; sibling — place after
             (condition-case nil
                 (let ((sibling (vulpea-select "Place after: "
                                               :require-match t))
                       (sib-id (vulpea-note-id sibling))
                       (parent-id (mm/parent-of (vulpea-note-id sibling))))
                   (if parent-id
                       (mm/set-parent note-id parent-id)
                     (mm/make-root note-id))
                   (mm/set-order-key note-id (mm/order-key-after sib-id))
                   (message "✓ %s → after %s"
                            (vulpea-note-title note)
                            (vulpea-note-title sibling)))
               (quit (message "Placement cancelled."))))
             (?u                                 ; unknown — leave as root
              (mm/make-root note-id)
              (message "✓ %s left as root" (vulpea-note-title note)))))))))

(defun mm/--apply-placement (placement target-id note)
  "Apply mind map PLACEMENT for NOTE with TARGET-ID.
Called as a post-apply function by the mind-map prompt."
  (let ((note-id (vulpea-note-id note)))
    (pcase placement
      ("parent"
       (when (and target-id (not (string= note-id target-id)))
         (mm/reparent target-id note-id)))
      ("child"
       (when (and target-id (not (string= note-id target-id)))
         (mm/reparent note-id target-id)))
      ("sibling"
       (when target-id
         (let ((parent-id (mm/parent-of target-id)))
           (if parent-id
               (mm/set-parent note-id parent-id)
             (mm/make-root note-id))
           (mm/set-order-key note-id (mm/order-key-after target-id)))))
      ("root"
       (mm/make-root note-id)))))
