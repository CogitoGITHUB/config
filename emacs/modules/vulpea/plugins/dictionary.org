* Database Table
#+begin_src emacs-lisp
(defun my/vulpea-dictionary--ensure-table ()
  "Create the dictionary_words table if it does not exist."
  (emacsql (vulpea-db)
    [:create :table :if :not :exists dictionary_words
     ([(note-id :not-null)
       (word :not-null)
       (pos :null)
       (source :null)
       (added :null)]
      (:primary-key [note-id])
      (:foreign-key [note-id] :references notes [id]
       :on-delete :cascade))]))
#+end_src

* DB Extractor
Reads WORD/POS/SOURCE/ADDED straight from note-data's properties and
title — no AST walk needed, so this stays fast and worker-eligible.

#+begin_src emacs-lisp
(defun my/vulpea-dictionary--extract (_ctx note-data)
  "Extract dictionary fields from NOTE-DATA and store in dictionary_words table."
  (let* ((note-id (plist-get note-data :id))
         (tags    (plist-get note-data :tags))
         (props   (plist-get note-data :properties)))
    (when (member "dictionary" tags)
      (let ((word   (or (cdr (assoc "WORD" props))
                        (plist-get note-data :title)))
            (pos    (cdr (assoc "POS" props)))
            (source (cdr (assoc "SOURCE" props)))
            (added  (cdr (assoc "ADDED" props))))
        (emacsql (vulpea-db)
          [:delete :from dictionary_words :where (= note-id $s1)]
          note-id)
        (emacsql (vulpea-db)
          [:insert :into dictionary_words :values $v1]
          (vector note-id word pos source added))))
    note-data))

(with-eval-after-load 'vulpea
  (my/vulpea-dictionary--ensure-table)
  (vulpea-db-register-extractor
   (make-vulpea-extractor
    :name       'dictionary-words
    :version    1
    :priority   80
    ;; No :requires-ast — works from note-data (title/tags/properties) only.
    :extract-fn #'my/vulpea-dictionary--extract)))
#+end_src

(with-eval-after-load 'vulpea
  (vulpea-schema-define
   'dictionary
   :predicate (lambda (note) (member "dictionary" (vulpea-note-tags note)))
   :fields
   (list
    (list :key "POS"
          :type 'string
          :required t
          :one-of my/vulpea-dictionary-pos-choices
          :validate (lambda (val _note)
                      (if (equal val "WARNING") "missing-required" t)))
    (list :key "ADDED"
          :type 'string
          :required t
          :validate (lambda (val _note)
                      (if (equal val "WARNING") "missing-required" t)))
    )))
#+end_src

* Word Capture
Dedicated capture command rather than the global prompt registry —
POS/SOURCE prompts should only ever fire for dictionary notes, not every
file-level note in the vault.

#+begin_src emacs-lisp
(defun my/vulpea-dictionary--slugify (word)
  "Turn WORD into a safe file-name stem."
  (let ((s (downcase word)))
    (setq s (replace-regexp-in-string "[^a-z0-9]+" "-" s))
    (replace-regexp-in-string "^-+\\|-+$" "" s)))

(defun my/vulpea-dictionary-new-word (word)
   "Create a new dictionary note for WORD.
Prompts for subdirectory under `dictionary/', fast mode level,
dictionary-specific prompts (POS, pronunciation, etc.), and
general prompts (state, tags, etc.).  Tags are set automatically:
`dictionary', the slugified WORD, and each subdirectory level
under `dictionary/'."
  (interactive "sWord: ")
  (let* ((root (expand-file-name "dictionary/" (my/notes-directory)))
         (_ (make-directory root t))
         (added (format-time-string "[%Y-%m-%d]"))
         (slug (my/vulpea-dictionary--slugify word))
         (subdir (my/vulpea-root-dir
                  (expand-file-name "dictionary/" (my/notes-directory))
                  "Subdirectory under dictionary/: "))
          (file-name (concat "dictionary/" subdir slug))
         (path-tags (if (string-empty-p subdir)
                        '()
                      (split-string (directory-file-name subdir) "/" t)))
         (vulpea-create-default-function (lambda (_) '()))
         (prompts (my/vulpea-collect-prompts-with-fast '(dictionary) '(file)))
         (tags (append (list "dictionary" slug) path-tags
                       (plist-get prompts :tags)))
         (properties (append
                      `(("ADDED" . ,added))
                      (plist-get prompts :properties)))
         (body (plist-get prompts :body))
         (post-apply (plist-get prompts :post-apply))
         (note (my/vulpea--create-file-wrapper
                word file-name tags properties body)))
    (dolist (fn post-apply)
      (funcall fn note))
    (let ((org-capture-templates
           `(("v" "Edit note body"
               plain
               (file+function
                ,(vulpea-note-path note)
                (lambda ()
                  (goto-char (point-max))
                  (skip-chars-backward " \t\n")))
               ,(or body "%?")
               :kill-buffer t :unnarrowed t))))
      (org-capture nil "v")
      (set-buffer-modified-p nil)
      (modaled-set-state "insert"))
    (my/vulpea-dictionary-regenerate-dictionary)
    note))
#+end_src

* Dictionary Regeneration
#+begin_src emacs-lisp
(defun my/vulpea-dictionary--all-words ()
  "Return the sorted, deduplicated list of words from dictionary_words."
  (let ((rows (emacsql (vulpea-db)
               [:select [word] :from dictionary_words])))
    (sort (delete-dups (mapcar #'car rows)) #'string<)))

(defun my/vulpea-dictionary-regenerate-dictionary ()
  "Rebuild the flat ispell .pws file from all notes tagged \"dictionary\"."
  (interactive)
  (make-directory (file-name-directory my/vulpea-dictionary-pws-file) t)
  (let ((words (my/vulpea-dictionary--all-words)))
    (with-temp-file my/vulpea-dictionary-pws-file
      (insert (format "personal_ws-1.1 en %d\n" (length words)))
      (dolist (w words)
        (insert w "\n"))))
  (message "Vocab dictionary regenerated (%d words)"
           (length (my/vulpea-dictionary--all-words))))

;; Point ispell at the generated file.
(with-eval-after-load 'ispell
  (setq ispell-personal-dictionary my/vulpea-dictionary-pws-file))
#+end_src

* Org-Drill Integration
#+begin_src emacs-lisp
(defun my/vulpea-dictionary--note-files ()
  "Return file paths of all notes tagged \"dictionary\"."
  (mapcar #'vulpea-note-path
          (vulpea-db-query-by-tags-some '("dictionary"))))

(defun my/vulpea-dictionary-drill ()
  "Run an org-drill session scoped to dictionary word files only."
  (interactive)
  (require 'org-drill)
  (let ((files (my/vulpea-dictionary--note-files)))
    (unless files
      (user-error "No dictionary notes found"))
    (org-drill files)))
#+end_src
