* Database Table
#+begin_src emacs-lisp
(defun my/vulpea-vocab--ensure-table ()
  "Create the vocab_words table if it does not exist."
  (emacsql (vulpea-db)
    [:create :table :if :not :exists vocab_words
     ([(note-id :not-null)
       (word :not-null)
       (pos :null)
       (source :null)
       (added :null)]
      (:primary-key [note-id])
      (:foreign-key [note-id] :references notes [id]
       :on-delete :cascade))]))
#+end_src

* DB Extractor
Reads WORD/POS/SOURCE/ADDED straight from note-data's properties and
title — no AST walk needed, so this stays fast and worker-eligible.

#+begin_src emacs-lisp
(defun my/vulpea-vocab--extract (_ctx note-data)
  "Extract vocab fields from NOTE-DATA and store in vocab_words table."
  (let* ((note-id (plist-get note-data :id))
         (tags    (plist-get note-data :tags))
         (props   (plist-get note-data :properties)))
    (when (member "vocab" tags)
      (let ((word   (or (cdr (assoc "WORD" props))
                        (plist-get note-data :title)))
            (pos    (cdr (assoc "POS" props)))
            (source (cdr (assoc "SOURCE" props)))
            (added  (cdr (assoc "ADDED" props))))
        (emacsql (vulpea-db)
          [:delete :from vocab_words :where (= note-id $s1)]
          note-id)
        (emacsql (vulpea-db)
          [:insert :into vocab_words :values $v1]
          (vector note-id word pos source added))))
    note-data))

(with-eval-after-load 'vulpea
  (my/vulpea-vocab--ensure-table)
  (vulpea-db-register-extractor
   (make-vulpea-extractor
    :name       'vocab-words
    :version    1
    :priority   80
    ;; No :requires-ast — works from note-data (title/tags/properties) only.
    :extract-fn #'my/vulpea-vocab--extract)))
#+end_src

* Schema Definition
#+begin_src emacs-lisp
(with-eval-after-load 'vulpea
  (vulpea-schema-define
   'vocab
   :predicate (lambda (note) (member "vocab" (vulpea-note-tags note)))
   :fields
   (list
    (list :key "POS"
          :type 'string
          :required t
          :one-of my/vulpea-vocab-pos-choices
          :validate (lambda (val _note)
                      (if (equal val "WARNING") "missing-required" t)))
    (list :key "ADDED"
          :type 'string
          :required t
          :validate (lambda (val _note)
                      (if (equal val "WARNING") "missing-required" t)))
    (list :key "SOURCE"
          :type 'string
          :required nil))))
#+end_src

* Word Capture
Dedicated capture command rather than the global prompt registry —
POS/SOURCE prompts should only ever fire for vocab notes, not every
file-level note in the vault.

#+begin_src emacs-lisp
(defun my/vulpea-vocab--slugify (word)
  "Turn WORD into a safe file-name stem."
  (let ((s (downcase word)))
    (setq s (replace-regexp-in-string "[^a-z0-9]+" "-" s))
    (replace-regexp-in-string "^-+\\|-+$" "" s)))

(defun my/vulpea-vocab-new-word (word)
  "Create a new vocab note for WORD: prompts for POS and SOURCE,
creates the file under `my/vulpea-vocab-directory`, tags it \"vocab\",
and seeds a drill-card skeleton heading."
  (interactive "sWord: ")
  (let* ((pos (completing-read "POS: " my/vulpea-vocab-pos-choices
                                nil t nil nil "⚠ WARNING"))
         (source (read-string "Source (optional): "))
         (added (format-time-string "[%Y-%m-%d]"))
         (file-name (format "vocab/%s.org" (my/vulpea-vocab--slugify word)))
         (note (vulpea-create
                word file-name
                :tags '("vocab")
                :properties `(("POS" . ,pos)
                              ("SOURCE" . ,source)
                              ("ADDED" . ,added)))))
    (when note
      (with-current-buffer (find-file-noselect (vulpea-note-path note))
        (goto-char (point-max))
        (unless (bolp) (insert "\n"))
        (insert (format "\n* %s :drill:\n\n%s\n" word
                        "  (edit: put the answer/definition here, hidden by the heading fold)"))
        (save-buffer)))
    (my/vulpea-vocab-regenerate-dictionary)
    note))
#+end_src

* Dictionary Regeneration
#+begin_src emacs-lisp
(defun my/vulpea-vocab--all-words ()
  "Return the sorted, deduplicated list of words from vocab_words."
  (let ((rows (emacsql (vulpea-db)
               [:select [word] :from vocab_words])))
    (sort (delete-dups (mapcar #'car rows)) #'string<)))

(defun my/vulpea-vocab-regenerate-dictionary ()
  "Rebuild the flat ispell .pws file from all notes tagged \"vocab\"."
  (interactive)
  (let ((words (my/vulpea-vocab--all-words)))
    (with-temp-file my/vulpea-vocab-pws-file
      (insert (format "personal_ws-1.1 en %d\n" (length words)))
      (dolist (w words)
        (insert w "\n"))))
  (message "Vocab dictionary regenerated (%d words)"
           (length (my/vulpea-vocab--all-words))))

;; Point ispell at the generated file.
(with-eval-after-load 'ispell
  (setq ispell-personal-dictionary my/vulpea-vocab-pws-file))
#+end_src

* Org-Drill Integration
#+begin_src emacs-lisp
(defun my/vulpea-vocab--note-files ()
  "Return file paths of all notes tagged \"vocab\"."
  (mapcar #'vulpea-note-path
          (vulpea-db-query-by-tags-some '("vocab"))))

(defun my/vulpea-vocab-drill ()
  "Run an org-drill session scoped to vocab word files only."
  (interactive)
  (require 'org-drill)
  (let ((files (my/vulpea-vocab--note-files)))
    (unless files
      (user-error "No vocab notes found"))
    (org-drill files)))
#+end_src
