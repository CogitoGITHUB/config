#+title: Vulpea Transclusion Plugin
#+priority: 15

Use [[id:UUID][TITLE]] links. Safe because org-agenda-files is
scoped to the notes directory only (see Notes Directory above),
so org-id-find never scans straight build symlinks.
#+begin_src emacs-lisp
(defun my/vulpea-transclusion-add (&optional only-body)
  "Select a vulpea note and insert a live transclusion at point.
With prefix arg (C-u), transclude body only — no heading line.

:level is computed automatically so the note lands as a child of
the current heading:
  at file level    → :level 1
  inside *  H      → :level 2
  inside ** H      → :level 3  …"
  (interactive "P")
  (require 'org-transclusion)
  (let ((org-transclusion-exclude-elements (list 'property-drawer))
        note id title ctx target opts line directive-pos)
    (setq note   (vulpea-select "Transclude note: " :require-match t))
    (setq id     (vulpea-note-id note))
    (setq title  (vulpea-note-title note))
    (setq ctx    (or (org-current-level) 0))
    (setq target (1+ ctx))
    (setq opts   (if only-body ":only-contents"
                   (format ":level %d" target)))
    (setq line   (format "#+transclude: [[id:%s][%s]] %s\n"
                          id title opts))
    (setq directive-pos (point))
    (insert line)
    (goto-char directive-pos)
    (unless org-transclusion-mode
      (let ((org-transclusion-add-all-on-activate nil))
        (org-transclusion-mode 1)))
    (org-transclusion-add)))

;; Transcluded ref notes show as collapsed headings — do NOT auto-unfold
;; them, so the ref's property drawer / ID stays hidden and you just see
;; "* Ref" closed at point.
(remove-hook 'org-transclusion-after-add-functions
             #'my/org-transclusion-show-subtree)

;; Auto-enable org-transclusion-mode in org files.
;; Safe now — org-agenda-files is scoped to notes directory only,
;; so org-id-find never scans straight build symlinks.
(setq org-transclusion-add-all-on-activate t)
;; Hide ref property drawers inside transcluded headings so a live ref
;; (e.g. STATE WORK: ACTIVE) renders as a single closed heading.
(setq org-transclusion-exclude-elements (list 'property-drawer))
(add-hook 'org-mode-hook (lambda () (org-transclusion-mode 1)))

;; Re-add transclusions after revert (e.g. auto-revert triggered by git ops).
(add-hook 'after-revert-hook #'org-transclusion-add-all)

;; Refresh transclusions in all open org buffers after any save.
;; Uses remove-all + add-all (the same pattern org-transclusion uses
;; internally for before-save / after-save).
(defun my/org-transclusion-refresh-all ()
  (dolist (buf (buffer-list))
    (with-current-buffer buf
      (when org-transclusion-mode
        (org-transclusion-remove-all)
        (org-transclusion-add-all)))))
(add-hook 'after-save-hook #'my/org-transclusion-refresh-all)
#+end_src
*** Transclusion Link Index
Indexes [[id:...][...]] links inside #+transclude: keywords into a
dedicated SQLite table so the deletion guard can find them as incoming links.
Also provides a text-based scanner for files not yet indexed by vulpea DB.

#+begin_src emacs-lisp
(defun my/transclude-links-ensure-table ()
  "Create the transclude_links table if it does not exist."
  (emacsql (vulpea-db)
           [:create :table :if :not :exists transclude_links
            ([(source :not-null)
              (dest   :not-null)]
             (:primary-key [source dest])
             (:foreign-key [source] :references notes [id]
              :on-delete :cascade)
             (:foreign-key [dest]   :references notes [id]
              :on-delete :cascade))]))

(defun my/vulpea-extract-transclude-links (ctx note-data)
  "Extract id: links from #+transclude: keywords in CTX."
  (let* ((ast     (vulpea-parse-ctx-ast ctx))
         (note-id (plist-get note-data :id))
         (found   nil))
    (org-element-map ast 'keyword
      (lambda (kw)
        (when (string= (org-element-property :key kw) "TRANSCLUDE")
          (let ((value (org-element-property :value kw)))
            (when (string-match "\\[\\[id:\\([^] \t\n]+\\)\\]\\[" value)
              (push (match-string-no-properties 1 value) found))))))
    (when found
      (let ((unique (delete-dups (nreverse found))))
        (emacsql (vulpea-db)
                 [:delete :from transclude_links :where (= source $s1)]
                 note-id)
        (emacsql (vulpea-db)
                 [:insert :into transclude_links :values $v1]
                 (mapcar (lambda (d) (vector note-id d)) unique))
        (plist-put note-data :transclude-links unique)))
    note-data))

(with-eval-after-load 'vulpea
  (my/transclude-links-ensure-table)
  (vulpea-db-register-extractor
   (make-vulpea-extractor
    :name       'transclude-links
    :version    1
    :priority   60
    :extract-fn #'my/vulpea-extract-transclude-links)))

(defun my/vulpea-db-query-transclude-links-to (target-id)
  "Return list of source note IDs that transclude TARGET-ID."
  (mapcar #'car
          (emacsql (vulpea-db)
                   [:select [source] :from transclude_links
                    :where (= dest $s1)]
                   target-id)))

(defun my/vulpea-links--find-transclude-refs (target-id)
  "Search all org files in vault for #+transclude: links to TARGET-ID.
Returns list of source note IDs. Acts as fallback for files not yet
indexed by the vulpea DB extractor."
  (let ((results nil)
        (root (my/notes-directory)))
    (dolist (file (nconc (directory-files-recursively root "\\.org\\'")
                         (directory-files-recursively root "/[^./]+\\'")))
      (with-temp-buffer
        (insert-file-contents file)
        (goto-char (point-min))
        (when (re-search-forward
               (format "#\\+transclude:.*\\[\\[id:%s\\]\\[" (regexp-quote target-id))
               nil t)
          (org-mode)
          (let ((id (org-entry-get (point-min) "ID")))
            (when id
              (push id results))))))
    (delete-dups results)))
#+end_src
