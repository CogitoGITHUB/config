#+begin_src emacs-lisp
(defun my/capture-directory ()
  "Return the org-capture (doct) templates directory."
  (expand-file-name "full-file-templates/" (my/templates-directory)))
(with-eval-after-load 'org-capture
  (require 'doct)
  (dolist (f (directory-files (my/capture-directory) t "\\.el$"))
    (load f))
  (setq org-capture-templates
        (append (when (boundp 'my/capture-vulpea) my/capture-vulpea)
                (when (boundp 'my/capture-tasks)  my/capture-tasks))))
#+end_src
