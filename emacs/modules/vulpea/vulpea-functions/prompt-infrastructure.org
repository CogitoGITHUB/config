Central registry so that adding a new .org to modules/vulpea/properties/ automatically
includes its prompt in the creation flow.

The registry defvar and defun live in
[[file:../properties/00-register.org]] so they load before any
prompt file calls ~my/vulpea-register-prompt~.

** Help File Mapping

A few prompt names don't match their filename
(e.g. `scheduled` lives in `schedule.org` with `deadline`).

#+begin_src emacs-lisp
(defvar my/vulpea-prompt-help-map
  '((scheduled . "schedule.org")
    (deadline  . "schedule.org"))
  "Alist mapping prompt :name to help filename.
Entries here override the default <name>.org.")

(defun my/vulpea--help-file (name)
  "Return the help filename for prompt NAME."
  (or (alist-get name my/vulpea-prompt-help-map)
      (format "%s.org" name)))

(defun my/vulpea--show-help (name)
  "Open prompt NAME's help file full-screen with org-tree-slide."
  (let (file path)
    (setq file (my/vulpea--help-file name))
    (setq path (expand-file-name
                (concat "modules/vulpea/properties/" file)
                user-emacs-directory))
    (when (file-exists-p path)
      (delete-other-windows)
      (find-file path)
      (goto-char (point-min))
      (org-tree-slide-mode 1))))

(defun my/vulpea--close-help ()
  "Stop org-tree-slide and kill the help buffer."
  (org-tree-slide-mode -1)
  (kill-buffer))
#+end_src

** TODO State Carrier

Global variable to pass TODO state from prompt collection through
to the heading-format file writer, so TODO appears on the heading
line (`* TODO Title`) instead of in the property drawer.

#+begin_src emacs-lisp
(defvar my/vulpea--creation-todo nil
  "TODO value for the current note creation.
Set by `my/vulpea-collect-prompts', read by the heading-format
advice on `vulpea--format-note-content'.")
#+end_src

** Collector

#+begin_src emacs-lisp
(defun my/vulpea-collect-prompts (context)
  "Run all registered prompts matching CONTEXT and merge results."
  (let (result)
    (dolist (spec my/vulpea-prompt-registry)
      (when (memq context (plist-get spec :contexts))
        (my/vulpea--show-help (plist-get spec :name))
        (let (val frag)
          (setq val (condition-case nil
                        (funcall (plist-get spec :prompt-fn))
                      (quit nil)))
          (setq frag (funcall (plist-get spec :to-plist) val context))
          (when-let ((v (plist-get frag :tags)))
            (setq result (plist-put result :tags
                         (append (plist-get result :tags) v))))
          (when-let ((v (plist-get frag :properties)))
            (setq result (plist-put result :properties
                         (append (plist-get result :properties)
                                 (cl-remove-if (lambda (p) (string-empty-p (cdr p))) v)))))
          (when-let ((v (plist-get frag :body)))
            (setq result (plist-put result :body
                         (concat (or (plist-get result :body) "") v))))
          (when-let ((v (plist-get frag :post-apply)))
            (setq result (plist-put result :post-apply
                         (append (plist-get result :post-apply) v))))
          (when-let ((v (plist-get frag :after)))
            (setq result (plist-put result :after v)))
          (when-let ((v (plist-get frag :file-name)))
            (setq result (plist-put result :file-name v)))
          (when (plist-member frag :todo)
            (setq result (plist-put result :todo (plist-get frag :todo))
                  my/vulpea--creation-todo (plist-get frag :todo))))
        (my/vulpea--close-help)))
    result))
#+end_src

** Fast Collector

When fast mode is active, all prompts return their "skip" defaults
(e.g. "WARNING" for state prompts) without user interaction.
This function temporarily overrides input primitives to achieve that.

#+begin_src emacs-lisp
(defun my/vulpea--fast-read-string (prompt &optional initial-input history default-value _inherit)
  "Return DEFAULT-VALUE or INITIAL-INPUT or empty — no prompt."
  (or default-value initial-input ""))

(defun my/vulpea--fast-completing-read (prompt collection &optional predicate _require-match initial-input history default-value _inherit)
  "Return DEFAULT-VALUE or first candidate."
  (or default-value
      (if (and collection (not (functionp collection))) (car collection) "")
      ""))

(defun my/vulpea--fast-read-char-choice (prompt chars &optional _inhibit-keyboard-quit)
  "Return \\r or first char — maps to skip/default in all prompt functions."
  (if (memq ?\r chars) ?\r (car chars)))

(defun my/vulpea--collect-default-prompts (context)
  "Like `my/vulpea-collect-prompts' but all prompts return defaults.
Temporarily overrides `read-string', `completing-read' and
`read-char-choice' to return their default/skip values."
  (cl-letf (((symbol-function 'read-string) #'my/vulpea--fast-read-string)
            ((symbol-function 'completing-read) #'my/vulpea--fast-completing-read)
            ((symbol-function 'read-char-choice) #'my/vulpea--fast-read-char-choice))
    (my/vulpea-collect-prompts context)))
#+end_src

** Missing Prompts Management

Manages MISSING PROMPTS.org — an auto-generated file tracking notes whose
state prompts were skipped (set to WARNING).

#+begin_src emacs-lisp
(defvar my/vulpea-missing--file "MISSING PROMPTS.org"
  "Filename for tracking notes with WARNING state properties.")

(defun my/vulpea-missing--path ()
  "Return full path to MISSING PROMPTS.org."
  (expand-file-name my/vulpea-missing--file
                    (my/notes-directory)))

(defun my/vulpea-missing--ensure-file ()
  "Create MISSING PROMPTS.org if it does not exist."
  (let ((path (my/vulpea-missing--path)))
    (unless (file-exists-p path)
      (with-temp-file path
        (insert "#+TITLE: MISSING PROMPTS\n")
        (insert "#+AUTHOR: Shape\n")
        (insert "#+DESCRIPTION: Auto-generated - notes with unanswered state prompts.\n")
        (insert "#+STARTUP: overview\n\n")))
    path))

(defun my/vulpea-missing--rebuild-from-health ()
  "Rebuild MISSING PROMPTS.org from schema health data.
Lists all notes with schema violations, grouped by schema.
Each entry includes violation details in the property drawer."
  (interactive)
  (let ((path (my/vulpea-missing--path))
        (by-id (make-hash-table :test #'equal)))
    (dolist (health (vulpea-schema-collection-health))
      (let ((schema-name (vulpea-schema-health-schema health)))
        (dolist (pair (vulpea-schema-health-invalid-notes health))
          (let* ((note (car pair))
                 (violations (cdr pair))
                 (id (vulpea-note-id note))
                 (title (vulpea-note-title note))
                 (existing (gethash id by-id)))
            (if existing
                (progn
                  (plist-put existing :schemas
                             (cons (symbol-name schema-name)
                                   (plist-get existing :schemas)))
                  (plist-put existing :violations
                             (append violations (plist-get existing :violations))))
              (puthash id
                       (list :id id :title title
                             :schemas (list (symbol-name schema-name))
                             :violations violations)
                       by-id))))))
    (let (entries)
      (maphash (lambda (_id plist) (push plist entries)) by-id)
      (if entries
          (with-temp-file path
            (insert "#+TITLE: MISSING PROMPTS\n")
            (insert "#+AUTHOR: Shape\n")
            (insert "#+DESCRIPTION: Auto-generated - notes with schema violations.\n")
            (insert "#+STARTUP: overview\n\n")
            (dolist (entry (nreverse entries))
              (insert (format "* TODO [[id:%s][%s]]\n"
                              (plist-get entry :id)
                              (plist-get entry :title)))
              (insert "  :PROPERTIES:\n")
              (insert (format "  :CREATED:       %s\n"
                              (format-time-string "<%Y-%m-%d %a>")))
              (insert (format "  :SCHEMAS:       %s\n"
                              (string-join (plist-get entry :schemas) ", ")))
              (let ((vs (plist-get entry :violations)))
                (when vs
                  (insert (format "  :VIOLATIONS:    %s\n"
                                  (string-join
                                   (mapcar #'vulpea-violation-message vs)
                                   "; ")))))
              (insert "  :END:\n\n")))
        (when (file-exists-p path)
          (delete-file path))))))

(defun my/vulpea--post-sync (_property _value _note)
  "No-op.  Schema health replaces per-property MISSING PROMPTS sync."
  nil)
#+end_src

** Rebuild & Hooks

#+begin_src emacs-lisp
(add-hook 'org-capture-after-finalize-hook #'my/vulpea-missing--rebuild-from-health)

(defvar my/vulpea-missing--startup-timer nil
  "Timer handle for retrying startup rebuild until DB is ready.")

(defun my/vulpea-missing--startup-rebuild ()
  "Run rebuild-from-health if vulpea DB has notes; retry later if not."
  (condition-case nil
      (if (> (length (vulpea-db-query)) 0)
          (progn
            (my/vulpea-missing--rebuild-from-health)
            (when my/vulpea-missing--startup-timer
              (cancel-timer my/vulpea-missing--startup-timer)
              (setq my/vulpea-missing--startup-timer nil)))
        (setq my/vulpea-missing--startup-timer
              (run-with-idle-timer 5 nil #'my/vulpea-missing--startup-rebuild)))
    (error
     (setq my/vulpea-missing--startup-timer
           (run-with-idle-timer 5 nil #'my/vulpea-missing--startup-rebuild)))))
