** Fast Mode
#+begin_src emacs-lisp
(defvar my/vulpea--fast-level 0
  "Fast mode level for current creation.
0=normal, 1=skip general, 2=skip all, 3=quick (no body).")

(defun my/vulpea--fast-level ()
  "Prompt for fast mode level.
Return 0 (normal), 1 (skip general), 2 (skip all), or 3 (quick, no body)."
  (let ((c (read-char-choice "Fast: (a)ormal (o)skip general (e)skip all (u)quick, no body [Enter=quick] " '(?a ?o ?e ?u ?\r ?\n))))
    (setq my/vulpea--fast-level
          (pcase c
            (?a 0)
            (?o 1)
            (?e 2)
            ((or ?u ?\r ?\n) 3)))))

(setq use-dialog-box nil)

(defun my/vulpea-title-to-slug-spaces (orig-fun title)
  "Replace underscores with spaces in the slug from ORIG-FUN."
  (replace-regexp-in-string "_" " " (funcall orig-fun title)))

(advice-add 'vulpea-title-to-slug :around #'my/vulpea-title-to-slug-spaces)
#+end_src
** Default Parameters
#+begin_src emacs-lisp
(defvar my/vulpea-default-subdir-root nil
  "When non-nil, use this as the default directory for the subdirectory prompt
instead of `my/notes-directory'.")

(with-eval-after-load 'vulpea
  (setq vulpea-create-default-function
        (lambda (title)
          (let* ((raw-name (read-string
                            (format "File name for \"%s\" (blank = slug): " title)))
                 (slug (if (string-empty-p raw-name)
                           (vulpea-title-to-slug title)
                         (file-name-sans-extension raw-name)))
                  (subdir (my/vulpea-root-dir (or my/vulpea-default-subdir-root (my/notes-directory))))
                 (fname (concat subdir slug))
                 (prompts (my/vulpea-collect-prompts-with-fast '() '(file))))
            (setq my/vulpea--pending-post-apply (plist-get prompts :post-apply))
            (list
             :file-name  fname
             :tags       (plist-get prompts :tags)
             :properties (plist-get prompts :properties)
             :body       (plist-get prompts :body)))))))
#+end_src
