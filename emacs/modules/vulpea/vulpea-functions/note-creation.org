** Fast Mode
#+begin_src emacs-lisp
(defvar my/vulpea--fast-mode nil
  "Non-nil when fast mode is active, skipping all state/tag prompts.")

(defun my/vulpea--fast-mode-p ()
  "Prompt for fast mode. Enter or y = fast, n = full.
Return non-nil when fast mode is selected."
  (let ((c (read-char-choice "Fast mode? [Y/n]: " '(?y ?n ?\r ?\n))))
    (setq my/vulpea--fast-mode (memq c '(?y ?\r ?\n)))))
#+end_src
** Default Parameters
#+begin_src emacs-lisp
(with-eval-after-load 'vulpea
  (setq vulpea-create-default-function
        (lambda (title)
          (let* ((fast (my/vulpea--fast-mode-p))
                 (raw-name (read-string
                            (format "File name for \"%s\" (blank = slug): " title)))
                 (slug (if (string-empty-p raw-name)
                           (concat (vulpea-title-to-slug title) ".org")
                         (if (string-suffix-p ".org" raw-name) raw-name
                           (concat raw-name ".org"))))
                 (subdir (let* ((root (my/notes-directory))
                                (raw (read-directory-name
                                      "Subdirectory: " root root)))
                           (if (string-equal
                                (file-name-as-directory
                                 (directory-file-name raw))
                                (file-name-as-directory root))
                               ""
                             (file-relative-name raw root))))
                 (fname (concat subdir slug))
                 (prompts (if fast
                              (my/vulpea--collect-default-prompts 'file)
                            (my/vulpea-collect-prompts 'file))))
            (setq my/vulpea--pending-post-apply (plist-get prompts :post-apply))
            (list
             :file-name  fname
             :tags       (plist-get prompts :tags)
             :properties (plist-get prompts :properties)
             :body       (plist-get prompts :body)))))))
#+end_src
