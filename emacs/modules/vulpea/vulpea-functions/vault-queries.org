** Relationship Queries
#+begin_src emacs-lisp
(defun my/vulpea-children-of (note)
  "Return list of vulpea notes whose :SOURCE: property is NOTE's ID."
  (let ((id (vulpea-note-id note)))
    (vulpea-db-query
     (lambda (n)
       (string= (alist-get "SOURCE" (vulpea-note-properties n)) id)))))

(defun my/vulpea-parent-of (note)
  "Return the vulpea note referenced by NOTE's :PARENT: property, or nil."
  (when-let ((parent-id (alist-get "PARENT" (vulpea-note-properties note))))
    (vulpea-db-get-by-id parent-id)))

(defun my/vulpea-source-of (note)
  "Return the vulpea note referenced by NOTE's :SOURCE: property, or nil."
  (when-let ((source-id (alist-get "SOURCE" (vulpea-note-properties note))))
    (vulpea-db-get-by-id source-id)))

** Vault Health
#+begin_src emacs-lisp
(defun my/vulpea-check-vault ()
  "Report dead links, orphan notes, isolated notes and title collisions."
  (interactive)
  (let ((dead       (vulpea-db-query-dead-links))
        (orphans    (vulpea-db-query-orphan-notes))
        (isolated   (vulpea-db-query-isolated-notes))
        (collisions (vulpea-db-query-title-collisions)))
    (with-current-buffer (get-buffer-create "*vulpea-health*")
      (erase-buffer)
      (insert (format "Dead links: %d\n" (length dead)))
      (dolist (d dead)
        (insert (format "  %s -> missing %s\n"
                        (vulpea-note-title (car d)) (cdr d))))
      (insert (format "\nOrphan notes: %d\n" (length orphans)))
      (dolist (o orphans)
        (insert (format "  %s\n" (vulpea-note-title o))))
      (insert (format "\nIsolated notes: %d\n" (length isolated)))
      (dolist (i isolated)
        (insert (format "  %s\n" (vulpea-note-title i))))
      (insert (format "\nTitle collisions: %d\n" (length collisions)))
      (dolist (c collisions)
        (insert (format "  \"%s\" -- %d notes\n"
                        (car c) (length (cdr c)))))
      (display-buffer (current-buffer)))))
#+end_src
