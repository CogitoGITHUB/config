** Open Capture for Note
#+begin_src emacs-lisp
(defvar my/vulpea--chosen-capture nil
  "The capture template name chosen in the current capture flow.
Set by `my/vulpea--choose-capture', used by
`my/vulpea--open-capture-for-note' to store `:TEMPLATE:' on the note.")

(defun my/vulpea--capture-file-hash (name)
  "Return a SHA-256 hash of capture file NAME, or nil."
  (let ((file (my/vulpea--capture-path name)))
    (when (file-exists-p file)
      (with-temp-buffer
        (insert-file-contents file)
        (secure-hash 'sha256 (buffer-string))))))

(defun my/vulpea--choose-capture (&optional note)
  "Choose a capture template via completing-read with preview.
When NOTE is a vulpea-note, preview shows the note's file with the
template appended — exactly what will land on disk after capture."
  (let ((title (if note (vulpea-note-title note) nil))
        (note-file (if note (vulpea-note-path note) nil))
        names current preview-win chosen)
    (setq names (cons "none" (my/vulpea--capture-names)))
    (setq current "none")
    (delete-other-windows)
    (setq preview-win (split-window-below
                       (- (floor (* 0.85 (frame-height))))))
    (when-let ((buf (my/vulpea--capture-buffer current title note-file)))
      (set-window-buffer preview-win buf))
    ;; ── completing-read with live buffer swap ──────────────
    (setq chosen
          (minibuffer-with-setup-hook
              (lambda ()
                (add-hook
                 'post-command-hook
                 (lambda ()
                   (let ((candidate
                          (ignore-errors
                            (when (fboundp 'vertico--candidate)
                              (vertico--candidate)))))
                     (when (and (stringp candidate)
                                (member candidate names)
                                (not (equal candidate current))
                                (window-live-p preview-win))
                       (if (equal candidate "none")
                           (let ((buf (get-buffer-create " *capture-preview-empty*")))
                             (when note-file
                               (with-current-buffer buf
                                 (erase-buffer)
                                 (insert-file-contents note-file)
                                 (org-mode)
                                 (outline-show-all)
                                 (when (fboundp 'org-indent-mode)
                                   (org-indent-mode 1))))
                             (set-window-buffer preview-win buf))
                         (when-let ((buf (my/vulpea--capture-buffer candidate title note-file)))
                           (set-window-buffer preview-win buf)))
                       (setq current candidate))))
                 nil 'local))
            (completing-read "Body template: " names nil t nil nil "none")))
    ;; ── Cleanup preview ────────────────────────────────────────
    (my/vulpea--capture-cleanup names)
    (delete-window preview-win)
    (setq my/vulpea--chosen-capture
          (if (and chosen
                   (not (string-empty-p chosen))
                   (not (equal chosen "none")))
              chosen
            nil))
    (if my/vulpea--chosen-capture
        (with-temp-buffer
          (insert-file-contents
           (my/vulpea--capture-path my/vulpea--chosen-capture))
          (buffer-string))
      "%?")))

(defun my/vulpea--open-capture-for-note (note &optional template)
  "Open org-capture for NOTE with capture preview.
When a template is chosen, stores `:TEMPLATE:' and `:TEMPLATE_HASH:'
properties on NOTE for schema validation."
  (let ((file (vulpea-note-path note))
        (tmpl (or template
                  (my/vulpea--choose-capture note))))
    (unless file
      (user-error "Note has no file path"))
    ;; store template metadata for schema validation
    (when my/vulpea--chosen-capture
      (my/vulpea--property-put note "TEMPLATE" my/vulpea--chosen-capture)
      (my/vulpea--property-put note "TEMPLATE_HASH"
                                (or (my/vulpea--capture-file-hash
                                     my/vulpea--chosen-capture)
                                    "unknown")))
    (let ((org-capture-templates
           `(("v" "Edit note body"
               plain
               (file+function
                ,file
                (lambda ()
                  (goto-char (point-max))
                  (skip-chars-backward " \t\n")))
               ,tmpl
               :kill-buffer t)))
          (org-capture-use-indirect-buffer nil))
      (when-let ((buf (get-file-buffer file)))
        (when (eq buf (current-buffer))
          (switch-to-buffer (other-buffer))))
      (org-capture nil "v"))))

(defun my/vulpea--cleanup-after-capture ()
  "After org-capture finalizes, prompt to open the new note.
Runs via `org-capture-after-finalize-hook'."
  (when (and (bound-and-true-p my/vulpea--new-note)
             my/vulpea--new-note)
    (let ((file (vulpea-note-path my/vulpea--new-note)))
      (when (and file (not my/vulpea--skip-open-prompt))
        (when (y-or-n-p (format "Open %s? " (file-name-nondirectory file)))
          (find-file file)))))
  (setq my/vulpea--new-note nil)
  (setq my/vulpea--skip-open-prompt nil))

(add-hook 'org-capture-after-finalize-hook #'my/vulpea--cleanup-after-capture)
#+end_src
** Property Utilities
#+begin_src emacs-lisp
(defun my/vulpea-links--current-note-id ()
  "Return the ID of the vulpea note at point, or nil.
Checks heading-level ID first, then file-level ID."
  (or (org-id-get)
      (org-entry-get (point-min) "ID")))

(defun my/vulpea--property-put (note key value)
  "Set org property KEY to VALUE on NOTE using `org-entry-put'.
This stores the property inside NOTE's property drawer, making it
queryable via `vulpea-note-properties' and the vulpea SQL properties table."
  (let* ((file (vulpea-note-path note))
         (pos (vulpea-note-pos note))
         (level (vulpea-note-level note)))
    (with-current-buffer (find-file-noselect file)
      (save-excursion
        (cond
         ((and level (> level 0) pos)
          (goto-char pos)
          (org-back-to-heading t))
         (t
          (goto-char (point-min))))
        (org-entry-put (point) key value)
        (save-buffer)))))
#+end_src
** Capture Advice
#+begin_src emacs-lisp
(defun my/vulpea--open-in-capture (note)
  "After vulpea-create: record source relationship, set up capture."
  (when note
    (when-let* ((source-id (my/vulpea-links--current-note-id))
                ((not (string= source-id (vulpea-note-id note)))))
      (my/vulpea--property-put note "SOURCE" source-id))
    (setq my/vulpea--new-note note))
  note)

(defun my/vulpea-find-advice (original-fn &rest args)
  "Wrap `vulpea-find' to open org-capture for newly created notes.
The `:filter-return' advice on `vulpea-create' `setq's the global
`my/vulpea--new-note' so `org-capture-after-finalize-hook' can
access it later for the \"Open file?\" prompt."
  (setq my/vulpea--new-note nil)
  (let* ((my/vulpea--skip-capture t)
         (result (apply original-fn args)))
    (when my/vulpea--new-note
      (my/vulpea--open-capture-for-note my/vulpea--new-note))
    result))

(with-eval-after-load 'vulpea
  (advice-add 'vulpea-create :filter-return #'my/vulpea--open-in-capture)
  (advice-add 'vulpea-find :around #'my/vulpea-find-advice)
  (advice-add 'vulpea-insert :around #'my/vulpea-find-advice))
#+end_src
