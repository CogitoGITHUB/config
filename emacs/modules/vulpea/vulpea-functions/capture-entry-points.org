* Capture Entry Points
:PROPERTIES:
:CATEGORY: editing
:END:
In vulpea v2 there is no org-roam-capture.  The pattern is:
=vulpea-create= makes the file and registers it in the DB; org-capture opens
that file so you fill in the body.  Put your =doct= declarations in
=vulpea/templates/vulpea.el= (loaded automatically by the doct loader).
#+begin_src emacs-lisp
;;; vulpea/templates/vulpea.el  --  define your doct templates here
;;
;; Use the helpers below as the :file / :function value in doct entries.
;; Declare your template list here as e.g.:
;;   (defvar my/capture-vulpea (doct `(("Vulpea" :keys "n" ...))))
;; -- file-level helpers ------------------------------------------------
(defun my/vulpea-capture-new-file ()
  "Prompt for title, subdir and tags; create the note; open capture for body."
  (let* ((my/vulpea--skip-capture t)
         (vulpea-create-default-function (lambda (_) '()))
         (fast (my/vulpea--fast-mode-p))
         (title  (read-string "Note title: "))
         (raw-name (read-string (format "File name for \"%s\" (blank = slug): " title)))
         (slug (if (string-empty-p raw-name)
                   (concat (vulpea-title-to-slug title) ".org")
                 (if (string-suffix-p ".org" raw-name) raw-name
                   (concat raw-name ".org"))))
         (subdir (let* ((root (my/notes-directory))
                        (raw (read-directory-name "Subdirectory: " root root)))
                   (if (string-equal (file-name-as-directory
                                      (directory-file-name raw))
                                     (file-name-as-directory root))
                       ""
                     (file-relative-name raw root))))
         (fname (concat subdir slug))
         (prompts (if fast (my/vulpea--collect-default-prompts 'file)
                     (my/vulpea-collect-prompts 'file)))
          (note   (my/vulpea--create-file-wrapper
                   title
                   fname
                   (plist-get prompts :tags)
                   (plist-get prompts :properties)
                   (plist-get prompts :body))))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (my/vulpea--open-capture-for-note note)))

(defun my/vulpea--create-file-wrapper (title fname tags properties body)
  (let ((vulpea-create-default-function (lambda (_) '())))
    (vulpea-create title fname
                   :tags tags
                   :properties properties
                   :body body)))

;; -- heading helper ----------------------------------------------------
(defun my/vulpea-capture-heading ()
  "Select a parent note, create a heading under it, then open via org-capture."
  (interactive)
  (let (vulpea-create-default-function parent title prompts note fast)
    (setq vulpea-create-default-function (lambda (_) '()))
    (setq fast (my/vulpea--fast-mode-p))
    (setq parent (let ((buffer-note (my/vulpea--buffer-note)))
                   (if buffer-note
                       (if (member (read-char-choice
                                    (format "Use \"%s\" as parent? (RET/y=yes, n=search) "
                                            (vulpea-note-title buffer-note))
                                    '(?y ?n ?\r ?\n))
                                   '(?y ?\r ?\n))
                           buffer-note
                         (vulpea-select "Parent note: " :require-match t))
                     (vulpea-select "Parent note: " :require-match t))))
    (setq title (read-string "Heading title: "))
    (setq prompts (if fast (my/vulpea--collect-default-prompts 'heading)
                     (my/vulpea-collect-prompts 'heading)))
    (setq my/vulpea--pending-post-apply nil)
    (setq note (vulpea-create title nil
                              :parent parent
                              :tags   (plist-get prompts :tags)
                              :properties (plist-get prompts :properties)
                              :after  (plist-get prompts :after)))
    (unless (vulpea-note-p note)
      (error "vulpea-create returned %S (expected a vulpea-note)" note))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (my/vulpea--open-capture-for-note note)))
;; -- insert heading in current buffer -----------------------------------
(defun my/vulpea-insert-heading (title)
  "Insert a vulpea heading at point in the current buffer.
Runs heading prompts for metadata, then opens a capture buffer
for body editing.  The new heading becomes a child of the heading
at point, or a child of the file if point is at file level."
  (interactive (list (read-string "Heading title: ")))
  (let (id parent-note prompts note fast)
    (setq fast (my/vulpea--fast-mode-p))
    (setq id (my/vulpea-links--current-note-id))
    (setq parent-note (vulpea-db-get-by-id id))
    (setq prompts (if fast (my/vulpea--collect-default-prompts 'heading)
                     (my/vulpea-collect-prompts 'heading)))
    (setq my/vulpea--pending-post-apply nil)
    (setq note (vulpea-create title nil
                              :parent parent-note
                              :tags (plist-get prompts :tags)
                              :properties (plist-get prompts :properties)
                              :after (plist-get prompts :after)))
    (dolist (fn (plist-get prompts :post-apply))
      (funcall fn note))
    (setq my/vulpea--skip-open-prompt t)
    (my/vulpea--open-capture-for-note note)))
