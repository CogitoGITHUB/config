#+title: Vulpea — Root Dir
#+priority: 11

* Root Dir
:PROPERTIES:
:CATEGORY: editing
:END:

Centralized directory prompt shared by all capture paths.
Each caller passes its own root; the function returns the relative path.

#+begin_src emacs-lisp
(defun my/vulpea-root-dir (&optional root prompt)
  "Prompt for a subdirectory under ROOT.
ROOT defaults to `(my/notes-directory)'.  PROMPT defaults to \"Subdirectory: \".
Returns the relative path from ROOT as a directory name (with trailing slash),
or \"\" if ROOT itself is selected (Enter at root prompt).
Type \".\" or press Enter for root level; select or type a subdirectory name."
  (let* ((root (expand-file-name (or root (my/notes-directory))))
         (dirs (cl-loop for f in (directory-files root nil "^[^.]" t)
                        when (file-directory-p (expand-file-name f root))
                        collect f))
         (candidates (cons "." dirs))
         (choice (completing-read (or prompt "Subdirectory: ") candidates nil nil nil nil ".")))
    (if (string-equal choice ".")
        ""
      (make-directory (expand-file-name choice root) t)
      (file-name-as-directory choice))))
#+end_src
