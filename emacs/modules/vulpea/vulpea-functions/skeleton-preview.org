#+begin_src emacs-lisp :lexical t
(defun my/vulpea--capture-dir ()
  (expand-file-name "modules/vulpea/templates/full-file-templates/"
                    user-emacs-directory))

(defun my/vulpea--capture-names ()
  (mapcar (lambda (f)
            (file-name-sans-extension (file-name-nondirectory f)))
          (directory-files (my/vulpea--capture-dir)
                           t "\\.org\\'")))

(defun my/vulpea--capture-path (name)
  (expand-file-name (format "%s.org" name)
                    (my/vulpea--capture-dir)))

(defun my/vulpea--capture-buffer (name &optional title note-file)
  "Return preview buffer for capture template NAME with TITLE.
When NOTE-FILE is given, prepend its contents to show the full
file as it will appear after capture.  Replaces %? with TITLE."
  (let* ((template-path (my/vulpea--capture-path name))
         (buf-name (format " *capture-preview-%s*" name))
         (display-title (or (and (stringp title) (not (string-empty-p title)) title)
                            "[title]")))
    (when (file-exists-p template-path)
      (with-current-buffer (get-buffer-create buf-name)
        (erase-buffer)
        (when (and note-file (file-exists-p note-file))
          (insert-file-contents note-file)
          (goto-char (point-max))
          (insert "\n"))
        (goto-char (point-max))
        (insert-file-contents template-path)
        (goto-char (point-min))
        (while (search-forward "%?" nil t)
          (replace-match display-title))
        (org-mode)
        (outline-show-all)
        (when (fboundp 'org-indent-mode)
          (org-indent-mode 1))
        (current-buffer)))))

(defun my/vulpea--capture-cleanup (names)
  "Kill all capture preview buffers for NAMES."
  (dolist (name names)
    (let ((buf (get-buffer (format " *capture-preview-%s*" name))))
      (when (buffer-live-p buf)
        (kill-buffer buf)))))
