(setq org-link-descriptive t)
(setq org-return-follows-link t)

(with-eval-after-load 'org
  (advice-add #'org-link--reveal :override #'ignore))

;; Standard org-activate-links skips property drawer contexts.
;; Activate links there manually.
(font-lock-add-keywords 'org-mode
  '((my/org--activate-property-links)))

(defun my/org--activate-property-links (limit)
  "Activate Org links inside property drawer values up to LIMIT."
  (when (re-search-forward org-link-any-re limit t)
    (let* ((start (match-beginning 0))
           (end   (match-end 0))
           (visible-start (or (match-beginning 3) (match-beginning 2)))
           (visible-end (or (match-end 3) (match-end 2)))
           (type  (match-string-no-properties 1))
           (path  (match-string-no-properties 2))
           (desc  (match-string-no-properties 3)))
      (save-excursion
        (goto-char (line-beginning-position))
        (when (looking-at ":\\([A-Z_][A-Z_0-9]*\\):")
          (let ((prop-start (match-beginning 0)))
            (org-remove-flyspell-overlays-in start end)
            (org-rear-nonsticky-at end)
            (remove-text-properties start end '(invisible nil))
            (when org-link-descriptive
              (add-text-properties start visible-start
                                   (list 'invisible 'org-link))
              (add-text-properties visible-end end
                                   (list 'invisible 'org-link)))
            (add-text-properties prop-start end
                                 (list 'mouse-face 'highlight
                                       'keymap org-mouse-map
                                       'help-echo (concat "LINK: " type ":" path)
                                       'font-lock-multiline t
                                       'rear-nonsticky t))
            (add-face-text-property prop-start end 'org-link)
            t))))))

(font-lock-add-keywords 'org-mode
  '((my/org--clean-property-display)))

(defun my/org--clean-property-display (limit)
  "Clean up property drawer display: hide colons, underscores as spaces."
  (catch 'found
    (while (re-search-forward "^:\\([A-Z_][A-Z_0-9]*\\):" limit t)
      (let ((pname (match-string-no-properties 1)))
        (unless (member pname '("PROPERTIES" "END"))
          (let ((name-start (match-beginning 1))
                (name-end (match-end 1))
                (trailing-colon (1- (match-end 0))))
            (put-text-property (match-beginning 0) (1+ (match-beginning 0)) 'display "")
            (put-text-property trailing-colon (match-end 0) 'display "")
            (save-excursion
              (goto-char name-start)
              (while (re-search-forward "_" name-end t)
                (put-text-property (match-beginning 0) (match-end 0) 'display " "))))
          (throw 'found t))))
    nil))
