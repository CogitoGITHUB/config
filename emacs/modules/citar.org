** Config
#+begin_src emacs-lisp
(setq citar-bibliography '("~/bib/references.bib"))

(setq org-cite-global-bibliography citar-bibliography
      org-cite-insert-processor   'citar
      org-cite-follow-processor   'citar
      org-cite-activate-processor 'citar)

(add-hook 'org-mode-hook
          (lambda () (add-to-list 'completion-at-point-functions #'citar-capf)))
(add-hook 'LaTeX-mode-hook
          (lambda () (add-to-list 'completion-at-point-functions #'citar-capf)))

(with-eval-after-load 'vulpea
  (defun my/citar-vulpea-get-candidates (keys)
    "Return alist of (citekey . display) for KEYS that have vulpea notes."
    (mapcar (lambda (k) (cons k k))
            (seq-filter
             (lambda (k)
               (not (null (vulpea-db-query-by-meta "citekey" k))))
             keys)))

  (defun my/citar-vulpea-has-notes (keys)
    "Return hash of citekey -> bool for KEYS."
    (let ((tbl (make-hash-table :test 'equal)))
      (dolist (k keys)
        (puthash k
                 (not (null (vulpea-db-query-by-meta "citekey" k)))
                 tbl))
      tbl))

  (defun my/citar-vulpea-open-note (citekey)
    "Open or create a vulpea note for CITEKEY."
    (let ((note (car (vulpea-db-query-by-meta "citekey" citekey))))
      (if note
          (vulpea-visit note)
        (vulpea-create citekey nil
                       :properties `(("CITEKEY" . ,citekey))))))

  (citar-register-notes-source
   'citar-vulpea
   (list :name      "Vulpea Notes"
         :category  'file
         :items     #'my/citar-vulpea-get-candidates
         :hasitems  #'my/citar-vulpea-has-notes
         :open      #'my/citar-vulpea-open-note
         :create    #'my/citar-vulpea-open-note))

  (setq citar-notes-source 'citar-vulpea))
#+end_src
