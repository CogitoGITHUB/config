    (pcase-dolist (`(,key . ,def) keybindings)

** Define-keys function
#+begin_src emacs-lisp
(defun manifolding-keyboard-define-keys (&rest body)
  "Define keybindings for states, substates, or globally.

:states / :substates / :global / :keymaps / :inherit / :bind"
  (declare (indent defun))
  (let* ((states (plist-get body :states))
         (substates (plist-get body :substates))
         (keymaps (append (mapcar #'manifolding-keyboard-get-state-keymap states)
                          (mapcar #'manifolding-keyboard-get-substate-keymap substates)
                          (plist-get body :keymaps)))
         (global (plist-get body :global))
         (inherit (plist-get body :inherit))
         (bind (plist-get body :bind)))
    (pcase-dolist (`(,inherit-keymap . ,inherit-keys) inherit)
      (dolist (k inherit-keys)
        (let ((binding (keymap-lookup (symbol-value inherit-keymap) k)))
          (dolist (keymap keymaps)
            (keymap-set (symbol-value keymap) k binding))
          (when global (keymap-global-set k binding)))))
    (pcase-dolist (`(,key . ,def) bind)
      (let ((keys (if (listp key) key (list key))))
        (dolist (k keys)
          (dolist (keymap keymaps)
            (keymap-set (symbol-value keymap) k def))
          (when global (keymap-global-set k def)))))))

(defmacro manifolding-keyboard--define-minor-mode (mode keymap body)
  "Define a minor MODE with KEYMAP and options in BODY."
  (let ((mode-name (symbol-name mode))
        (sparse (plist-get body :sparse))
        (no-suppress (plist-get body :no-suppress))
        (lighter (plist-get body :lighter))
        (cursor-type (plist-get body :cursor-type))
        (no-emulation (plist-get body :no-emulation)))
    `(progn
      (defvar ,keymap
        (if ,sparse (make-sparse-keymap) (make-keymap))
        ,(format "Keymap for %s." mode-name))
      (unless ,no-suppress
        (suppress-keymap ,keymap))
      (define-minor-mode ,mode
        ,(format "Manifolding-Keyboard minor mode %s" mode-name)
        :lighter ,lighter
        :keymap ,keymap
        (when ,cursor-type (setq-local cursor-type ,cursor-type)))
      (unless ,no-emulation
        (unless (boundp 'manifolding-keyboard--emulation-mode-map-alist)
          (setq manifolding-keyboard--emulation-mode-map-alist nil))
        (add-to-list 'manifolding-keyboard--emulation-mode-map-alist (cons ',mode ,keymap))
        (add-to-list 'emulation-mode-map-alists manifolding-keyboard--emulation-mode-map-alist)))))

(defun manifolding-keyboard-enable-substate-on-state-change (substate &rest body)
  "Enable SUBSTATE under conditions specified in BODY."
  (declare (indent defun))
  (let ((states (plist-get body :states))
        (major (plist-get body :major))
        (minor (plist-get body :minor))
        (pred (plist-get body :pred))
        (check-minor (lambda (mode)
                       (and (boundp mode) (symbol-value mode)))))
    (add-variable-watcher
     'manifolding-keyboard-state
     (lambda (_ new-val _ _)
       (funcall (manifolding-keyboard-get-substate-mode substate)
                (if (and (or (not major) (memq major-mode major))
                         (or (not minor) (memq t (mapcar check-minor minor)))
                         (or (not pred) (funcall pred))
                         (or (not states) (member new-val states)))
                    1 -1))))))

** TUI ESC / Meta disambiguation (Meow-extracted, ~my/esc-mode~)
Extracted from ~meow-esc.el~ (meow 1.5.0, GPLv3). In the terminal, ESC can
double as META because they share a keycode. This mode rewires the terminal's
~input-decode-map~ binding for ESC through a ~:filter~ that waits
~my/esc-delay~: if another key follows within the delay it is treated as a Meta
combo, otherwise a standalone ESC. This gives BOTH standalone-ESC and Meta /
ESC-prefix usage in the TUI simultaneously. Layout-independent, Dvorak-safe.
~my/smart-escape~ (above) handles *state* selection on ESC; this handles the
terminal input layer. Not enabled by default — call ~(my/esc-mode 1)~ to turn on.
#+begin_src emacs-lisp
(defcustom my/esc-delay 0.1
  "Maximum time in seconds between ESC and the following key to be treated
as a Meta / ESC-prefix combo rather than a standalone ESC, when
~my/esc-mode~ is active in the terminal."
  :type 'number
  :group 'manifolding-keyboard)

(defvar my/esc-sequence [?\e]
  "The key sequence used to reach the terminal's original ESC binding.")

(define-minor-mode my/esc-mode
  "Ensure ESC works as a standalone key AND as Meta/prefix in the terminal.
When enabled, the terminal ~input-decode-map~'s ESC binding is wrapped in a
filter (see ~my/esc~) that waits ~my/esc-delay~: a key pressed within the delay
is read as a Meta/ESC-prefix combo; otherwise it is a standalone ESC."
  :init-value nil
  :global t
  (if my/esc-mode
      (progn
        (add-hook 'after-make-frame-functions #'my/esc-init-if-tui)
        (mapc #'my/esc-init-if-tui (frame-list)))
    (progn
      (remove-hook 'after-make-frame-functions #'my/esc-init-if-tui)
      (mapc #'my/esc-deinit-if-tui (frame-list)))))

(defun my/esc-init-if-tui (frame)
  "Wrap FRAME's terminal ESC binding with the ~my/esc~ filter, if in TUI."
  (with-selected-frame frame
    (unless window-system
      (let ((term (frame-terminal frame)))
        (when (not (terminal-parameter term 'my-esc-map))
          (let ((my-esc-map (lookup-key input-decode-map [?\e])))
            (set-terminal-parameter term 'my-esc-map my-esc-map)
            (define-key input-decode-map my/esc-sequence
              `(menu-item "" ,my-esc-map :filter ,#'my/esc))))))))

(defun my/esc-deinit-if-tui (frame)
  "Restore FRAME's terminal ESC binding, removing the ~my/esc~ filter."
  (with-selected-frame frame
    (unless window-system
      (let ((term (frame-terminal frame)))
        (when (terminal-live-p term)
          (let ((my-esc-map (terminal-parameter term 'my-esc-map)))
            (when my-esc-map
              (define-key input-decode-map my/esc-sequence my-esc-map)
              (set-terminal-parameter term 'my-esc-map nil))))))))

(defun my/esc (map)
  "Menu-item filter for ESC in the terminal.

If the ESC is promptly followed by another key (within ~my/esc-delay~),
return MAP (allowing the Meta / ESC-prefix keymap to handle it); otherwise
return [escape] (a standalone ESC). MAP is the original ESC keymap."
  (if (and (let ((keys (this-single-command-keys)))
             (and (> (length keys) 0)
                  (= (aref keys (1- (length keys))) ?\e)))
           (sit-for my/esc-delay))
      (prog1 [escape]
        (when defining-kbd-macro
          (end-kbd-macro)
          (setq last-kbd-macro (vconcat last-kbd-macro [escape]))
          (start-kbd-macro t t)))
    map))
#+end_src

** Atomic undo wrapper (Meow-extracted, ~my/wrap-collapse-undo~)
Extracted from ~meow--wrap-collapse-undo~ (meow-util.el:569–589, meow 1.5.0,
GPLv3). Runs a body of multiple edits but collapses them into a SINGLE undo
step via change groups, so downstream undo/undo-amalgamate behave atomically.
If the body signals an error the change group is cancelled instead. Layout-
independent, Dvorak-safe. Useful for multi-step edit commands (replace-all,
bulk ks/visual edits) that should feel like one atomic operation.
#+begin_src emacs-lisp
(defmacro my/wrap-collapse-undo (&rest body)
  "Like `progn' but perform BODY with its edits collapsed into one undo step.

If BODY completes, its changes are accepted and amalgamated with the preceding
undo entry so they undo as a single action. If BODY signals an error, the
change group is cancelled (no partial changes, no undo entry)."
  (declare (indent 0) (debug t))
  (let ((handle (make-symbol "--change-group-handle--"))
        (success (make-symbol "--change-group-success--")))
    `(let ((,handle (prepare-change-group))
           ;; Don't truncate any undo data in the middle of this.
           (undo-outer-limit nil)
           (undo-limit most-positive-fixnum)
           (undo-strong-limit most-positive-fixnum)
           (,success nil))
       (unwind-protect
           (progn
             (activate-change-group ,handle)
             (prog1 ,(macroexp-progn body)
               (setq ,success t)))
         (if ,success
             (progn
               (accept-change-group ,handle)
               (undo-amalgamate-change-group ,handle))
           (cancel-change-group ,handle))))))
#+end_src

(defun my/thing-parse-pair-search (push-token pop-token back near)
  "Search for the nearest PUSH-TOKEN or POP-TOKEN (regexps) around point.

When BACK is non-nil search backward, else forward. Returns (push|pop . POS)
with POS at the delimiter boundary; moves point to the token start for the
next iteration. Returns nil if neither token is found."
  (let* ((search-fn (if back #'re-search-backward #'re-search-forward))
         (match-fn (if back #'match-end #'match-beginning))
         (cmp-fn (if back #'> #'<))
         (push-next-pos nil)
         (pop-next-pos nil)
         (push-pos (save-mark-and-excursion
                     (when (funcall search-fn push-token nil t)
                       (setq push-next-pos (point))
                       (if near (funcall match-fn 0) (point)))))
         (pop-pos (save-mark-and-excursion
                    (when (funcall search-fn pop-token nil t)
                      (setq pop-next-pos (point))
                      (if near (funcall match-fn 0) (point))))))
    (cond
     ((and (not pop-pos) (not push-pos)) nil)
     ((not pop-pos) (goto-char push-next-pos) (cons 'push push-pos))
     ((not push-pos) (goto-char pop-next-pos) (cons 'pop pop-pos))
     ((funcall cmp-fn push-pos pop-pos)
      (goto-char push-next-pos) (cons 'push push-pos))
     (t
      (goto-char pop-next-pos) (cons 'pop pop-pos)))))

(defun my/thing-pair-function (push-token pop-token near)
  "Return (beg . end) of the balanced pair of regexps PUSH-TOKEN/POP-TOKEN
surrounding point, or nil. With NEAR, boundaries hug the delimiters; else beg
is the opening token's position and end is just past the closing token."
  (let* ((found nil)
         (depth 0)
         (beg (save-mark-and-excursion
                (prog1
                    (let ((case-fold-search nil))
                      (while (and (<= depth 0)
                                  (setq found (my/thing-parse-pair-search
                                               push-token pop-token t near)))
                        (let ((push-or-pop (car found)))
                          (if (eq 'push push-or-pop)
                              (cl-incf depth)
                            (cl-decf depth))))
                      (when (> depth 0) (cdr found)))
                  (setq depth 0 found nil))))
         (end (save-mark-and-excursion
                (let ((case-fold-search nil))
                  (while (and (>= depth 0)
                              (setq found (my/thing-parse-pair-search
                                           push-token pop-token nil near)))
                    (let ((push-or-pop (car found)))
                      (if (eq 'push push-or-pop)
                          (cl-incf depth)
                        (cl-decf depth))))
                  (when (< depth 0) (cdr found))))))
    (when (and beg end)
      (cons beg end))))

(defun my/select-inside-delimiters ()
  "Activate the INNER content of the innermost enclosing ~() [] {} <>~ around point.

Finds the nearest balanced delimiter pair surrounding point, sets point and
mark to the text strictly between the delimiters, and enters visual state."
  (interactive)
  (let ((pairs '(("\\\\(" . "\\\\)") ("\\\\[" . "\\\\]") ("{" . "}") ("<" . ">")))
        (best nil))
    (dolist (p pairs)
      (let* ((found (my/thing-pair-function (car p) (cdr p) nil))
             (beg (and found (car found)))
             (end (and found (cdr found))))
        (when (and beg end (<= beg (point) end)
                   (or (null best) (> beg (car best))))
          (setq best (cons beg end)))))
    (if best
        (progn
          (goto-char (car best))
          (push-mark (1+ (point)) t t)
          (goto-char (1- (cdr best)))
          (activate-mark)
          (manifolding-keyboard-set-state "visual"))
      (message "No enclosing delimiter pair"))))

(defun my/chain (command &optional first then)
  "Run FIRST commands, then COMMAND, then THEN commands.
Each element of FIRST/THEN is called interactively if a command, else applied

** Act-on-all-candidates (Embark-extracted, ~my/act-all~)
Extracted from ~embark-act-all~ (embark.el:2482–2558) + the candidate
collection of ~embark--maybe-transform-candidates~ (embark.el:2445–2480,
reduced to same-buffer occurrences for M-K), from Embark (GPLv3, Omar Antolín
Camarena). Embark's act-all performs ONE action on EVERY candidate, not just the
one at point. This port collects all buffer occurrences of the symbol/word at
point and offers a plain-key choice of actions on them all: ~k~ kill all
(destructive — asks confirmation, matching Embark's ~embark-confirm-act-all~),
~c~ copy all to the kill ring (non-destructive), anything else cancels.
Distinct from the existing occurrence machinery: B06 (~my/hl-*~, ~/~) highlights
+ navigates, Meow #04 (~my/jump-to-symbol~, ~f~) jumps — ~my/act-all~ ACTS
(batches a mutation/copy) on every occurrence, which nothing else does.
Internal helpers ~my/act-all-markers~ / ~my/act-all-delete~ / ~my/act-all-copy~
are consumed only by the bound ~my/act-all~ (bound-command rule). Deps: builtins
(~thing-at-point~, ~bounds-of-thing-at-point~, ~re-search-forward~,
~regexp-quote~, ~copy-marker~, ~delete-region~, ~kill-new~, ~read-char-choice~).
Layout-independent, Dvorak-safe. Bound ~my/act-all~ → plain key ~M~ (free in
both normal + motion) in normal + motion states.
#+begin_src emacs-lisp
(defun my/act-all-markers (thing)
  "Return markers for every occurrence of THING in the buffer, in order."
  (let (markers)
    (save-excursion
      (goto-char (point-min))
      (while (re-search-forward (regexp-quote thing) nil t)
        (push (cons (copy-marker (match-beginning 0))
                    (copy-marker (match-end 0)))
              markers)))
    (nreverse markers)))

(defun my/act-all-delete (thing markers)
  "Delete every occurrence of THING marked by MARKERS."
  (dolist (m markers)
    (let ((beg (car m)) (end (cdr m)))
      (when (and (marker-buffer beg) (<= (point-min) beg end (point-max)))
        (delete-region beg end))))
  (message "Deleted all %s" thing))

(defun my/act-all-copy (thing markers)
  "Copy every occurrence of THING marked by MARKERS to the kill ring."
  (let ((text (mapconcat (lambda (m)
                           (buffer-substring-no-properties (car m) (cdr m)))
                         markers "\n")))
    (kill-new text)
    (message "Copied %s: %d occurrence(s)" thing (length markers))))

(defun my/act-all ()
  "Act on all occurrences of the thing at point (Embark-extracted).
Embark's act-all performs one action on EVERY candidate, not just the thing at
point. This port collects all buffer occurrences of the symbol/word at point and
offers a plain-key action on all of them:

  k  kill all occurrences (destructive — asks confirmation)
  c  copy all occurrences to the kill ring (non-destructive)

Any other key cancels. Distinct from the occurrence highlighters/jumpers (B06,
Meow #04): this batches an action on every occurrence."
  (interactive)
  (let* ((b (or (bounds-of-thing-at-point 'symbol)
                (bounds-of-thing-at-point 'word)))
         (thing (and b (buffer-substring-no-properties (car b) (cdr b)))))
    (if (null thing)
        (message "No symbol/word at point")
      (let* ((markers (my/act-all-markers thing))
             (n (length markers)))
        (message "%s: %d occurrence%s" thing n (if (= n 1) "" "s"))
        (when (> n 1)
          (let ((choice (read-char-choice "Act on all: (k)ill, (c)opy, any key cancels"
                                          '(?k ?c))))
            (pcase choice
              (?k (when (y-or-n-p (format "Kill all %d occurrences of %s? " n thing))
                     (my/act-all-delete thing markers)))
              (?c (my/act-all-copy thing markers))
              (_ (message "Canceled")))))))))

* Hydra Infrastructure

** Defhydra-state macro
Single declaration generates both a manifolding-keyboard state's keybindings and its hydra menu. Bindings with `:exit nil` become state keys; all bindings appear in the hydra. The "Back → parent" group is automatic.
#+begin_src emacs-lisp
(defmacro my/defhydra-state (hydra-name state-name title back-target &rest spec)
  "Define manifolding-keyboard STATE-NAME and its hydra menu from SPEC.
STATE-NAME may be nil for hydras without a manifolding-keyboard state.
BACK-TARGET is (HYDRA/BODY LABEL) for the back link, or nil for root.
SPEC is alternating group-name / binding-list pairs:
  GROUP-NAME  ((KEY CMD LABEL &key EXIT) ...)
Bindings with :exit nil go into the manifolding-keyboard state keymap.
A \"Back\" group linking to BACK-TARGET is appended automatically."
  (declare (indent defun))
  (let ((groups '())
        (state-binds '()))
    (while spec
      (let ((group-name (pop spec))
            (bindings (pop spec)))
        (push (list group-name bindings) groups)
        (dolist (b bindings)
          (unless (plist-get (cdddr b) :exit)
            (push (list (nth 0 b) (nth 1 b)) state-binds)))))
    (setq groups (nreverse groups)
          state-binds (nreverse state-binds))
    (let ((back-group
           (when back-target
             `("Back" (("SPC" ,(car back-target) ,(cadr back-target) :exit t))))))
      `(progn
         ,@(when state-name
             `((manifolding-keyboard-define-keys
                :states '(,state-name)
                :bind ',state-binds)))
         (pretty-hydra-define ,hydra-name
           (:title ,title :quit-key "q" :separator "─")
           (,@(apply #'append (mapcar (lambda (g)
                                        (list (car g) (cadr g)))
                                      groups))
            ,@back-group))))))
#+end_src

#+begin_src emacs-lisp
(defmacro ks-multipop (lst n)
  "Remove LST's first N elements and return them."
  `(if (<= (length ,lst) ,n)
       (prog1 ,lst
         (setq ,lst nil))
     (prog1 ,lst
       (setcdr
        (nthcdr (1- ,n) (prog1 ,lst (setq ,lst (nthcdr ,n ,lst))))
        nil))))

(defun ks--de-bruijn (keys n)
  "De Bruijn sequence for alphabet KEYS and subsequences of length N."
  (let* ((k (length keys))
         (a (make-list (* n k) 0))
         sequence)
    (cl-labels ((db (T p)
                  (if (> T n)
                      (if (eq (% n p) 0)
                          (setq sequence
                                (append sequence
                                        (cl-subseq a 1 (1+ p)))))
                    (setf (nth T a) (nth (- T p) a))
                    (db (1+ T) p)
                    (cl-loop for j from (1+ (nth (- T p) a)) to (1- k) do
                         (setf (nth T a) j)
                         (db (1+ T) T)))))
      (db 1 1)
      (mapcar (lambda (n)
                (nth n keys))
              sequence))))

(defun ks--path-alist-1 (lst seq-len keys)
  "Build a De Bruin sequence from LST.
SEQ-LEN is how many elements of KEYS it takes to identify a match."
  (let ((db-seq (ks--de-bruijn keys seq-len))
        prev-pos prev-seq prev-win path-alist)
    ;; The De Bruijn seq is cyclic, so append the seq-len - 1 first chars to
    ;; the end.
    (setq db-seq (nconc db-seq (cl-subseq db-seq 0 (1- seq-len))))
    (cl-labels ((subseq-and-pop ()
                  (when (nth (1- seq-len) db-seq)
                    (prog1 (cl-subseq db-seq 0 seq-len)
                      (pop db-seq)))))
      (while lst
        (let* ((cur (car lst))
               (pos (cond
                      ;; ace-window has matches of the form (pos . wnd)
                      ((integerp (car cur)) (car cur))
                      ;; ks-jump have form ((start . end) . wnd)
                      ((consp (car cur)) (caar cur))
                      (t (error "Unexpected match representation: %s" cur))))
               (win (cdr cur))
               (path (if prev-pos
                         (let ((diff (if (eq win prev-win)
                                         (- pos prev-pos)
                                       0)))
                           (when (and (> diff 0) (< diff seq-len))
                             (while (and (nth (1- seq-len) db-seq)
                                         (not
                                          (eq 0
                                              (cl-search
                                               (cl-subseq prev-seq diff)
                                               (cl-subseq db-seq 0 seq-len)))))
                               (pop db-seq)))
                           (subseq-and-pop))
                       (subseq-and-pop))))
          (if (not path)
              (setq lst nil
                    path-alist nil)
            (push (cons path (car lst)) path-alist)
            (setq prev-pos pos
                  prev-seq path
                  prev-win win
                  lst (cdr lst))))))
    (nreverse path-alist)))

(defun ks-order-closest (x)
  (abs (- (if (numberp (car x))
              (car x)
            (caar x))
          (point))))

(defvar ks-command nil
  "Store the current command symbol.
E.g. `ks-goto-line' or `ks-goto-char'.")

(defun ks-tree (lst keys)
  "Coerce LST into a balanced tree.
The degree of the tree is the length of KEYS.
KEYS are placed appropriately on internal nodes."
  (let* ((len (length keys))
         (order-fn (cdr (assq ks-command ks-orders-alist)))
         (lst (if order-fn
                  (cl-sort lst #'< :key order-fn)
                lst)))
    (cl-labels
        ((rd (ls)
           (let ((ln (length ls)))
             (if (< ln len)
                 (cl-pairlis keys
                             (mapcar (lambda (x) (cons 'leaf x)) ls))
               (let ((ks (copy-sequence keys))
                     res)
                 (dolist (s (ks-subdiv ln len))
                   (push (cons (pop ks)
                               (if (eq s 1)
                                   (cons 'leaf (pop ls))
                                 (rd (ks-multipop ls s))))
                         res))
                 (nreverse res))))))
      (rd lst))))

(defun ks-subdiv (n b)
  "Distribute N in B terms in a balanced way."
  (let* ((p (1- (floor (+ (log n b) 1e-6))))
         (x1 (expt b p))
         (x2 (* b x1))
         (delta (- n x2))
         (n2 (/ delta (- x2 x1)))
         (n1 (- b n2 1)))
    (append
     (make-list n1 x1)
     (list
      (- n (* n1 x1) (* n2 x2)))
     (make-list n2 x2))))

(defun ks-traverse (tree walker &optional recur-key)
  "Traverse TREE generated by `ks-tree'.

(defun ks-read (tree display-fn cleanup-fn)
  "Select a leaf from TREE using consecutive `read-key'.

DISPLAY-FN should take CHAR and LEAF and signify that LEAFs
associated with CHAR will be selected if CHAR is pressed.  This is
commonly done by adding a CHAR overlay at LEAF position.

CLEANUP-FN should take no arguments and remove the effects of
multiple DISPLAY-FN invocations."
  (catch 'done
    (setq ks-current-path "")
    (while tree
      (let ((ks--leafs nil))
        (ks-traverse tree
                      (lambda (path leaf)
                        (push (cons path leaf) ks--leafs)))
        (dolist (x ks--leafs)
          (funcall display-fn (car x) (cdr x))))
      (let ((char (funcall ks-translate-char-function (read-key)))
            window
            branch)
        (funcall cleanup-fn)
        (if (setq window (ks-mouse-event-window char))
            (throw 'done (cons char window))

              (progn

                ;; exit so other packages can utilize its value.
                (setq ks-current-path
                      (concat ks-current-path (string (ks--key-to-char char))))
                (if (eq (car (setq tree (cdr branch))) 'leaf)
                    (throw 'done (cdr tree))))
            (funcall ks-handler-function char)))))))

(defun ks-read-de-bruijn (lst keys)
  "Select from LST dispatching on KEYS."
  ;; In theory, the De Bruijn sequence B(k,n) has k^n subsequences of length n
  ;; (the path length) usable as paths, thus that's the lower bound.  Due to
  ;; partially overlapping matches, not all subsequences may be usable, so it's
  ;; possible that the path-len must be incremented, e.g., if we're matching
  ;; for x and a buffer contains xaxbxcx only every second subsequence is
  ;; usable for the four matches.
  (catch 'done
    (let* ((path-len (ceiling (log (length lst) (length keys))))
           (alist (ks--path-alist-1 lst path-len keys)))
      (while (not alist)
        (cl-incf path-len)
        (setq alist (ks--path-alist-1 lst path-len keys)))
      (let* ((len (length (caar alist)))
             (i 0))
        (setq ks-current-path "")
        (while (< i len)
          (dolist (x (reverse alist))
            (ks--overlay-at-full (reverse (car x)) (cdr x)))
          (let ((char (funcall ks-translate-char-function (read-key))))
            (ks--remove-leading-chars)
            (setq alist
                  (delq nil
                        (mapcar (lambda (x)
                                  (when (eq (caar x) char)
                                    (cons (cdr (car x)) (cdr x))))
                                alist)))
            (setq ks-current-path
                  (concat ks-current-path (string (ks--key-to-char char))))
            (cl-incf i)
            (unless alist
              (funcall ks-handler-function char))))
        (cdar alist)))))

(defun ks-read-words (lst words)
  "Select from LST using WORDS."
  (catch 'done
    (let ((num-words (length words))
          (num-entries (length lst))
          alist)
      ;; If there are not enough words to cover all the candidates,
      ;; we use a De Bruijn sequence to generate the remaining ones.
      (when (< num-words num-entries)
        (let ((keys ks-keys)
              (bad-keys '(?a ?e ?i ?o ?u ?y))
              (path-len 1)
              (num-remaining (- num-entries num-words))
              tmp-alist)
          ;; Delete all keys which could lead to duplicates.
          ;; We want at least three keys left to work with.
          (dolist (x bad-keys)
            (when (memq x keys)
              (setq keys (delq ?a keys))))
          (when (< (length keys) 3)
            (signal 'user-error
                    '("Please add more keys to the variable `ks-keys'.")))
          ;; Generate the sequence and add the keys to the existing words.
          (while (not tmp-alist)
            (cl-incf path-len)
            (setq tmp-alist (ks--path-alist-1 lst path-len keys)))
          (while (>= (cl-decf num-remaining) 0)
            (push (mapconcat 'string (caar tmp-alist) nil) (cdr (last words)))
            (setq tmp-alist (cdr tmp-alist)))))
      (dolist (x lst)
        (push (cons (string-to-list (pop words)) x) alist))
      (setq ks-current-path "")
      (while (or (> (length alist) 1)
                 (caar alist))
        (dolist (x (reverse alist))
          (ks--overlay-at-full (reverse (car x)) (cdr x)))
        (let ((char (funcall ks-translate-char-function (read-key))))
          (ks--remove-leading-chars)
          (setq alist
                (delq nil
                      (mapcar (lambda (x)
                                (when (eq (caar x) char)
                                  (cons (cdr (car x)) (cdr x))))
                              alist)))
          (setq ks-current-path
                (concat ks-current-path (string (ks--key-to-char char))))
          (unless alist
            (funcall ks-handler-function char))))
      (cdar alist))))

  (let ((len (length candidates)))
    (cond ((= len 0)
           nil)
          ((and (= len 1) ks-single-candidate-jump)
           (car candidates))
          (t
           (unwind-protect
                (progn
                  (ks--make-backgrounds
                   (ks-window-list))
                  (cond ((eq ks-style 'de-bruijn)
                         (ks-read-de-bruijn
                          candidates ks-keys))
                        ((eq ks-style 'words)
                         (ks-read-words
                          candidates ks-words))
                        (t

                                    (or cleanup-fn #'ks--remove-leading-chars)))))
             (ks--done))))))

(defun ks--last-candidates-cycle (advancer)
  (let* ((ks-last-candidates
          (cl-remove-if-not
           (lambda (x) (equal (cdr x) (selected-window)))
           ks-last-candidates))
         (min-dist
          (apply #'min
                 (mapcar (lambda (x) (abs (- (if (listp (car x)) (caar x) (car x)) (point)))) ks-last-candidates)))
         (pos
          (cl-position-if
           (lambda (x)
             (= (- (if (listp (car x)) (caar x) (car x)) (point)) min-dist))
           ks-last-candidates)))
    (funcall advancer pos ks-last-candidates)))

(defun ks-prev ()

  (interactive)
  (ks--last-candidates-cycle
   (lambda (pos lst)
     (when (> pos 0)
       (let ((candidate (nth (1- pos) lst)))
         (goto-char (if (listp (car candidate)) (caar candidate) (car candidate))))))))

(defun ks-next ()
  "Go to the next candidate of the last `ks-read'."
  (interactive)
  (ks--last-candidates-cycle
   (lambda (pos lst)
     (when (< pos (1- (length lst)))
       (let ((candidate (nth (1+ pos) lst)))
         (goto-char (if (listp (car candidate)) (caar candidate) (car candidate))))))))

(defun ks-process (candidates &optional overlay-fn cleanup-fn)
  "Select one of CANDIDATES using `ks-read'.
Use OVERLAY-FN to visualize the decision overlay.
CLEANUP-FN should take no arguments and remove the effects of
multiple OVERLAY-FN invocations."
  (setq overlay-fn (or overlay-fn (ks--style-fn ks-style)))
  (setq cleanup-fn (or cleanup-fn #'ks--remove-leading-chars))
  (unless (and (consp (car candidates))
               (windowp (cdar candidates)))
    (setq candidates
          (mapcar (lambda (x) (cons x (selected-window)))
                  candidates)))
  (setq ks-last-candidates (copy-sequence candidates))
  (let ((original-cands (copy-sequence candidates))
        (res (ks--process-1 candidates overlay-fn cleanup-fn)))
    (cond
      ((null res)
       (if (and (eq ks-style 'words) candidates)
           (ks-process original-cands overlay-fn cleanup-fn)
         (message "zero candidates")
         t))
      ((eq res 'restart)
       (ks-process original-cands overlay-fn cleanup-fn))
      ;; ignore exit from `ks-handler-function'
      ((eq res 'exit))
      ((eq res 'abort)
       nil)
      (t
       (funcall ks-pre-action res)
       (setq res (car res))
       (let ((action (or ks-action ks-action-oneshot 'ks-action-goto)))
         (funcall action
                  (if (consp res)
                      (car res)
                    res)))
       res))))

               (string (car (last path)))
               'face 'ks-lead-face)))
    (ks--overlay
     str
     (ks-candidate-beg leaf) nil
     (ks-candidate-wnd leaf)
     (lambda (str old-str)
       (cond ((string= old-str "\n")
              (concat str "\n"))
             ;; add padding for wide-width character
             ((eq (string-width old-str) 2)
              (concat str " "))
             (t
              str))))))

(defun ks--overlay-at-full (path leaf)
  "Create an overlay with PATH at LEAF.
PATH is a list of keys from tree root to LEAF.
LEAF is normally ((BEG . END) . WND)."
  (let* ((path (mapcar #'ks--key-to-char path))
         (str (propertize
               (apply #'string (reverse path))
               'face 'ks-lead-face))
         (len (length path))
         (beg (ks-candidate-beg leaf))
         (wnd (cdr leaf))
         end)
    (dotimes (i len)
      (set-text-properties i (1+ i)
                           `(face ,(nth i ks-lead-faces))
                           str))
    (when (eq ks-style 'de-bruijn)
      (setq str (concat
                 (propertize ks-current-path
                             'face 'ks-lead-face-1)
                 str))
      (setq len (length str)))
    (with-selected-window wnd
      (save-excursion
        (goto-char beg)
        (let* ((lep (if (bound-and-true-p visual-line-mode)
                        (save-excursion
                          (end-of-visual-line)
                          (point))
                      (line-end-position)))
               ;; `end-of-visual-line' is bugged sometimes
               (lep (if (< lep beg)
                        (line-end-position)
                      lep))
               (len-and-str (ks--update-offset-and-str len str lep)))
          (setq len (car len-and-str))
          (setq str (cdr len-and-str))
          (setq end (if (= beg lep)
                        (1+ beg)
                      (min (+ beg
                              (if (eq (char-after) ?\t)
                                  1
                                len))
                           lep)))
          (when (and (bound-and-true-p visual-line-mode)
                     (> len (- end beg))
                     (not (eq lep beg)))
            (setq len (- end beg))
            (let ((old-str (apply #'string (reverse path))))
              (setq str
                    (substring
                     (propertize
                      old-str
                      'face
                      (if (= (length old-str) 1)
                          'ks-lead-face
                        'ks-lead-face-0))
                     0 len)))))))
    (ks--overlay
     str beg end wnd
     (lambda (str old-str)
       (cond ((string= old-str "\n")
              (concat str "\n"))
             ((string= old-str "\t")
              (concat str (make-string (max (- tab-width len) 0) ?\ )))
             (t
              ;; add padding for wide-width character
              (if (eq (string-width old-str) 2)
                  (concat str " ")
                str)))))))

(defun ks--overlay-post (path leaf)
  "Create an overlay with PATH at LEAF.
PATH is a list of keys from tree root to LEAF.
LEAF is normally ((BEG . END) . WND)."
  (let* ((path (mapcar #'ks--key-to-char path))
         (str (propertize (apply #'string (reverse path))
                          'face 'ks-lead-face)))
    (when (or ks-highlight-first (> (length str) 1))
      (set-text-properties 0 1 '(face ks-lead-face-0) str))
    (setq str (concat
               (propertize ks-current-path
                           'face 'ks-lead-face-1)
               str))
    (ks--overlay
     str
     (ks-candidate-end leaf) nil
     (ks-candidate-wnd leaf))))

(defun ks--update-offset-and-str (offset str lep)
  "Recalculate the length of the new overlay at point.

OFFSET is the previous overlay length.
STR is the overlay string that we wish to add.
LEP is the line end position.

We want to add an overlay between point and END=point+OFFSET.
When other overlays already exist between point and END, set
OFFSET to be the difference between the start of the first
overlay and point.  This is equivalent to truncating our new
overlay, so that it doesn't intersect with overlays that already
exist."
  (let* ((wnd (selected-window))
         (beg (point))
         (oov (delq nil
                    (mapcar
                     (lambda (o)
                       (and (eq (overlay-get o 'category) 'ks)
                            (eq (overlay-get o 'window) wnd)
                            (overlay-start o)))
                     (overlays-in beg (min (+ beg offset) lep))))))
    (when oov
      (setq offset (- (apply #'min oov) beg))
      (setq str (substring str 0 offset)))
    (let ((other-ov (cl-find-if
                     (lambda (o)
                       (and (eq (overlay-get o 'category) 'ks)
                            (eq (overlay-start o) beg)
                            (not (eq (overlay-get o 'window) wnd))))
                     (overlays-in (point) (min (+ (point) offset) lep)))))
      (when (and other-ov
                 (> (overlay-end other-ov)
                    (+ beg offset)))
        (setq str (concat str (buffer-substring
                               (+ beg offset)
                               (overlay-end other-ov))))
        (setq offset (- (overlay-end other-ov)
                        beg))))
    (cons offset str)))

(defun ks--style-fn (style)
  "Transform STYLE symbol to a style function."

    (pre #'ks--overlay-pre)
    (at #'ks--overlay-at)

    (de-bruijn #'ks--overlay-at-full)

    (t (error "Unexpected style %S" style))))

(defun ks--linum-update-window (_ win)
  "Update line numbers for the portion visible in window WIN."
  (goto-char (window-start win))
  (let ((line (line-number-at-pos))
        (limit (window-end win t))
        (fmt (cond ((stringp linum-format) linum-format)
                   ((eq linum-format 'dynamic)
                    (let ((w (length (number-to-string
                                      (count-lines (point-min) (point-max))))))
                      (concat "%" (number-to-string w) "d")))))
        (width 0)
        (ks-strs (when ks-linum-mode
                   (ks--linum-strings))))
    (run-hooks 'linum-before-numbering-hook)
    ;; Create an overlay (or reuse an existing one) for each
    ;; line visible in this window, if necessary.
    (while (and (not (eobp)) (< (point) limit))
      (let* ((str
              (cond (ks-linum-mode
                     (pop ks-strs))
                    (fmt
                     (propertize (format fmt line) 'face 'linum))
                    (t
                     (funcall linum-format line))))
             (visited (catch 'visited
                        (dolist (o (overlays-in (point) (point)))
                          (when (equal-including-properties
                                 (overlay-get o 'linum-str) str)
                            (unless (memq o linum-overlays)
                              (push o linum-overlays))
                            (setq linum-available (delq o linum-available))
                            (throw 'visited t))))))
        (setq width (max width (length str)))
        (unless visited
          (let ((ov (if (null linum-available)
                        (make-overlay (point) (point))
                      (move-overlay (pop linum-available) (point) (point)))))
            (push ov linum-overlays)
            (overlay-put ov 'before-string
                         (propertize " " 'display `((margin left-margin) ,str)))
            (overlay-put ov 'linum-str str))))
      ;; Text may contain those nasty intangible properties, but that
      ;; shouldn't prevent us from counting those lines.
      (let ((inhibit-point-motion-hooks t))
        (forward-line))
      (setq line (1+ line)))
    (when (display-graphic-p)
      (setq width (ceiling
                   (/ (* width 1.0 (linum--face-width 'linum))
                      (frame-char-width)))))
    (set-window-margins win width (cdr (window-margins win)))))

The window scope is determined by `ks-all-windows'.
When ARG is non-nil, do the opposite of `ks-all-windows'.
BEG and END narrow the scope where candidates are searched.
When BOTTOM-UP is non-nil, display ks candidates from top to bottom"
  (setq ks-action (or ks-action #'identity))
  (let ((ks-style (if ks-linum-mode
                      (progn
                        (message "Goto line:")
                        'ignore)
                    ks-style)))
    (ks-process
     (ks--line-cands arg beg end bottom-up))))

(defun ks-goto-line (&optional arg)
  "Jump to a line start in current buffer.

When ARG is 1, jump to lines currently visible, with the option
to cancel to `goto-line' by entering a number.

When ARG is 4, negate the window scope determined by
`ks-all-windows'.

Otherwise, forward to `goto-line' with ARG."
  (interactive "p")
  (setq arg (or arg 1))
  (if (not (memq arg '(1 4)))
      (progn
        (goto-char (point-min))
        (forward-line (1- arg)))
    (ks-with ks-goto-line
      (let* ((ks-handler-old ks-handler-function)
             (ks-handler-function
              (lambda (char)
                (if (or (< char ?0)
                        (> char ?9))
                    (funcall ks-handler-old char)
                  (let ((line (read-from-minibuffer
                               "Goto line: " (string char))))
                    (when line
                      (ks-push-mark)
                      (save-restriction
                        (widen)
                        (goto-char (point-min))
                        (forward-line (1- (string-to-number line))))
                      (throw 'done 'exit))))))
             (r (ks--line (eq arg 4))))
        (when (and (not (memq r '(t nil))) (eq ks-action #'identity))
          (ks-action-goto r))))))

the cursor
When BOTTOM-UP is non-nil, display ks candidates from top to bottom"

    (setq offset (+ 2 (- offset))))

         (r (ks--line nil (window-start)

		       bottom-up)))

      (delete-region beg end)

      (insert text))))

(defun ks-kill-region (arg)
  "Select two lines and kill the region between them.

The window scope is determined by `ks-all-windows' or
`ks-all-windows-alt' when ARG is non-nil."
  (interactive "P")
  (let ((initial-window (selected-window)))
    (ks-with ks-kill-region
      (let* ((beg (save-selected-window
                    (list (ks--line arg) (selected-window))))
             (end (list (ks--line arg) (selected-window))))
        (cond
          ((not (numberp (car beg)))
           (user-error "Fail to select the beginning of region"))
          ((not (numberp (car end)))
           (user-error "Fail to select the end of region"))
          ;; Restrict operation to same window. It's better if it can be
          ;; different windows but same buffer; however, then the cloned
          ;; buffers with different narrowed regions might cause problem.
          ((not (equal (cdr beg) (cdr end)))
           (user-error "Selected points are not in the same window"))
          ((< (car beg) (car end))
           (save-excursion
             (kill-region
              (car beg)
              (progn (goto-char (car end)) (forward-visible-line 1) (point)))))
          (t
           (save-excursion
             (kill-region
              (progn (goto-char (car beg)) (forward-visible-line 1) (point))
              (car end)))))))
    (select-window initial-window)))

;;;###autoload
(defun ks-setup-default ()
  "Setup the default shortcuts."
  (eval-after-load "isearch"
    '(define-key isearch-mode-map (kbd "C-'") 'ks-isearch)))

(defcustom ks-timeout-seconds 0.5
  "How many seconds to wait for the second char."
  :type 'float)

(defcustom ks-enter-times-out t
  "Whether enter exits ks-goto-char-timer early. If nil it matches newline"
  :type 'boolean)

(defvar ks-text ""
  "Store the input read by `ks--read-candidates'.")

(defun ks--read-candidates (&optional re-builder)
  "Read as many chars as possible and return their occurrences.
At least one char must be read, and then repeatedly one next char
may be read if it is entered before `ks-timeout-seconds'.  DEL
deletes the last char entered, and RET exits with the currently
read string immediately instead of waiting for another char for
`ks-timeout-seconds'.
The format of the result is the same as that of `ks--regex-candidates'.
This function obeys `ks-all-windows' setting.
RE-BUILDER is a function that takes a string and returns a regex.
When nil, `regexp-quote' is used.
If a group is captured, the first group is highlighted.
Otherwise, the whole regex is highlighted."
  (setq ks-text "")
  (let ((re-builder (or re-builder #'regexp-quote))
        char break overlays regex)
    (unwind-protect
         (progn
           (ks--make-backgrounds
            (ks-window-list))
           (while (and (not break)
                       (setq char
                             (read-char (format "%d  char%s: "
                                                (length overlays)
                                                (if (string= ks-text "")
                                                    ks-text
                                                  (format " (%s)" ks-text)))
                                        t
                                        (and (not (string= ks-text ""))
                                             ks-timeout-seconds))))
             ;; Unhighlight
             (dolist (ov overlays)
               (delete-overlay ov))
             (setq overlays nil)
             (cond
               ;; Handle RET
               ((= char 13)
                (if ks-enter-times-out
                    (setq break t)
                  (setq ks-text (concat ks-text (list ?\n)))))
               ;; Handle C-h, DEL
               ((memq char ks-del-last-char-by)
                (let ((l (length ks-text)))
                  (when (>= l 1)
                    (setq ks-text (substring ks-text 0 (1- l))))))
               ;; Handle ESC
               ((= char 27)
                (keyboard-quit))
               (t
                (setq ks-text (concat ks-text (list char)))))
             ;; Highlight
             (when (>= (length ks-text) 1)
               (let ((case-fold-search
                      (or ks-case-fold-search (string= ks-text (downcase ks-text))))
                     found)
                 (ks-dowindows current-prefix-arg
                   (dolist (pair (ks--find-visible-regions
                                  (window-start)
                                  (window-end (selected-window) t)))
                     (save-excursion
                       (goto-char (car pair))
                       (setq regex (funcall re-builder ks-text))
                       (while (re-search-forward regex (cdr pair) t)
                         (unless (not (ks--visible-p (1- (point))))
                           (let* ((idx (if (= (length (match-data)) 4) 1 0))
                                  (ov (make-overlay
                                       (match-beginning idx) (match-end idx))))
                             (setq found t)
                             (push ov overlays)
                             (overlay-put
                              ov 'window (selected-window))
                             (overlay-put
                              ov 'face 'ks-goto-char-timer-face)))))))
                 ;; No matches at all, so there's surely a typo in the input.
                 (unless found (beep)))))
           (nreverse (mapcar (lambda (ov)
                               (cons (cons (overlay-start ov)
                                           (overlay-end ov))
                                     (overlay-get ov 'window)))
                             overlays)))
      (dolist (ov overlays)
        (delete-overlay ov))
      (ks--done))))

(defun ks-push-mark ()
  "Store the current point and window."
  (let ((inhibit-message t))
    (ring-insert ks-ring
                 (cons (point) (selected-window)))
    (unless (region-active-p)
      (push-mark))))

(defun ks-pop-mark ()
  "Jump back to the last location of `ks-push-mark'."
  (interactive)
  (let (res)
    (condition-case nil
        (progn
          (while (not (window-live-p
                       (cdr (setq res (ring-remove ks-ring 0))))))
          (let* ((window (cdr res))
                 (frame (window-frame window)))
            (when (and (frame-live-p frame)
                       (not (eq frame (selected-frame))))
              (select-frame-set-input-focus frame))
            (select-window window)
            (goto-char (car res))))
      (error
       (set-mark-command 4)))))
