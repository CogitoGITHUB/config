(provide 'manifolding-atlas-db)

(defvar manifolding-atlas-db--connection nil
  "DB connection.")

(defun manifolding-atlas-db (&optional _args)
  "Return DB connection, opening it if needed."
  (require 'emacsql)
  (require 'emacsql-sqlite)
  (unless (and manifolding-atlas-db--connection
               (ignore-errors (emacsql-live-p manifolding-atlas-db--connection)))
    (setq manifolding-atlas-db--connection
          (ignore-errors (emacsql-sqlite manifolding-atlas-db-location))))
  manifolding-atlas-db--connection)

(defun manifolding-atlas-db-query (&optional filter)
  "Query notes, filtered by FILTER if given."
  (let* ((db (ignore-errors (manifolding-atlas-db)))
         (db-result (when db
                      (if filter
                          (seq-filter filter (ignore-errors (emacsql db [:select * :from notes])))
                        (ignore-errors (emacsql db [:select * :from notes]))))))
    (if (and db-result (consp db-result))
        db-result
      ;; Fallback: scan vault directly when DB not available or empty (same fallback as elsewhere tonight)
      (let* ((root (my/manifolding-atlas-root-dir))
             (files (ignore-errors (my/manifolding-atlas-db-sync--list-org-files root)))
             notes)
        (dolist (file (or files (directory-files-recursively root "^[^.]" t)))
          (when (and (file-regular-p file)
                     (not (string-match-p "/\\.git/" file))
                     (not (string-match-p "/admin/" file)))
            (let ((id (ignore-errors
                        (with-temp-buffer
                          (insert-file-contents file)
                          (when (re-search-forward "^:ID:[ \t]+\\(.+\\)" nil t)
                            (string-trim (match-string 1)))))))
              (when id
                (let ((title (ignore-errors
                               (with-temp-buffer
                                 (insert-file-contents file)
                                 (when (re-search-forward "^\\*+[ \t]+\\(.*\\)" nil t)
                                   (let ((raw (string-trim (match-string 1))))
                                     (if (string-match "^TODO[ \t]+\\(.*\\)" raw)
                                         (string-trim (match-string 1 raw))
                                       raw)))))))
                  (push (list :id id :title (or title (file-name-nondirectory file))
                              :path file :level 0 :pos 1
                              :tags nil :properties nil :aliases nil :primary-title nil :meta nil)
                        notes))))))
        (if filter
            (seq-filter filter notes)
          notes)))))

(defun manifolding-atlas-db-query-by-property (prop val)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (emacsql db [:select * :from notes :where (= properties $s1)] (format "\"%s\":\"%s\"" prop val)))
      ;; Fallback: scan vault and filter by property (uses same vault-scan as db-query)
      (seq-filter (lambda (n)
                    (let ((props (manifolding-atlas-note-properties n)))
                      (equal (cdr (assoc (upcase prop) props)) val)))
                  (ignore-errors (manifolding-atlas-db-query))))))

(defun manifolding-atlas-db-query-by-ids (ids)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if (and db ids)
        (ignore-errors (emacsql db [:select * :from notes :where (in id $v1)] (vconcat ids)))
      ;; Fallback: scan vault and filter by ids
      (when ids
        (seq-filter (lambda (n) (member (manifolding-atlas-note-id n) ids))
                    (ignore-errors (manifolding-atlas-db-query)))))))

(defun manifolding-atlas-db-query-by-file-path (path)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (emacsql db [:select * :from notes :where (= path $s1)] path))
      ;; Fallback: scan vault and filter by path
      (seq-filter (lambda (n) (string-equal (manifolding-atlas-note-path n) path))
                  (ignore-errors (manifolding-atlas-db-query))))))

(defun manifolding-atlas-db-query-by-tags-some (tags)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (seq-filter (lambda (n) (seq-intersection tags (manifolding-atlas-note-tags n)))
                    (ignore-errors (manifolding-atlas-db-query)))
      ;; Fallback: same filter on vault scan when DB unavailable
      (seq-filter (lambda (n) (seq-intersection tags (manifolding-atlas-note-tags n)))
                  (ignore-errors (manifolding-atlas-db-query))))))

(defun manifolding-atlas-db-query-by-links-some (links)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (emacsql db [:select * :from links :where (in source $v1)] (vconcat (mapcar #'car links))))
      ;; Fallback: links require DB, vault scan has no link index
      (progn
        (message "manifolding-atlas-db-query-by-links-some: DB unavailable, links not indexed")
        nil))))

(defun manifolding-atlas-db-query-links-to (id)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (emacsql db [:select * :from links :where (= dest $s1)] id))
      (progn
        (message "manifolding-atlas-db-query-links-to: DB unavailable")
        nil))))

(defun manifolding-atlas-db-query-links-from (id)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (emacsql db [:select * :from links :where (= source $s1)] id))
      (progn
        (message "manifolding-atlas-db-query-links-from: DB unavailable")
        nil))))

(defun manifolding-atlas-db-get-by-id (id)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (or (car (ignore-errors (emacsql db [:select * :from notes :where (= id $s1)] id)))
            (user-error "manifolding-atlas-db-get-by-id: ID %s not found" id))
      (let ((file (condition-case nil (org-id-find-id-file id) (error nil))))
        (if (and file (file-exists-p file))
            (with-temp-buffer
              (insert-file-contents file)
              (let ((title (when (re-search-forward "^\\*+[ \t]+\\(.*\\)" nil t)
                             (let ((raw (string-trim (match-string 1))))
                               (if (string-match "^TODO[ \t]+\\(.*\\)" raw)
                                   (string-trim (match-string 1 raw))
                                 raw)))))
                (list :id id
                      :title (or title (file-name-nondirectory file))
                      :path file
                      :level 0 :pos 1
                      :tags nil :properties nil
                      :aliases nil :primary-title nil
                      :meta nil)))
          (user-error "manifolding-atlas-db-get-by-id: ID %s not found" id))))))

(defun manifolding-atlas-db-update-file (path)
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (ignore-errors (manifolding-atlas-db--update-file path))
      1)))

(defun manifolding-atlas-db-count-notes ()
  (let ((db (ignore-errors (manifolding-atlas-db))))
    (if db
        (or (caar (ignore-errors (emacsql db [:select (funcall count *) :from notes])))
            (length (ignore-errors (manifolding-atlas-db-query))))
      ;; Fallback: count vault files via same scan as db-query
      (length (ignore-errors (manifolding-atlas-db-query))))))

(defun manifolding-atlas-select-from (prompt candidates &rest _args)
  "Real select-from: completing-read over CANDIDATES titles, return existing or new plist."
  (require 'seq)
  (let* ((titles (mapcar (lambda (n) (or (manifolding-atlas-note-title n) (plist-get n :title) "")) candidates))
         (choice (completing-read (cond ((string-suffix-p ": " prompt) prompt)
                                         ((string-suffix-p ":" prompt) (concat prompt " "))
                                         (t (concat prompt ": ")))
                                   titles nil nil nil nil nil))
         (found (seq-find (lambda (n) (string= (or (manifolding-atlas-note-title n) (plist-get n :title) "") choice)) candidates)))
    (if found
        found
      (list :id nil :title choice :path nil :properties nil :tags nil))))

(cl-defun manifolding-atlas-select (prompt &key require-match filter-fn candidates)
  "Select a note with PROMPT via completing-read.
CANDIDATES defaults to every note in the database (via
`manifolding-atlas-db-query'). FILTER-FN, if given, is applied to
narrow the auto-fetched candidates down first. REQUIRE-MATCH is
passed through to the underlying prompt."
  (let* ((cands (or candidates (ignore-errors (manifolding-atlas-db-query))))
         (cands (if filter-fn (seq-filter filter-fn cands) cands)))
    (manifolding-atlas-select-from prompt cands :require-match require-match)))

(defun manifolding-atlas-db--extract-aliases (properties)
  "Extract aliases from PROPERTIES alist.

Looks for property defined by `manifolding-atlas-buffer-alias-property'.
Handles both quoted aliases (with spaces) and unquoted aliases properly.
Strips org links and emphasis markers from each alias.

The property name is upcased before lookup because PROPERTIES keys
are stored upcased, so a lowercase or mixed-case
`manifolding-atlas-buffer-alias-property' still matches."
  (when-let* ((aliases-str (cdr (assoc (upcase manifolding-atlas-buffer-alias-property)
                                       properties))))
    (setq aliases-str (string-trim aliases-str))
    (let ((result nil)
          (pos 0))
      (while (< pos (length aliases-str))
        ;; Skip whitespace
        (while (and (< pos (length aliases-str))
                    (= (aref aliases-str pos) ? ))
          (setq pos (1+ pos)))
        (when (< pos (length aliases-str))
          (let ((char (aref aliases-str pos)))
            (cond
             ;; Quoted alias - find matching closing quote
             ((= char ?\")
              (let ((end (string-match "\"" aliases-str (1+ pos))))
                (if end
                    (progn
                      (push (manifolding-atlas-db--strip-emphasis
                             (org-link-display-format
                              (substring aliases-str (1+ pos) end)))
                            result)
                      (setq pos (1+ end)))
                  (error "Unmatched quote in %s" manifolding-atlas-buffer-alias-property))))
             ;; Unquoted alias - find next space or end of string
             (t
              (let ((end (or (string-match " " aliases-str pos)
                             (length aliases-str))))
                (push (manifolding-atlas-db--strip-emphasis
                       (org-link-display-format
                        (substring aliases-str pos end)))
                       result)
                (setq pos end)))))))
      (nreverse result))))
