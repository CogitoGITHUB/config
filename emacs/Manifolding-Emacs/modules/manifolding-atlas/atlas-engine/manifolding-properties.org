
#+title: Manifolding Atlas — Live Properties
#+priority: -5

* Live Properties
:PROPERTIES:
:CATEGORY: editing
:END:

Replaces static property values with Org links to canonical or scoped ref notes.
Three ref types: ~'closed~ (shared, deduped), ~'scoped~ (per-entity), ~nil~ (plain string).

** Canonicalize Value

#+begin_src emacs-lisp
(defun my/manifolding-atlas--canonicalize (value)
  "Downcase, dasherize, strip non-alphanumeric for use as a filename."
  (let ((s (downcase value)))
    (setq s (replace-regexp-in-string "[^a-z0-9]+" "-" s))
    (replace-regexp-in-string "^-+\\|-+$" "" s)))
#+end_src

(defun my/manifolding-atlas--note-id-by-path (path)
  "Return the ID of the note at PATH, or nil."
  (when (file-exists-p path)
    (with-temp-buffer
      (insert-file-contents path)
      (org-mode)
      (goto-char (point-min))
      (when (re-search-forward "^:ID:[[:blank:]]+\\(.+\\)" nil t)
        (match-string-no-properties 1)))))

** Tolerant Drawer Access

*** Drawer region locator
Finds the property drawer of the heading at point without org's own
strict parsing — vault notes carry raw transclude lines in drawers.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-region ()
  (save-excursion
    (when (condition-case nil (org-back-to-heading t) (error nil))
      (let* ((beg (line-beginning-position))
             (hend (save-excursion
                     (goto-char beg)
                     (forward-line 1)
                     (if (re-search-forward "^\\*+[ \t]" (point-max) t)
                         (line-beginning-position)
                       (point-max)))))
        (goto-char beg)
        (when (re-search-forward "^:PROPERTIES:[ \t]*$" hend t)
          (let ((a (line-beginning-position 2)))
            (when (re-search-forward "^:END:[ \t]*$" hend t)
              (cons a (line-beginning-position)))))))))
#+end_src

*** Drawer read
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-get (key)
  (save-excursion
    (when-let ((r (my/manifolding-atlas--drawer-region)))
      (goto-char (car r))
      (or (when (re-search-forward
                 (format "^:%s:[[:blank:]]+\\(.*\\)$" (regexp-quote key))
                 (cdr r) t)
            (match-string-no-properties 1))
          (progn
            (goto-char (car r))
            (when (re-search-forward
                   "^#\\+transclude: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
                   (cdr r) t)
              (let ((label (match-string-no-properties 2)))
                (when (string-equal key (my/manifolding-atlas--key-from-label label))
                  (format "[[id:%s][%s]]"
                          (match-string-no-properties 1) label)))))))))
#+end_src

*** Drawer write
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-put (key value)
  (save-excursion
    (let ((r (my/manifolding-atlas--drawer-region)))
      (if r
          (progn
            (goto-char (car r))
            (if (re-search-forward
                 (format "^:%s:[[:blank:]]*\\(.*\\)$" (regexp-quote key))
                 (cdr r) t)
                (progn (beginning-of-line)
                       (delete-region (line-beginning-position)
                                      (line-end-position))
                       (insert (format ":%s: %s" key value)))
              (goto-char (cdr r))
              (beginning-of-line)
              (insert (format ":%s: %s\n" key value))))
        (org-back-to-heading t)
        (end-of-line)
        (forward-line 1)
        (unless (bolp) (insert "\n"))
        (insert (format ":PROPERTIES:\n:%s: %s\n:END:\n" key value))))))
#+end_src

*** org-entry fallbacks
Drawer storage stays transparent: org-entry-get/put fall through to
the tolerant primitives when the standard paths find nothing.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--org-entry-get-fallback (orig-fn pom property &optional inherit literal-nil)
  (or (funcall orig-fn pom property inherit literal-nil)
      (when (and property
                 (or (null pom)
                     (and (integerp pom)
                          (<= (point-min) pom) (<= pom (point-max)))))
        (save-excursion
          (when (integerp pom) (goto-char pom))
          (my/manifolding-atlas--drawer-get
           (if (symbolp property) (symbol-name property) property))))))

(defun my/manifolding-atlas--org-entry-put-fallback (orig-fn pom property value)
  (if (and (or (null pom)
               (and (integerp pom)
                    (<= (point-min) pom) (<= pom (point-max))))
           (my/manifolding-atlas--drawer-region))
      (my/manifolding-atlas--drawer-put
       (if (symbolp property) (symbol-name property) property)
       value)
    (funcall orig-fn pom property value)))

(advice-add 'org-entry-get :around #'my/manifolding-atlas--org-entry-get-fallback)
(advice-add 'org-entry-put :around #'my/manifolding-atlas--org-entry-put-fallback)
#+end_src

(defun my/manifolding-atlas--ensure-ref-note (property-type value &optional _note)
  (when (and value (not (string-equal value "WARNING")))
    (let* ((slug (my/manifolding-atlas--canonicalize value))
           (root (expand-file-name (format "admin/refs/%s/" property-type) (my/manifolding-atlas-root-dir)))
           (path (expand-file-name slug root))
           (id (my/manifolding-atlas--note-id-by-path path))
           (title (format "%s: %s" (my/manifolding-atlas--property-display-name property-type) value)))

          (my/manifolding-atlas--make-id-link id title)
        (make-directory root t)
        (let ((manifolding-atlas-create-default-function (lambda (_) '())))
          (let ((note (manifolding-atlas-create title path
                       :tags nil :properties nil :body nil)))
            (when note
              (my/manifolding-atlas--make-id-link (manifolding-atlas-note-id note) title))))))))

** Scoped To-Plist Factory

*** Cross-buffer property writer
NOTE may live in a different buffer than the one being edited:
locate its heading by file, level, and pos first.
Uses drawer-put directly for live-properties domain's transclude handling.
#+begin_src emacs-lisp
(require 'cl-lib)

(defun my/manifolding-atlas--property-put (note key value)
  "Set property KEY to VALUE in NOTE's file on disk using drawer-put."
  (let* ((file (manifolding-atlas-note-path note))
         (level (manifolding-atlas-note-level note))
         (pos   (manifolding-atlas-note-pos note)))
    (when file
      (with-current-buffer (find-file-noselect file)
        (save-excursion
          (if (and level (> level 0) pos)
              (progn (goto-char pos) (org-back-to-heading t))
            (goto-char (point-min)))
          (my/manifolding-atlas--drawer-put key value))
        (save-buffer)))))
#+end_src

*** Heading property scan
Reads plain drawer entries AND in-drawer transclude lines as
uniform (KEY . VALUE) pairs.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--props-in-region (beg end props)
  "Collect (KEY . VALUE) pairs between BEG/END into PROPS."
  (goto-char beg)
  (while (re-search-forward
          "^:\\([A-Z][A-Z0-9_]*\\):[[:blank:]]+\\(.*\\)$"
          end t)
    (push (cons (match-string-no-properties 1)
                (match-string-no-properties 2))
          props))
  (goto-char beg)
  (while (re-search-forward
          "^#\\+transclude: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
          end t)
    (let ((key (my/manifolding-atlas--key-from-label
                (match-string-no-properties 2))))
      (when key
        (push (cons key
                    (format "[[id:%s][%s]]"
                            (match-string-no-properties 1)
                            (match-string-no-properties 2)))
              props))))
  props)

(defun my/manifolding-atlas--heading-properties ()
  (let (props)
    (when (condition-case nil (progn (org-back-to-heading t) t) (error nil))
      (save-excursion
        (let* ((beg (condition-case nil (progn (org-back-to-heading t) (line-beginning-position)) (error (point-min))))
               (end (save-excursion
                      (goto-char beg)
                      (forward-line 1)
                      (if (re-search-forward "^\\*+[ \t]" (point-max) t)
                          (line-beginning-position)
                        (point-max)))))
          (goto-char beg)
          (when (re-search-forward "^:PROPERTIES:" end t)
            (let ((dr-beg (line-beginning-position 2)))
              (when (re-search-forward "^:END:" end t)
                (setq props
                      (my/manifolding-atlas--props-in-region
                       dr-beg (line-beginning-position) props))))))))
    props))
#+end_src

*** Live transclude insertion
#+begin_src emacs-lisp
(defun my/manifolding-atlas--add-live-transcludes (note)
  (let* ((file (manifolding-atlas-note-path note))
         (level (manifolding-atlas-note-level note))
         (pos   (manifolding-atlas-note-pos note)))
    (when file
      (with-current-buffer (find-file-noselect file)
        (save-excursion
          (cond ((and level (> level 0) pos)
                 (goto-char pos)
                 (org-back-to-heading t))
                (t (goto-char (point-min))))
          (let ((props (my/manifolding-atlas--heading-properties))
                (anchor (or (my/manifolding-atlas--drawer-insert-point)
                            (my/manifolding-atlas--body-anchor))))
            (dolist (pair props)
              (when-let ((m (my/manifolding-atlas--ref-link-value (cdr pair))))
                (let ((id (car m)) (label (cdr m)))
                  (unless (my/manifolding-atlas--live-transclude-p id)
                    (goto-char anchor)
                    (beginning-of-line)
                    (unless (bolp) (insert "\n"))
                    (insert (format "#+transclude: [[id:%s][%s]] :property\n"
                                    id label))))))))
        (save-buffer)))))

(defun my/manifolding-atlas--live-transcludes-after-create (note)
  (when (manifolding-atlas-note-p note)
    (my/manifolding-atlas--add-live-transcludes note))
  note)
(advice-add 'manifolding-atlas-create :filter-return #'my/manifolding-atlas--live-transcludes-after-create)

*** Transclude removal sweep
Drops stale :property transclude lines of this type, keeping REF-ID,
and reports which ids were removed for later cleanup.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--remove-other-type-transcludes (property-type ref-id)
  "Delete PROPERTY-TYPE transclude lines except REF-ID; return removed ids."
  (let (removed)
    (goto-char (point-min))
    (while (re-search-forward
            "^#\\+transclude: \\[\\[id:\\([^]]+\\)\\]\\[" nil t)
      (let ((lid (match-string-no-properties 1)))
        (when (and (my/manifolding-atlas--ref-of-type-p property-type lid)
                   (not (string-equal lid ref-id)))
          (push lid removed)
          (delete-region (line-beginning-position)
                         (1+ (line-end-position))))))
    removed))
#+end_src

(defun my/manifolding-atlas--refresh-property-transclude (property-type drawer-value ref-id)
  (let (removed)
    (save-excursion
      (org-back-to-heading t)
      (my/manifolding-atlas--drawer-delete (my/manifolding-atlas--property-key-from-type property-type))

                       (point-min)))
        (my/manifolding-atlas--insert-property-transclude
         property-type drawer-value ref-id)
        (when marker (set-marker marker nil)))
      (save-buffer))
    removed))

*** Ref usage checks and orphan sweep
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--ref-in-use-p (id)
  (let* ((root (my/manifolding-atlas-root-dir))
         (refs-dir (expand-file-name "admin/refs/" root))
         (files (nconc (directory-files-recursively root "\\.org\\'")
                       (directory-files-recursively root "[^./]+\\'")))
         (in-use nil))
    (dolist (f files)
      (unless in-use
        (let ((exp (expand-file-name f)))
          (when (and (not (file-in-directory-p exp refs-dir))
                     (my/manifolding-atlas-links--ref-id-in-file-p id exp))
            (setq in-use t)))))
    in-use))

(defun my/manifolding-atlas--cleanup-unused-ref (id)
  (when (and id (not (my/manifolding-atlas--ref-in-use-p id)))
    (let ((path (ignore-errors (org-id-find-id-file id))))
      (when (and path
                 (file-exists-p path)
                 (file-in-directory-p
                  path
                  (expand-file-name "admin/refs/" (my/manifolding-atlas-root-dir))))
        (message "my/manifolding-atlas: deleting unused ref %s" path)
        (delete-file path)))))

(defun my/manifolding-atlas--sweep-orphaned-refs ()
  (interactive)
  (let* ((root (my/manifolding-atlas-root-dir))
         (refs-dir (expand-file-name "admin/refs/" root))
         (files (directory-files-recursively refs-dir "[^./]+\\'" t))
         count)
    (dolist (f files)
      (when (file-regular-p f)
        (with-temp-buffer
          (insert-file-contents f)
          (goto-char (point-min))
          (when (re-search-forward "^:ID:[[:blank:]]+\\(.+\\)" nil t)
            (let ((id (string-trim (match-string-no-properties 1))))
              (unless (my/manifolding-atlas--ref-in-use-p id)
                (message "my/manifolding-atlas: deleting unused ref %s" f)
                (delete-file f)
                (setq count (1+ (or count 0)))))))))
    (message "my/manifolding-atlas: swept %d orphaned ref%s"
             (or count 0) (if (and count (= count 1)) "" "s"))))

*** State post-apply
The state is represented ONLY by an in-drawer #+transclude: line, so
this drops the plain drawer entry and refreshes that transcription.
Transclusion rendering is stripped first so the drawer edits see raw
#+transclude: directives; after-save-hook re-renders them.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--apply-state-post (property-type drawer-value ref-id note)
  "Post-apply wrapper for a live state property.
Navigates by NOTE's own path, so it works from any buffer."
  (when-let ((file (manifolding-atlas-note-path note)))
    (with-current-buffer (find-file-noselect file)
      (let ((re-render (bound-and-true-p org-transclusion-mode))
            removed)
        (when re-render (org-transclusion-remove-all))
        (unwind-protect
            (save-excursion
              (let ((pos (manifolding-atlas-note-pos note)))
                (when (and pos (> pos 0)) (goto-char pos)))
              (setq removed
                    (my/manifolding-atlas--refresh-property-transclude
                     property-type drawer-value ref-id)))
          (when re-render
            (ignore-errors (org-transclusion-add-all))))
        (dolist (old-id removed)
          (my/manifolding-atlas--cleanup-unused-ref old-id))))))

(defun my/manifolding-atlas--clear-state-transcribes (property-type _note)
  (save-excursion
    (goto-char (point-min))
    (while (re-search-forward
            "^#\\+transclude: \\[\\[id:\\([^]]+\\)\\]\\[" nil t)
      (let ((lid (match-string-no-properties 1)))
        (when (my/manifolding-atlas--ref-of-type-p property-type lid)
          (delete-region (line-beginning-position)
                         (1+ (line-end-position))))))
    (save-buffer)))

(defun my/manifolding-atlas--create-scoped-note (property-type value parent-id &optional _note)
  (when (and value (not (string-equal value "WARNING")))
    (let* ((slug (my/manifolding-atlas--canonicalize (or parent-id "orphan")))
           (val-slug (my/manifolding-atlas--canonicalize value))

                                                    ("REF_TYPE" . ,(symbol-name property-type))))
                     :body nil)))

             (my/manifolding-atlas--make-id-link (manifolding-atlas-note-id note) value)))))))

*** Per-drawer stripping
Swaps each in-drawer transclude line for a plain `:KEY:` synonym so
org's parser sees ordinary properties; returns a restore cell.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--strip-one-drawer ()
  "Strip transclude lines in the drawer at point; return restore cell."
  (let* ((pbeg (line-beginning-position 2))
         (pm (save-excursion
               (if (re-search-forward "^:END:[ \t]*$" nil t)
                   (copy-marker (line-beginning-position))
                 nil)))
         restore)
    (when pm
      (goto-char pbeg)
      (while (re-search-forward "^[^:].*$" pm t)
        (let* ((text (match-string 0))
               (m (when (string-match
                         "^#\\+transclude: \\[\\[id:\\([^]]+\\)\\]\\[\\([^]]*\\)\\]\\]"
                         text)
                    (let* ((label (match-string-no-properties 2 text))
                           (key (my/manifolding-atlas--key-from-label label)))
                      (when key
                        (list key
                              (match-string-no-properties 1 text)
                              label)))))
               (syn (when m
                      (format ":%s: [[id:%s][%s]]"
                              (nth 0 m) (nth 1 m) (nth 2 m)))))
          (push (cons (point-marker) (cons text syn)) restore)
          (delete-region (line-beginning-position)
                         (1+ (line-end-position)))
          (when syn
            (insert syn "\n"))))
      (goto-char pm)
      (set-marker pm nil))
    restore))
#+end_src

*** Restore pass
Puts the raw transclude lines back exactly where they were.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--restore-stripped-lines (restore)
  (dolist (cell (nreverse restore))
    (goto-char (car cell))
    (when (cddr cell)
      (delete-region (line-beginning-position)
                     (1+ (line-end-position))))
    (insert (cadr cell) "\n")
    (set-marker (car cell) nil)))
#+end_src

** RET on a property -- select/create, open, update or change
Press RET on a property line inside a :PROPERTIES: drawer. If it is
still unset (WARNING), select an existing option or type a new one --
either way the resulting note's file opens directly so writing can
start immediately. If it is already set, choose to update the linked

(defun my/manifolding-atlas--current-property-key ()

  (save-excursion

(defun my/manifolding-atlas--property-refs-as-candidates (property-type)
  "Return the option strings defined as level-2 headings in the
PROPERTY-TYPE blueprint file, excluding the WARNING sentinel."
  (let ((file (my/manifolding-atlas--property-blueprint-file property-type)))

(defvar-local my/manifolding-atlas--picker-callback nil)

(defun my/manifolding-atlas--picker-select ()
  (interactive)
  (unless (org-at-heading-p)
    (user-error "Point must be on an option"))
  (let ((title (string-trim (org-get-heading t t t t)))
        (cb my/manifolding-atlas--picker-callback))
    (kill-buffer)
    (when cb (funcall cb title))))

(defvar my/manifolding-atlas--picker-map
  (let ((m (make-sparse-keymap)))
    (define-key m (kbd "RET") #'my/manifolding-atlas--picker-select)
    (define-key m (kbd "q") (lambda () (interactive) (kill-buffer)))
    m))

(defun my/manifolding-atlas--open-picker (property-type on-select)
  "Open the PROPERTY-TYPE blueprint file in an org-capture buffer and
immediately arm a key-selection jump restricted to level-2 (**)
option headings. Pressing an option's letter selects it, closes this
capture buffer, and calls ON-SELECT with its title -- one keypress,
no separate mark/accept step."
  (let* ((file (my/manifolding-atlas--property-blueprint-file property-type))
         (org-capture-templates
          `(("p" "Pick property option" plain (file ,file) ""
             :unnarrowed t :empty-lines 1))))
    (message "TRACE picker: BEFORE org-capture, current-buffer=%s" (current-buffer))
    (org-capture nil "p")
    (message "TRACE picker: AFTER org-capture, current-buffer=%s" (current-buffer))
    (goto-char (point-min))
    (let ((ks-all-windows nil))
      (ks-jump "^\\*\\* .+$"
               :beg (point-min)
               :end (point-max)
               :action
               (lambda (pt)
                 (goto-char pt)
                 (org-back-to-heading t)
                 (let ((title (string-trim (org-get-heading t t t t))))
                   (message "TRACE picker: selected title=%s, current-buffer=%s BEFORE org-capture-kill" title (current-buffer))
                   (org-capture-kill)
                   (message "TRACE picker: current-buffer=%s AFTER org-capture-kill" (current-buffer))
                   (funcall on-select title)))))))

(defun my/manifolding-atlas--open-ref-file-capture (ref-id)
  "Open REF-ID's file in a plain org-capture buffer for editing."
  (let ((file (ignore-errors (org-id-find-id-file ref-id))))
    (unless (and file (file-exists-p file))
      (user-error "Could not find the reference file for %s" ref-id))
    (message "TRACE ref-capture: BEFORE org-capture file=%s current-buffer=%s" file (current-buffer))
    (let ((org-capture-templates
           `(("n" "Edit ref" plain (file ,file) "" :unnarrowed t :empty-lines 1))))
      (org-capture nil "n"))
    (message "TRACE ref-capture: AFTER org-capture current-buffer=%s" (current-buffer))))

(defun my/manifolding-atlas--property-choose-and-link (property-type key)
  "Open the PROPERTY-TYPE blueprint file so the user can pick an
option with a single keypress. On selection, create/reuse the ref
note for that value and write its id-link as KEY's plain property
VALUE inside the note's existing :PROPERTIES: drawer -- never as a
separate transcluded block -- then open the ref note for editing."
  (message "TRACE choose-link: START property-type=%s key=%s current-buffer=%s" property-type key (current-buffer))
  (let ((marker (point-marker)))
    (my/manifolding-atlas--open-picker
     property-type
     (lambda (final-title)
       (message "TRACE choose-link: on-select fired final-title=%s current-buffer=%s marker-buffer=%s" final-title (current-buffer) (marker-buffer marker))
       (let* ((link (my/manifolding-atlas--ensure-ref-note property-type final-title))
              (ref-id (my/manifolding-atlas--id-from-link link)))
         (unless ref-id
           (user-error "Could not create a reference note for %s" final-title))
         (with-current-buffer (marker-buffer marker)
           (save-excursion
             (goto-char marker)
             (org-entry-put nil key link)
             (save-buffer)))
         (message "TRACE choose-link: about to open ref-file-capture ref-id=%s current-buffer=%s" ref-id (current-buffer))
         (my/manifolding-atlas--open-ref-file-capture ref-id))))))

(defun my/manifolding-atlas--property-pick-via-session (property-type key)
  "Open the real declarative-prompt session for PROPERTY-TYPE/KEY on
the note at point -- the same picker used at note creation time
(letters jump+mark, C-l makes it live, C-e edits the file, RET
advances). PROPERTY-TYPE doubles as the registry key-sym."
  (when (zerop (hash-table-count my/manifolding-atlas-org-prompt--specs))
    (my/manifolding-atlas-org-prompt--register-all))
  (let* ((note (my/manifolding-atlas--note-at-point))
         (spec (gethash property-type my/manifolding-atlas-org-prompt--specs)))
    (unless note
      (user-error "Not on a note (no :ID: on this heading)"))
    (unless spec
      (user-error "No declarative spec registered for %s" key))
    (my/manifolding-atlas-org-prompt--session-begin
     (list (list :spec spec :default nil))
     note
     nil)))

(defun my/manifolding-atlas-property-make-live ()
  "RET on a property: select/create its value and open the resulting
note directly. If the property already has a live value, ask whether
to update the existing note or change the selected option."
  (interactive)
  (let ((key (my/manifolding-atlas--current-property-key)))
    (unless key
      (user-error "Not on a property line"))
    (when (member key '("ID"))
      (user-error "Cannot make the ID property live"))
    (let* ((property-type (intern (downcase (string-replace "_" "-" key))))
           (raw (or (org-entry-get nil key) ""))
           (existing-link (my/manifolding-atlas--ref-link-value raw))
           (existing-id (car existing-link)))
      (if existing-id
          (if (y-or-n-p (format "%s is set -- update the note (y) or change the option (n)? " key))
              (my/manifolding-atlas--open-ref-file existing-id)
            (my/manifolding-atlas--property-choose-and-link property-type key))
        (my/manifolding-atlas--property-choose-and-link property-type key)))))

(defun my/manifolding-atlas-return-dispatch ()
  "RET dispatcher: make the property at point live if on one, else
fall back to normal `org-return' behavior everywhere else."
  (interactive)
  (if (my/manifolding-atlas--current-property-key)
      (my/manifolding-atlas-property-make-live)
    (call-interactively #'org-return)))

(with-eval-after-load 'org
  (define-key org-mode-map (kbd "RET") #'my/manifolding-atlas-return-dispatch))
