(defun my/manifolding-atlas-blueprint--legacy-choices (prompt-name)
  "Legacy regex extraction of choices from elisp prompt PROMPT-NAME."
  (let* ((dir (my/manifolding-atlas-blueprint--prompt-dir))
         (files (directory-files-recursively
                 dir (format "^%s\\.org$" prompt-name))))
    (when files
      (with-temp-buffer
        (insert-file-contents (car files))
        (goto-char (point-min))
        (when (re-search-forward "completing-read" nil t)
          (condition-case nil
              (when (re-search-forward "'(\"" nil t)
                (let ((items (save-excursion
                               (backward-char 2)
                               (read (current-buffer)))))
                  (cl-remove-if (lambda (s) (string-match-p "WARNING" s)) items)))
            (error nil)))))))

(defun my/manifolding-atlas-blueprint--template-props ()
  "Return alist of (template-name . (prop-key ...)) from capture template files."
  (let ((template-dir (my/manifolding-atlas-blueprint--template-dir))
        result)
    (when (file-directory-p template-dir)
      (dolist (file (directory-files template-dir t "\\.org\\'"))
        (let* ((name (file-name-sans-extension (file-name-nondirectory file)))
               (keys (nreverse (my/manifolding-atlas-blueprint--drawer-keys-in-file file))))
          (when keys
            (push (cons name keys) result)))))
    (nreverse result)))
(let ((props-dir (expand-file-name "properties/" (my/manifolding-atlas-blueprint--prompt-dir)))
      (fields '()))
  (when (file-directory-p props-dir)
    (dolist (file (directory-files props-dir t "\\.org\\'"))
      (let ((name (file-name-sans-extension (file-name-nondirectory file))))
        (push (my/manifolding-atlas-blueprint--field-from-prompt (intern name) t)
              fields))))
  (manifolding-atlas-blueprint-define
   'base
   :predicate (lambda (_note) t)
    :fields (nreverse fields)))

(defun my/manifolding-atlas-blueprint--hash-validator ()
  "Validator comparing TEMPLATE_HASH against the current template file."
  (lambda (val note)
    (let* ((name (or (car (manifolding-atlas-note-meta-get-list
                           note 'TEMPLATE 'string))
                     (cdr (assoc "TEMPLATE"
                                 (manifolding-atlas-note-properties note)))))
           (current (if name
                        (my/manifolding-atlas--capture-file-hash name)
                      nil)))
      (cond ((and val current (not (string-equal val current)))
              "template-outdated")
             ((null val) "missing-required")
             (t t)))))
(cl-loop for (template-name . keys) in (my/manifolding-atlas-blueprint--template-props)
         for schema-name = (intern (concat "template-" template-name))
         do (manifolding-atlas-blueprint-define
             schema-name
             :predicate
             `(lambda (note)
                (or (member ,template-name
                            (manifolding-atlas-note-meta-get-list note 'TEMPLATE 'string))
                    (equal ,template-name
                           (cdr (assoc "TEMPLATE"
                                       (manifolding-atlas-note-properties note))))))
             :include 'base
:fields
             `((:key "TEMPLATE_HASH"
                :type string
                :required t
                :validator #'my/manifolding-atlas-blueprint--hash-validator))))
(cl-loop for (template-name . keys) in (my/manifolding-atlas-blueprint--template-props)
             'task
             :predicate (lambda (note) (member "task" (manifolding-atlas-note-tags note)))
             :include 'base
             :fields
(list (my/manifolding-atlas-blueprint--field-from-prompt 'todo)
                    (my/manifolding-atlas-blueprint--field-from-prompt 'priority)))
