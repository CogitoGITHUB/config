#+title: Manifolding Atlas — Module Paths
#+priority: -2

Single source of truth for every in-module filesystem location.
Nothing else in the module should hardcode modules/manifolding-atlas
paths; change them here only.

* Paths

** Eval depth guard
Lexical compilation makes frames cheap, but too-high eval-depth
lets runaway recursion overflow the real C stack before the guard
fires.  4000 is ~2.5x default — generous yet C-safe.
#+begin_src emacs-lisp
(setq max-lisp-eval-depth 4000
      max-specpdl-size 4000)
#+end_src

** Module directories
#+begin_src emacs-lisp
(defun my/manifolding-atlas-module-dir ()
  "Return the manifolding-atlas module root."
  (expand-file-name "manifolding-atlas/"
                    (expand-file-name "modules" (locate-user-emacs-file "Manifolding-Emacs"))))

(defun my/manifolding-atlas-schema-dir ()
  "Return the schema/ directory (prompt definitions)."
  (expand-file-name "schema/" (my/manifolding-atlas-module-dir)))

(defun my/manifolding-atlas-templates-dir ()
  "Return the templates/ directory root."
  (expand-file-name "templates/" (my/manifolding-atlas-module-dir)))

(defun my/manifolding-atlas-template-files-dir ()
  "Return the templates/files/ directory (body templates + doct el)."
  (expand-file-name "files/" (my/manifolding-atlas-templates-dir)))

(defun my/manifolding-atlas-infra-fast-dir ()
  "Return the infra/fast/ directory (fast captures)."
  (expand-file-name "fast/"
                    (expand-file-name "infra/"
                                      (my/manifolding-atlas-module-dir))))
#+end_src
