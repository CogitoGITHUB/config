#+title: Manifolding Atlas Routines Plugin
#+priority: 16

Modular routines system for the atlas, modeled on the mastering plugin.
A routine is an atlas note tagged =routine= whose child headings are
routine entries referencing sources via =SOURCE=, =ROUTINE_ID= and
=ROUTINE_ORDER= properties. Every mutation dual-writes the org drawer
and the SQLite index, so queries run against the DB without re-parsing
org. Chains are routines whose sources are themselves routines.

*** Require
#+begin_src emacs-lisp
(require 'manifolding-atlas-db)
(require 'manifolding-atlas-db-query)
(require 'org-id)
(require 'transient)
#+end_src

*** Vocabulary Constants
Three axes define every routine: its type (atomic/composite/chain),
its life domain (SELF-CARE through CUSTOM), and its temporal rhythm
(HOUR through YEARLY, plus day-parts and seasons).
#+begin_src emacs-lisp
(defconst my/manifolding-atlas-routines--types
  '("atomic" "composite" "chain"))

(defconst my/manifolding-atlas-routines--domains
  '("SELF-CARE" "TRAINING" "NUTRITION" "WORK" "STUDY" "RECOVERY"
    "HOUSE" "ADMINISTRATION" "CUSTOM"))

(defconst my/manifolding-atlas-routines--temporal
  '("HOUR" "DAY-PART" "DAY" "DAILY" "NIGHTLY" "WEEKLY" "MONTHLY"
    "QUARTERLY" "SEASONAL" "YEARLY"))

(defconst my/manifolding-atlas-routines--day-parts
  '("WAKE" "MORNING" "MIDDAY" "AFTERNOON" "EVENING" "NIGHT"))

(defconst my/manifolding-atlas-routines--seasons
  '("SPRING" "SUMMER" "AUTUMN" "WINTER"))
#+end_src

*** Database Tables
Three tables: =routines= indexes routine notes by type/domain/temporal,
=routine_entries= maps entry headings to their parent routine with
ordering, and =routine_sessions= journals each run with done/skip counts.

**** Routines table
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--ensure-table-routines ()
  (emacsql (manifolding-atlas-db)
    [:create :table :if :not :exists routines
     ([(note-id :text :primary-key :not-null)
       (rtype :text :not-null)
       (domain :text)
       (temporal :text)]
      (:foreign-key [note-id] :references notes [id]
       :on-delete :cascade))]))
#+end_src

*** DB Extractor
Indexes ROUTINE_* properties from NOTE-DATA. Routine notes land in
=routines=; entry headings carrying ROUTINE_ID, SOURCE and
ROUTINE_ORDER land in =routine_entries=. Session notes with SESSION_OF
and SESSION_DATE land in =routine_sessions=.

**** Routine row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-routine (note-id props)
  "Index the routine row for NOTE-ID from PROPS."
  (let ((rtype (cdr (assoc "ROUTINE_TYPE" props))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routines :where (= note-id $s1)]
      note-id)
    (when (and rtype
               (not (string-equal rtype "WARNING"))
               (member rtype my/manifolding-atlas-routines--types))
      (emacsql (manifolding-atlas-db)
        [:insert :into routines :values $v1]
        (vector note-id rtype
                (cdr (assoc "ROUTINE_DOMAIN" props))
                (cdr (assoc "ROUTINE_TEMPORAL" props)))))))
#+end_src
**** Entry row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-entry (note-id props)
  "Index the routine_entries row for NOTE-ID from PROPS."
  (let ((rid (cdr (assoc "ROUTINE_ID" props)))
        (source (cdr (assoc "SOURCE" props)))
        (order (cdr (assoc "ROUTINE_ORDER" props))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routine_entries :where (= note-id $s1)]
      note-id)
    (when (and rid source order
               (not (string-equal rid "WARNING"))
               (not (string-equal source "WARNING")))
      (emacsql (manifolding-atlas-db)
        [:insert :into routine_entries :values $v1]
        (vector note-id rid source
                (string-to-number order)
                (cdr (assoc "ROUTINE_GROUP" props)))))))
#+end_src
**** Session row extraction
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract-session (note-id props)
  "Index the routine_sessions row for NOTE-ID from PROPS."
  (let ((sid (cdr (assoc "SESSION_OF" props)))
        (sdate (cdr (assoc "SESSION_DATE" props))))
    (emacsql (manifolding-atlas-db)
      [:delete :from routine_sessions :where (= note-id $s1)]
      note-id)
    (when (and sid sdate
               (not (string-equal sid "WARNING"))
               (not (string-equal sdate "WARNING"))
               (string-match-p "^[0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}$" sdate))
      (emacsql (manifolding-atlas-db)
        [:insert :into routine_sessions :values $v1]
        (vector note-id sid
                (my/manifolding-atlas-routines--datestring-int sdate)
                (string-to-number (or (cdr (assoc "SESSION_DONE" props)) "0"))
                (string-to-number (or (cdr (assoc "SESSION_SKIPPED" props)) "0"))
                (string-to-number (or (cdr (assoc "SESSION_TOTAL" props)) "0")))))))
#+end_src
**** Extractor dispatch and registration
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--extract (_ctx note-data)
  "Index ROUTINE_* properties from NOTE-DATA.
Routine notes land in `routines'; entry headings carrying ROUTINE_ID,
SOURCE and ROUTINE_ORDER land in `routine_entries'."
  (let* ((note-id (plist-get note-data :id))
         (props (plist-get note-data :properties)))
    (my/manifolding-atlas-routines--extract-routine note-id props)
    (my/manifolding-atlas-routines--extract-entry note-id props)
    (my/manifolding-atlas-routines--extract-session note-id props)
    note-data))

(with-eval-after-load 'manifolding-atlas
  (my/manifolding-atlas-routines--ensure-tables)
  (manifolding-atlas-db-register-extractor
   (make-manifolding-atlas-extractor
    :name       'routines
    :version    1
    :priority   85
    :extract-fn #'my/manifolding-atlas-routines--extract)))
#+end_src

*** Schema
Two schemas: =routine= requires ROUTINE_TYPE/DOMAIN/TEMPORAL on notes
tagged =routine= (but not =session=); =routines-session= requires the
SESSION_* properties on notes tagged =session=.

**** Validator factories
Every field validates as required, optionally constrained to a
vocabulary — one factory per shape keeps both schemas terse.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--required-validator ()
  "Validator accepting any non-WARNING value."
  (lambda (val _n)
    (if (or (null val) (equal val "WARNING"))
        "missing-required"
      t)))

(defun my/manifolding-atlas-routines--member-validator (allowed bad-msg)
  "Validator requiring a non-WARNING member of ALLOWED."
  (lambda (val _n)
    (cond ((or (null val) (equal val "WARNING")) "missing-required")
          ((not (member val allowed)) bad-msg)
          (t t))))
#+end_src
**** Routine schema
#+begin_src emacs-lisp
(manifolding-atlas-blueprint-define
 'routine
 :predicate (lambda (note)
              (and (member "routine" (manifolding-atlas-note-tags note))
                   (not (member "session" (manifolding-atlas-note-tags note)))))
 :include 'base
 :fields
 (list
  (list :key "ROUTINE_TYPE" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator
                   my/manifolding-atlas-routines--types "invalid-type"))
  (list :key "ROUTINE_DOMAIN" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator
                   my/manifolding-atlas-routines--domains "invalid-domain"))
  (list :key "ROUTINE_TEMPORAL" :type 'string
        :validate (my/manifolding-atlas-routines--member-validator
                   (my/manifolding-atlas-routines--temporal-values)
                   "invalid-temporal"))))
#+end_src
**** Session schema
#+begin_src emacs-lisp
(manifolding-atlas-blueprint-define
 'routines-session
 :predicate (lambda (note) (member "session" (manifolding-atlas-note-tags note)))
 :include 'base
 :fields
 (list
  (list :key "SESSION_OF" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
  (list :key "SESSION_DATE" :type 'string
        :validate (lambda (val _n)
                    (cond ((or (null val) (equal val "WARNING")) "missing-required")
                          ((not (string-match-p "^[0-9]\\{4\\}-[0-9]\\{2\\}-[0-9]\\{2\\}$" val)) "invalid-date")
                          (t t))))
  (list :key "SESSION_DONE" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
  (list :key "SESSION_SKIPPED" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))
  (list :key "SESSION_TOTAL" :type 'string
        :validate (my/manifolding-atlas-routines--required-validator))))
#+end_src

*** Entry Writing
Entries are appended as child headings with SOURCE, ROUTINE_ID and
ROUTINE_ORDER properties. Optional group labels (Warm-Up through
Recovery) and workout prescriptions (=3x10 @20kg /180s=) are parsed
into WORKOUT_* properties.

**** Prescription format
#+begin_src emacs-lisp
(defconst my/manifolding-atlas-routines--groups
  '("Warm-Up" "Main" "Accessory" "Conditioning" "Cooldown" "Recovery"))

(defconst my/manifolding-atlas-routines--rx-regexp
  "\\`\\s-*\\([0-9]+\\)\\s-*x\\s-*\\([0-9]+\\)\\(?:\\s-*@\\s-*\\([^/]+?\\)\\)?\\(?:\\s*/\\s-*\\([0-9]+\\)\\s-*s?\\)?\\s-*\\'")
#+end_src

**** Entry insertion
Dual-writes: org drawer properties plus immediate routine_entries rows.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entry-insert-row (eid rid source order group)
  "Insert one routine_entries row immediately; tolerate FK timing."
  (condition-case nil
      (emacsql (manifolding-atlas-db)
        [:insert :into routine_entries :values $v1]
        (vector eid rid source order group))
    (error nil)))
#+end_src

**** Group and prescription prompts
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--prompt-details (items)
  "Annotate reordered ITEMS with optional group and prescription.
ITEMS are (id . title) conses; returns (id title group rx) lists."
  (let (out)
    (dolist (item items)
      (let* ((group (completing-read
                     (format "Group for %s (blank = none): " (cdr item))
                     (append my/manifolding-atlas-routines--groups '(""))
                     nil t nil nil ""))
             (rx (read-string
                  (format "Prescription for %s (e.g. 3x10 @20kg /180, blank = none): "
                          (cdr item)))))
        (push (list (car item) (cdr item)
                    (unless (string-empty-p group) group)
                    (unless (string-empty-p rx) rx))
              out)))
    (nreverse out)))
#+end_src
**** Prescription parsing
Unparseable prescriptions are kept verbatim in WORKOUT_RX.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--rx-parts (rx)
  "Return (sets reps load rest) parsed from RX, or nil."
  (when rx
    (when (string-match my/manifolding-atlas-routines--rx-regexp rx)
      (list (match-string 1 rx) (match-string 2 rx)
            (match-string 3 rx) (match-string 4 rx)))))
#+end_src
**** Entry heading text
Builds the full drawer for one entry heading.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entry-text (title eid sid routine-id idx group rx)
  "Return the entry heading text with its :PROPERTIES: drawer."
  (let* ((parts (my/manifolding-atlas-routines--rx-parts rx))
         (sets (nth 0 parts)) (reps (nth 1 parts))
         (load (nth 2 parts)) (rest (nth 3 parts)))
    (unless (or (null rx) (and sets reps))
      (message "Prescription %S stored verbatim as WORKOUT_RX" rx))
    (concat
     (format "** %s\n:PROPERTIES:\n:ID: %s\n:SOURCE: %s\n:ROUTINE_ID: %s\n:ROUTINE_ORDER: %d\n"
             title eid sid routine-id idx)
     (when group (format ":ROUTINE_GROUP: %s\n" group))
     (when sets (format ":WORKOUT_SETS: %s\n" sets))
     (when reps (format ":WORKOUT_REPS: %s\n" reps))
     (when load (format ":WORKOUT_LOAD: %s\n" load))
     (when rest (format ":WORKOUT_REST: %ss\n" rest))
     (when rx (format ":WORKOUT_RX: %s\n" rx))
     ":END:\n")))
#+end_src
**** Entry writer
Dual-writes: org drawer properties plus immediate routine_entries rows.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--write-entries (note items)
  "Append ITEMS as entry child headings of NOTE.
ITEMS are (id title group rx) lists from `--prompt-details'.
Dual-writes: org drawer properties plus immediate routine_entries rows."
  (let* ((file (manifolding-atlas-note-path note))
         (routine-id (manifolding-atlas-note-id note))
         (idx 0))
    (with-current-buffer (find-file-noselect file)
      (goto-char (point-max))
      (dolist (item items)
        (setq idx (1+ idx))
        (unless (bolp) (insert "\n"))
        (let ((eid (org-id-new)))
          (insert (my/manifolding-atlas-routines--entry-text
                   (nth 1 item) eid (nth 0 item) routine-id idx
                   (nth 2 item) (nth 3 item)))
          (org-id-add-location eid file)
          (my/manifolding-atlas-routines--entry-insert-row
           eid routine-id (nth 0 item) idx (nth 2 item))))
      (save-buffer))))
#+end_src

*** Queries
Entries are read from the DB index first (fast), falling back to a
direct org scan when unsynced. Workout detail always reads live from
the routine file.

**** Browsing
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines-all ()
  "Return list of all routine notes known to the DB."
  (let ((ids (mapcar #'car
                     (emacsql (manifolding-atlas-db)
                              [:select [note-id] :from routines]))))
    (seq-map #'manifolding-atlas-db-get-by-id ids)))

(defun my/manifolding-atlas-routines-find ()
  "Browse and open routine notes."
  (interactive)
  (let ((ids (mapcar #'car
                     (emacsql (manifolding-atlas-db)
                              [:select [note-id] :from routines]))))
    (manifolding-atlas-find
     :filter-fn (lambda (note)
                  (member (manifolding-atlas-note-id note) ids)))))
#+end_src
**** Routine selection helpers
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--routine-candidates ()
  "Return alist of (label . note-id) for every known routine."
  (mapcar (lambda (row)
            (cons (format "%s (%s)" (cadr row) (car row)) (car row)))
          (emacsql (manifolding-atlas-db)
                   [:select [routines:note-id notes:title]
                    :from routines
                    :inner :join notes
                    :on (= routines:note-id notes:id)])))

(defun my/manifolding-atlas-routines--select-routine (prompt candidates)
  "Completing-read a routine from PROMPT and CANDIDATES, return its id."
  (cdr (assoc (completing-read prompt (mapcar #'car candidates) nil nil)
              candidates)))

(defun my/manifolding-atlas-routines--select-routine-interactive (prompt)
  "Prompt for a routine and return the note."
  (let ((candidates (my/manifolding-atlas-routines--routine-candidates)))
    (when (null candidates) (user-error "No routines defined yet"))
    (manifolding-atlas-db-get-by-id
     (my/manifolding-atlas-routines--select-routine prompt candidates))))
#+end_src
**** Org fallback reader
Used when the DB index has no rows for a routine.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entries-org (routine)
  "Read entries of ROUTINE directly from org as fallback.
Returns (source-id title order group sets reps load rest rx) lists."
  (let ((entries nil))
    (with-current-buffer
        (find-file-noselect (manifolding-atlas-note-path routine))
      (org-map-entries
       (lambda ()
         (let ((source (org-entry-get nil "SOURCE"))
               (order  (org-entry-get nil "ROUTINE_ORDER"))
               (group  (org-entry-get nil "ROUTINE_GROUP")))
           (when (and source order)
             (push (list source
                         (org-get-heading t t t t)
                         (string-to-number order)
                         group
                         (org-entry-get nil "WORKOUT_SETS")
                         (org-entry-get nil "WORKOUT_REPS")
                         (org-entry-get nil "WORKOUT_LOAD")
                         (org-entry-get nil "WORKOUT_REST")
                         (org-entry-get nil "WORKOUT_RX"))
                   entries))))
       nil nil))
    (sort (nreverse entries)
          (lambda (a b) (< (nth 2 a) (nth 2 b))))))
#+end_src
**** Live workout detail
Workout properties always come from the file, never the stale index.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entry-extras (routine)
  "Hash entry-ID to its (sets reps load rest rx) from ROUTINE's file."
  (let ((map (make-hash-table :test 'equal)))
    (with-current-buffer
        (find-file-noselect (manifolding-atlas-note-path routine))
      (org-map-entries
       (lambda ()
         (let ((id (org-entry-get nil "ID")))
           (when id
             (puthash id
                      (list (org-entry-get nil "WORKOUT_SETS")
                            (org-entry-get nil "WORKOUT_REPS")
                            (org-entry-get nil "WORKOUT_LOAD")
                            (org-entry-get nil "WORKOUT_REST")
                            (org-entry-get nil "WORKOUT_RX"))
                      map))))
       nil nil))
    map))
#+end_src
**** Row expansion
Joins one DB row with its live workout extras and source title.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--expand-entry-row (row extras)
  "Expand one DB ROW with EXTRAS into an entry list."
  (let* ((eid (nth 0 row))
         (sid (nth 1 row))
         (extra (or (gethash eid extras) '(nil nil nil nil nil)))
         (snote (manifolding-atlas-db-get-by-id sid)))
    (list sid
          (if snote
              (manifolding-atlas-note-title snote)
            sid)
          (nth 2 row)
          (nth 3 row)
          (nth 0 extra)
          (nth 1 extra)
          (nth 2 extra)
          (nth 3 extra)
          (nth 4 extra))))
#+end_src
**** Entries dispatcher
Ordering and groups come from the DB index; workout detail is read
live from the routine file.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--entries (routine)
  "Return ordered entries of ROUTINE as
(source-id title order group sets reps load rest rx)."
  (let* ((rid (manifolding-atlas-note-id routine))
         (rows (condition-case nil
                   (emacsql (manifolding-atlas-db)
                            [:select [note-id source-id entry-order entry-group]
                             :from routine_entries
                             :where (= routine-id $s1)
                             :order :by entry-order]
                            rid)
                 (error nil))))
    (if rows
        (let ((extras (my/manifolding-atlas-routines--entry-extras routine)))
          (mapcar (lambda (row)
                    (my/manifolding-atlas-routines--expand-entry-row row extras))
                  rows))
      (my/manifolding-atlas-routines--entries-org routine))))
#+end_src

*** Session Journaling
Each run creates or updates a session note under
=sessions/<Y>/<Q>/<m>/<W>/<date>-<slug>= with per-step results.
Idempotent per day: same routine + date rewrites in place, preserving ID.

**** Date conversion
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--dateint-days (int)
  "Convert YYYYMMDD INT to org day-number."
  (time-to-days
   (encode-time 0 0 0 (% int 100) (% (/ int 100) 100) (/ int 10000))))

(defun my/manifolding-atlas-routines--dateint-string (int)
  "Format YYYYMMDD INT as YYYY-MM-DD."
  (format "%04d-%02d-%02d" (/ int 10000) (% (/ int 100) 100) (% int 100)))
#+end_src
**** Session lookup and path
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--session-note-for (routine-id date-int)
  "Return today's existing session note for ROUTINE-ID on DATE-INT, or nil."
  (let ((row (caar (emacsql (manifolding-atlas-db)
                     [:select [note-id] :from routine_sessions
                      :where (and (= routine-id $s1)
                                  (= session-date $s2))]
                     routine-id date-int))))
    (when row (manifolding-atlas-db-get-by-id row))))

(defun my/manifolding-atlas-routines--session-path (date slug)
  "Journal-style absolute path for a session of SLUG on ISO DATE:
sessions/<year>/<quarter>/<month>/<week>/<date>-<slug>."
  (let* ((parsed (date-to-time date))
         (year (format-time-string "%Y" parsed))
         (quarter (format "Q%d" (1+ (/ (1- (string-to-number
                                        (substring date 5 7))) 3))))
         (month (format-time-string "%Y-%m" parsed))
         (week (format-time-string "W%V" parsed))
         (dir (expand-file-name
               (format "%s/%s/%s/%s/" year quarter month week)
               my/manifolding-atlas-routines-sessions-directory)))
    (make-directory dir t)
    (expand-file-name (format "%s-%s" date slug) dir)))

*** Streaks
Temporal-aware streak counting: the period unit derives from the
routine's TEMPORAL property. A streak survives if the current or
previous period has a session.

**** Period mapping and bucketing
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--streak-unit (temporal)
  "Map TEMPORAL context to a streak period unit."
  (pcase temporal
    ("WEEKLY" 'week)
    ("MONTHLY" 'month)
    ((or "QUARTERLY" "SEASONAL") 'quarter)
    ("YEARLY" 'year)
    (_ 'day)))

(defun my/manifolding-atlas-routines--bucket (date-int unit)
  "Bucket YYYYMMDD DATE-INT into an ordinal of UNIT."
  (let* ((y (/ date-int 10000))
         (m (% (/ date-int 100) 100))
         (d (% date-int 100))
         (days (time-to-days (encode-time 0 0 0 d m y))))
    (pcase unit
      ('week (/ days 7))
      ('month (+ (* y 12) m))
      ('quarter (+ (* y 4) (1+ (/ (1- m) 3))))
      ('year y)
      (_ days))))
#+end_src
**** Streak counting
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--streak-for (routine-id &optional temporal)
  "Consecutive-period streak of ROUTINE-ID ending this or the previous
period. The period derives from TEMPORAL (day by default; week, month,
quarter/season, or year for the matching temporal contexts)."
  (let* ((unit (my/manifolding-atlas-routines--streak-unit temporal))
         (buckets (delete-dups
                   (sort (mapcar (lambda (row)
                                   (my/manifolding-atlas-routines--bucket
                                    (nth 0 row) unit))
                                 (my/manifolding-atlas-routines--sessions-for
                                  routine-id))
                         #'>)))
         (today-int (my/manifolding-atlas-routines--datestring-int
                     (my/manifolding-atlas-routines--today-string)))
         (anchor (my/manifolding-atlas-routines--bucket today-int unit))
         (streak 0))
    (when buckets
      (cond ((= (car buckets) anchor))
            ((= (car buckets) (1- anchor)) (setq anchor (car buckets)))
            (t (setq buckets nil)))
      (while (and buckets (= (car buckets) anchor))
        (setq streak (1+ streak)
              anchor (1- anchor)
              buckets (cdr buckets))))
    streak))

  "Append ordering problems for TITLE's ENTRIES to REPORT."
  (let ((orders (mapcar (lambda (e) (nth 2 e)) entries)))
    (when (/= (length orders) (length (delete-dups orders)))
      (push (format "%s: duplicate ROUTINE_ORDER values" title) report))
    (let ((sorted (sort (copy-sequence orders) #'<)))
      (unless (equal sorted (number-sequence 1 (length sorted)))
        (push (format "%s: non-contiguous ordering %s" title sorted)
              report))))
  report)
#+end_src
**** Per-routine validation
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--validate-one (rid rtype report)
  "Validate routine RID; return extended REPORT."
  (let* ((note (manifolding-atlas-db-get-by-id rid))
         (title (if note (manifolding-atlas-note-title note) rid))
         (entries (condition-case err
                      (my/manifolding-atlas-routines--entries-org note)
                    (error
                     (push (format "%s: cannot read entries (%s)" title err)
                           report)
                     nil))))
    (unless note
      (push (format "%s: routine row without resolvable note" rid) report))
    (when entries
      (dolist (entry entries)
        (unless (manifolding-atlas-db-get-by-id (nth 0 entry))
          (push (format "%s: unresolvable source id:%s" title (nth 0 entry))
                report)))
      (setq report (my/manifolding-atlas-routines--check-orders
                    title entries report))
      (when (equal rtype "chain")
        (dolist (entry entries)
          (unless (caar (emacsql (manifolding-atlas-db)
                          [:select [note-id] :from routines
                           :where (= note-id $s1)]
                          (nth 0 entry)))
            (push (format "%s: chain member id:%s is not a routine"
                          title (nth 0 entry))
                  report)))))
    report))
#+end_src

(defun my/manifolding-atlas-routines-validate ()
  "Validate all routines and report problems."
  (interactive)

        (count 0))
    (dolist (row (emacsql (manifolding-atlas-db)
                   [:select [note-id rtype] :from routines]))

        (insert line "\n"))
      (display-buffer (current-buffer)))
    (message "Validated %d routines, %d problem(s)" count (length report))))

*** Run Session
Step-by-step execution in a review buffer. Steps that are in the
mastering pool show their review status; due steps can be graded with
l (learn) or r (review-only) before being practiced. At the end, offers
to journal the session.

**** Step display
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--display-step (buf step total title sid badge)
  (with-current-buffer buf
    (let ((inhibit-read-only t))
      (erase-buffer)
      (insert (format "[%d/%d] %s\n\n" step total title))
      (when badge
        (insert (format "mastering: %s\n" badge)))
      (insert (format "[[id:%s][open]]\n" sid)))))
#+end_src
**** Open-note followup prompt
After opening a step's note, the user is asked again; returns
'done-learn, 'done, 'skip, 'quit, or nil when there is no note.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--open-followup (snote sid due-p)
  "Open SNOTE for SID and ask what to do next."
  (when snote
    (find-file (manifolding-atlas-note-path snote))
    (let* ((g-choices (if due-p '(?l ?r ?d ?s ?q)
                        '(?d ?s ?q)))
           (g (read-char-choice
               (if due-p
                   "l=learn d=done s=skip q=quit session: "
                 "d=done s=skip q=quit session: ")
               g-choices)))
      (pcase g
        (?l 'done-learn)
        (?d 'done)
        (?s 'skip)
        (?q 'quit)))))
#+end_src
**** Single-step handler
Returns (RESULT . GRADED-P), or 'quit, or nil when nothing was
recorded for this step.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--run-step (buf entry step total)
  "Interactively run ENTRY as step STEP of TOTAL in review BUF."
  (let* ((sid (nth 0 entry))
         (stitle (nth 1 entry))
         (snote (manifolding-atlas-db-get-by-id sid))
         (mastery (my/manifolding-atlas-routines--mastery-status sid))
         (badge (car mastery))
         (due-p (cdr mastery))
         (choices (if due-p '(?l ?r ?d ?s ?o ?q) '(?d ?s ?o ?q)))
         (prompt
          (concat (format "[%d/%d] %s%s — "
                          step total stitle
                          (if badge (format " [%s]" badge) ""))
                  (if due-p
                      "l=learn+done r=review-only d=done s=skip o=open q=quit: "
                    "d=done s=skip o=open q=quit: "))))
    (my/manifolding-atlas-routines--display-step
     buf step total stitle sid badge)
    (pcase (read-char-choice prompt choices)
      (?l (my/manifolding-atlas-routines--grade-step sid) '(done . t))
      (?r (my/manifolding-atlas-routines--grade-step sid)
          (message "Reviewed without marking practice.") nil)
      (?d '(done . nil))
      (?s '(skip . nil))
      (?o (let ((res (my/manifolding-atlas-routines--open-followup
                      snote sid due-p)))
            (cond ((eq res 'quit) 'quit)
                  ((eq res 'done-learn)
                   (my/manifolding-atlas-routines--grade-step sid)
                   '(done . t))
                  (res (cons res nil))
                  (t nil))))
      (?q 'quit))))
#+end_src
**** Run command
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines-run (&optional routine)
  "Execute ROUTINE (or prompt for one) step by step in a review buffer.
Steps that are in the mastering pool show their review status; due
steps can be graded with r (learn) before being practiced.  At the
end, offer to journal the session."
  (interactive)
  (let* ((routine (or routine
                      (my/manifolding-atlas-routines--select-routine-interactive
                       "Run routine: ")))
         (entries (my/manifolding-atlas-routines--entries routine)))
    (unless entries (user-error "Routine has no entries"))
    (let ((buf (get-buffer-create "*manifolding-atlas-routines-run*"))
          (total (length entries))
          (done 0)
          (skipped 0)
          (learned 0)
          (step 0)
          (results nil))
      (with-current-buffer buf
        (special-mode)
        (setq buffer-read-only t))
      (pop-to-buffer-same-window buf)
      (catch 'routines-quit
        (dolist (entry entries)
          (setq step (1+ step))
          (let ((res (my/manifolding-atlas-routines--run-step
                      buf entry step total)))
            (cond
             ((eq res 'quit) (throw 'routines-quit nil))
             (res
              (pcase (car res)
                ('done (setq done (1+ done)))
                ('skip (setq skipped (1+ skipped))))
              (when (cdr res) (setq learned (1+ learned)))
              (push (list (nth 0 entry) (nth 1 entry) (car res))
                    results)))))))
        (kill-buffer buf)
        (when (and results
                   (y-or-n-p (format "Log session (%d done / %d skipped / %d learned)? "
                                     done skipped learned)))
          (my/manifolding-atlas-routines--session-create
           routine (nreverse results) done skipped))
        (message "Routine finished: %d done, %d skipped, %d learned"
                 done skipped learned)))
#+end_src

(defun my/manifolding-atlas-routines--add-entry-source ()
  "Return (id . title) for a new entry source."
  (if (and (derived-mode-p 'org-mode)
           (org-at-heading-p)
           (buffer-file-name)
           (file-in-directory-p
            (buffer-file-name)
            (my/manifolding-atlas-root-dir)))
      (progn (org-id-get-create)
             (when buffer-file-name (save-buffer))
             (cons (org-id-get) (org-get-heading t t t t)))
    (let* ((notes (manifolding-atlas-db-query))
           (cands (mapcar (lambda (n)
                            (cons (manifolding-atlas-note-title n)
                                  (manifolding-atlas-note-id n)))
                          notes))
           (choice (completing-read "Entry note: "
                                    (mapcar #'car cands) nil t)))
      (cons (cdr (assoc choice cands)) choice))))

**** Order correction
write-entries assigns sequential orders; a manual append forces the
next free order instead.
#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--fix-entry-order (routine next-order)
  "Rewrite the last entry's ROUTINE_ORDER on ROUTINE to NEXT-ORDER."
  (with-current-buffer (find-file-noselect
                        (manifolding-atlas-note-path routine))
    (goto-char (point-max))
    (when (re-search-backward ":ROUTINE_ORDER: [0-9]+" nil t)
      (let ((inhibit-read-only t))
        (delete-region (match-beginning 0) (match-end 0))
        (insert (format ":ROUTINE_ORDER: %d" next-order))))
    (save-buffer)))
#+end_src

(defun my/manifolding-atlas-routines-add-entry (&optional routine)
  "Append an entry to ROUTINE (or prompt for one).
Source is the heading at point when in a vault note, else a
vault-wide selection.  Regenerates the rendered file afterwards."
  (interactive)
  (let* ((routine (or routine
                      (my/manifolding-atlas-routines--select-routine-interactive
                       "Add entry to routine: ")))
         (sid (my/manifolding-atlas-routines--add-entry-source))
         (existing (my/manifolding-atlas-routines--entries-org routine))
         (next-order (1+ (or (car (sort (mapcar (lambda (e) (nth 2 e)) existing)

                             0))))

     routine (list (list (car sid) (cdr sid) nil nil)))
    (my/manifolding-atlas-routines--fix-entry-order routine next-order)
    (let ((note routine))
      (my/manifolding-atlas-routines-generate note)
      (message "Entry added to %s" (manifolding-atlas-note-title routine)))
    routine))

#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines--clear-entry-headings (note)
  "Delete all child entry headings of NOTE."
  (with-current-buffer (find-file-noselect
                        (manifolding-atlas-note-path note))
    (goto-char (point-min))
    (while (re-search-forward "^\\*\\* " nil t)
      (org-back-to-heading t)
      (let ((beg (point)))
        (org-end-of-subtree t t)
        (delete-region beg (point))))
    (save-buffer)))

#+begin_src emacs-lisp
(defun my/manifolding-atlas-routines-due-reviews ()
  "Create or refresh the \"Due Reviews\" composite routine from
today's due mastering notes.  Stable ID across refreshes; renders
like any other routine and appears in streaks/report."
  (interactive)
  (let ((ids (my/manifolding-atlas-mastering--due-note-ids)))
    (when (null ids) (user-error "Nothing due today"))
    (let* ((title "Due Reviews")
           ;; Reuse an existing Due Reviews routine if present.
           (rows (emacsql (manifolding-atlas-db)
                          [:select [routines:note-id notes:title]
                           :from routines
                           :inner :join notes :on (= routines:note-id notes:id)]))
           (rid (cl-loop for row in rows
                         when (string-equal (cadr row) title)
                         return (car row)))
           (note (if rid
                     (manifolding-atlas-db-get-by-id rid)
                   (my/manifolding-atlas--create-file-wrapper
                    title
                    (concat "routines/due-reviews")
                    '("routine")
                    '(("ROUTINE_TYPE" . "composite"))
                    "")))
           (items (mapcar (lambda (id)
                            (let ((n (manifolding-atlas-db-get-by-id id)))
                              (cons id (if n (manifolding-atlas-note-title n)
                                         id))))
                          ids))
           (ordered (my/manifolding-atlas-routines--reorder items))
           (details (my/manifolding-atlas-routines--prompt-details ordered)))
      (unless details (user-error "Aborted"))
      (my/manifolding-atlas-routines--clear-entry-headings note)
      (my/manifolding-atlas-routines--write-entries note details)
      (my/manifolding-atlas-routines-generate note)
      (message "Due Reviews refreshed (%d entries)" (length details))
      note)))
#+end_src
