*** Note Contexts
#+begin_src emacs-lisp
(defun my/manifolding-atlas--note-contexts (note)
  "Return contexts applicable to NOTE.
If the note carries a tag matching a specialized registered context
(contact, dictionary, rule), return only those specialized contexts so
the menu adapts to the note's kind.  Otherwise return the base
'file/'heading context derived from the note's level."
  (let* ((level (manifolding-atlas-note-level note))
         (base (if (and level (> level 0)) 'heading 'file))
         (tags (manifolding-atlas-note-tags note))
         (registered (delete-dups
                      (seq-mapcat (lambda (s) (plist-get s :contexts))
                                  my/manifolding-atlas-prompt-registry)))
         (specialized (cl-remove-if-not (lambda (c) (member (symbol-name c) tags))
                                        (cl-remove 'file (cl-remove 'heading registered)))))
    (if specialized specialized (list base))))

(defun my/manifolding-atlas--update-prompt-spec (key)
  "Find prompt spec with KEY matching the current note context."
  (seq-find (lambda (s)
              (and (eq (plist-get s :key) key)

                    my/manifolding-atlas--current-contexts

**** Menu line formatting
Strips Org link brackets so values stay readable; WARNING stays visible.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--clean-display-val (val)
  "Return clean display for property value VAL."
  (cond
   ((null val) "—")
   ((string-empty-p val) "—")
   ((string-match "\\]\\[\\([^]]+\\)\\]" val)
    (match-string-no-properties 1 val))
   (t val)))

(defun my/manifolding-atlas--update-short-label (spec)
  "Return SPEC's display label without any KEY: prefix."
  (let ((label (or (plist-get spec :label) "")))
    (if (string-match "^[^:]+: \\(.*\\)" label)
        (match-string 1 label)
      label)))

(defun my/manifolding-atlas--update-format-prompt (key)
  "Return description string for prompt KEY: toggle state, label, current value."
  (let* ((spec (my/manifolding-atlas--update-prompt-spec key))
         (short (my/manifolding-atlas--update-short-label spec))
         (current (when-let ((fn (plist-get spec :read-current-fn)))
                    (condition-case nil
                        ;; read-current-fn may run while a non-Org
                        ;; buffer is current (e.g. dired captures) —
                        ;; org-element warns instead of signaling, so
                        ;; hush just that type around the call.
                        (let ((warning-suppress-types
                               (cons '(org-element)
                                     warning-suppress-types)))
                          (funcall fn))
                      (error nil))))
         (display (my/manifolding-atlas--clean-display-val
                   (pcase current
                     ((pred listp) (string-join current ", "))
                     ((pred stringp) current)
                     (_ nil))))
         (mark (if (memq key my/manifolding-atlas--selected-keys) "✓" " ")))
    (format "[%s] %-15s %s" mark short display)))

*** Dynamic Suffixes

**** Capture grid — entries
Dedicated buffer engine for capture AND updates: we own all input,
so SPC and picking keys can never conflict. Rows are plain display;
REAL avy overlays handle jumping (normal avy behavior — any number
of rows, automatic narrowing, no code bookkeeping).
#+begin_src emacs-lisp
(defun my/manifolding-atlas--capture-grid-entries ()
  "Build grid entries: plists (:key :label :selected :pos).
Starting selection depends on mode: captures open ALL-selected
(first avy pick then SOLO-selects), updates open UNSELECTED."
  (let* ((prompts (seq-filter
                   (lambda (s)
                     (let ((k (plist-get s :key)))
                       (and (not (memq k '(position)))
                            (plist-get s :read-current-fn)
                            (plist-get s :prompt-fn)
                            (seq-intersection
                             my/manifolding-atlas--current-contexts
                             (plist-get s :contexts) #'eq))))
                    my/manifolding-atlas-prompt-registry)))
    (mapcar
     (lambda (s)
       (list :key (plist-get s :key)
             :label (my/manifolding-atlas--update-short-label s)
             :selected (and my/manifolding-atlas--capture-selection-mode t)
             :pos nil))
     prompts)))
#+end_src

(defun my/manifolding-atlas--capture-grid-line (e)
  "Render entry E as one grid line (gutter for avy overlays)."
  (format "   [%s] %s"
          (if (plist-get e :selected) "✓" " ")

(defun my/manifolding-atlas--capture-grid-render (buf entries)
  "Paint the whole grid BUF from ENTRIES."
  (with-current-buffer buf
    (erase-buffer)
    (insert "Properties — letters jump+toggle · SPC all/none · RET done · q cancel\n\n")
    (dolist (e entries)

      (insert (my/manifolding-atlas--capture-grid-line e) "\n"))
    (goto-char (point-min))))

  "Repaint one entry's line in place."
  (save-excursion
    (goto-char (plist-get e :pos))
    (delete-region (point) (line-end-position))
    (insert (my/manifolding-atlas--capture-grid-line e))))

(defun my/manifolding-atlas--capture-grid-refresh (entries buf)
  "Repaint every entry and re-point at the grid top."
  (with-current-buffer buf
    (dolist (e entries)
      (my/manifolding-atlas--capture-grid-redraw e))

*** Transient window cleanup
Killing a transient buffer without managing its window makes that window revert to its previous buffer (often Dired), duplicating it.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--kill-buffer-reuse-window (buf &optional note-buf)
  "Kill BUF. For windows showing it: place NOTE-BUF into exactly one
of them if it's the last window on its frame, delete the rest."
  (when (buffer-live-p buf)
    (let ((placed nil))
      (dolist (win (get-buffer-window-list buf nil t))
        (when (window-live-p win)
          (if (and (not placed)
                   (= 1 (length (window-list (window-frame win) 'no-minibuf)))
                   note-buf (buffer-live-p note-buf))
              (progn (set-window-buffer win note-buf) (setq placed t))
            (ignore-errors (delete-window win)))))
      (kill-buffer buf))))
#+end_src

#+begin_src emacs-lisp
(defun my/manifolding-atlas--run-update-key (key)
  "Run prompt KEY against the current context and apply its fragment."
  (let (spec val frag)
    (setq spec (my/manifolding-atlas--update-prompt-spec key))
    (setq val (condition-case nil
                 (funcall (plist-get spec :prompt-fn))
               (quit nil)))
    ;; skip the WARNING sentinel — user didn't change anything
    (when (and (stringp val) (string-equal val "WARNING"))
      (setq val nil))
    (setq frag (funcall (plist-get spec :to-plist) val
                        my/manifolding-atlas--current-context))
    (when frag
      (my/manifolding-atlas--apply-fragment frag spec))))

**** Buffer fallback
Prefers the note under the cursor, falls back to the deepest note at

#+begin_src emacs-lisp
(defun my/manifolding-atlas--buffer-note ()
  "Return the manifolding-atlas note for the current buffer, or nil."
  (or (my/manifolding-atlas--note-at-point)
      (when-let ((file (buffer-file-name))
                 (notes (manifolding-atlas-db-query-by-file-path (file-truename file))))
        (let ((pos-point (point)))
          (or (let ((best nil))
                (dolist (n notes best)
                  (let ((npos (manifolding-atlas-note-pos n)))
                    (when (and npos (<= npos pos-point)
                               (or (null best) (> npos (manifolding-atlas-note-pos best))))
                      (setq best n)))))
              (car notes))))))

*** Property Drawer Fallback
Schema validation reads from the description list (meta), but we store
state values in the property drawer.  This advice makes validation fall
back to the property drawer when no description list value exists.

**** Drawer value resolution
Visits the note's file only when the current buffer isn't already it.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--drawer-value-for-note (note key)
  "Return KEY's drawer value for NOTE from whichever buffer holds it."
  (let* ((file (manifolding-atlas-note-path note)))
    (if (and (buffer-file-name)
             (equal (file-truename (buffer-file-name))
                    (file-truename file)))
        ;; current buffer is already the note's file
        (save-excursion
          (when-let ((pos (manifolding-atlas-note-pos note)))
            (ignore-errors (goto-char pos)))
          (org-entry-get nil key))
      ;; visit the note's file
      (with-current-buffer (find-file-noselect file)
        (save-excursion
          (when-let ((pos (manifolding-atlas-note-pos note)))
            (ignore-errors (goto-char pos)))
          (org-entry-get nil key))))))

(defun my/manifolding-atlas--meta-get-fallback (orig-fun note prop &optional type)
  "Fall back to the property drawer when the description list has no value for PROP.
Skips non-string types to avoid interfering with note/link resolution."
  (or (funcall orig-fun note prop type)
      (when (memq type '(nil string))
        (when-let* ((file (manifolding-atlas-note-path note))
                    (key (if (symbolp prop) (symbol-name prop) prop))
                    (val (my/manifolding-atlas--drawer-value-for-note note key)))
          (unless (string-empty-p val)
            (list val))))))

(advice-add 'manifolding-atlas-note-meta-get-list :around #'my/manifolding-atlas--meta-get-fallback)

*** Fix Violations Suffix

**** Schema field collection
Builds the key→field lookup from every schema applicable to NOTE.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--schema-field-lookup (note)
  "Return (ALL-FIELDS . LOOKUP) for all schema fields of NOTE."
  (let ((all-fields
         (cl-loop for name in (manifolding-atlas-schema-applicable note)
                  append (manifolding-atlas-schema-fields (manifolding-atlas-schema-get name))))
        (lookup (make-hash-table :test 'equal)))
    (dolist (f all-fields)
      (puthash (plist-get f :key) f lookup))
    (cons all-fields lookup)))
#+end_src

**** Schema field repair
Prompts once per missing or WARNING-valued field; one-of fields get
completion, free fields get read-string.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--fix-schema-fields (schema-keys field-lookup)
  "Prompt for each missing/WARNING schema field. Returns count fixed."
  (let ((prompted 0))
    (dolist (key schema-keys)
      (let* ((val (org-entry-get nil key))
             (field (gethash key field-lookup)))
        (when (or (null val) (and (stringp val) (string-match-p "WARNING" val)))
          (condition-case nil
              (let* ((one-of (plist-get field :one-of))
                     (new-val (if one-of
                                  (completing-read
                                   (format "%s: " key) one-of nil t nil nil)
                                (read-string (format "%s: " key)))))
                (unless (string-empty-p new-val)
                  (org-entry-put nil key new-val)
                  (setq prompted (1+ prompted))))
            (quit nil)))))
    prompted))
#+end_src

**** Legacy WARNING property repair
Non-schema properties still carrying the old sentinel get one last
chance before being left visible.
#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--fix-warning-props (schema-keys)
  "Prompt for non-schema drawer entries with WARNING values."
  (let ((prompted 0))
    (dolist (pair (org-entry-properties nil 'standard))
      (let* ((key (car pair))
             (val (cdr pair)))
        (when (and (stringp val) (string-match-p "WARNING" val)
                   (not (member key schema-keys))
                   (not (member key '("ID"))))
          (condition-case nil
              (let ((new-val (read-string (format "%s: " key))))
                (unless (string-empty-p new-val)
                  (org-entry-put nil key new-val)
                  (setq prompted (1+ prompted))))
            (quit nil)))))
    prompted))
#+end_src

#+begin_src emacs-lisp :lexical t
(defun my/manifolding-atlas--fix-violations ()
  "Prompt for each missing or WARNING-valued schema field.
Writes the value to the property drawer only; the
`manifolding-atlas-note-meta-get-list' advice makes schema validation fall back
to the property drawer when no description-list value exists."
  (interactive)
  (let* ((note (my/manifolding-atlas--buffer-note))
         (fields-and-lookup (my/manifolding-atlas--schema-field-lookup note))
         (all-fields (car fields-and-lookup))
         (field-lookup (cdr fields-and-lookup))
         (schema-keys (hash-table-keys field-lookup))
         prompted)
    (unless note
      (user-error "Not in a manifolding-atlas note buffer"))
    (setq prompted (my/manifolding-atlas--fix-schema-fields schema-keys field-lookup))
    (setq prompted (+ prompted (my/manifolding-atlas--fix-warning-props schema-keys)))
    (save-buffer)
    (message "Updated %d properties in drawer." prompted)))

#+begin_src emacs-lisp
(defun my/manifolding-atlas--capture-grid-toggle-all (entries buf)
  "SPC: if anything is selected, deselect all; else select all."
  (let ((any (seq-some (lambda (e) (plist-get e :selected)) entries)))
    (dolist (e entries)
      (plist-put e :selected (not any))
      (my/manifolding-atlas--capture-grid-redraw e))))

(defun my/manifolding-atlas--capture-grid-row-at (entries)
  "Return the entry whose label appears on the current line."
  (let ((line (buffer-substring-no-properties
               (line-beginning-position) (line-end-position))))
    (seq-find (lambda (e)
                (string-match-p
                 (regexp-quote (plist-get e :label)) line))
              entries)))

(defun my/manifolding-atlas--capture-grid-loop ()
  "Run the property grid. Returns 'confirm or 'cancel.
REAL avy overlays every row: jump to a row to toggle it (the first
jump SOLO-selects in capture mode). SPC flips all/none without
leaving avy; pausing surfaces RET/q."
  (let* ((entries (my/manifolding-atlas--capture-grid-entries))
         (buf (get-buffer-create "*Atlas Grid*"))
         (solo my/manifolding-atlas--capture-selection-mode))
    (my/manifolding-atlas--capture-grid-render buf entries)
    (my/manifolding-atlas--grid-display buf)
    (catch 'exit
      (while t
        (let ((win (get-buffer-window buf t))
              spc jumped ret quit)
          (unless win
            (throw 'exit 'cancel))
           (with-selected-window win
             (let* ((avy-all-windows nil)
                    (avy-dispatch-alist
                     (append (list (cons ?\s (lambda (_) (message "atlas-grid SPC dispatch") (setq spc t) t)) (cons ?\r (lambda (_) (message "atlas-grid RET dispatch") (setq ret t) t)) (cons ?\n (lambda (_) (message "atlas-grid RET dispatch \\n") (setq ret t) t)) (cons ?q (lambda (_) (message "atlas-grid Q dispatch") (setq quit t) t))) avy-dispatch-alist)))
               (condition-case err
                   (progn (avy-goto-line)
                          (let ((e (my/manifolding-atlas--capture-grid-row-at
                                    entries)))
                           (when e
                             (if solo
                                 (progn
                                   (dolist (x entries)
                                     (unless (eq x e)
                                       (plist-put x :selected nil)))
                                   (plist-put e :selected t)
                                   (setq solo nil))
                                 (plist-put e :selected
                                            (not (plist-get e :selected))))
                                (setq jumped t)
                                (my/manifolding-atlas--capture-grid-refresh
                                 entries buf))))
                 (error (warn "atlas-grid avy: %s" (error-message-string err)) nil))))
           (when spc
             (setq spc nil)
             (setq solo nil)
             (my/manifolding-atlas--capture-grid-toggle-all entries buf))
          (when ret
            (setq my/manifolding-atlas--selected-keys
                  (mapcar (lambda (e) (plist-get e :key)) (seq-filter (lambda (e) (plist-get e :selected)) entries)))
            (throw (quote exit) (if my/manifolding-atlas--selected-keys (quote confirm) (quote cancel))))
          (when quit
            (setq my/manifolding-atlas--selected-keys nil)
            (throw (quote exit) (quote cancel)))
          (unless (or spc ret quit)
            (pcase (read-key "letters jump · SPC all/none · RET confirm · q cancel")
              ((or ?\r ?\n) (setq my/manifolding-atlas--selected-keys
                         (mapcar (lambda (e) (plist-get e :key))
                                 (seq-filter
                                  (lambda (e) (plist-get e :selected))
                                  entries)))
                   (throw 'exit
                          (if my/manifolding-atlas--selected-keys
                              'confirm 'cancel)))
              (?q  (setq my/manifolding-atlas--selected-keys nil)
                   (throw 'exit 'cancel))
              (_ nil))))))))

(defun my/manifolding-atlas--update-grid-session ()
  "SPC v P: run the grid UNSELECTED until cancel.
Confirm applies every picked prompt to the note, then reopens the
grid so more properties can be updated in the same sitting."
  (catch 'done
    (while t
      (unless (eq 'confirm (my/manifolding-atlas--capture-grid-loop))
        (throw 'done nil))
      (let ((keys my/manifolding-atlas--selected-keys))
        (setq my/manifolding-atlas--selected-keys nil)
        (dolist (key keys)
          (my/manifolding-atlas--run-update-key key))
        (when (fboundp 'my/manifolding-atlas--add-live-transcludes)
          (my/manifolding-atlas--add-live-transcludes
           my/manifolding-atlas--current-note))
        (save-buffer)
        (message "Updated properties on %s"
                 (manifolding-atlas-note-title
                  my/manifolding-atlas--current-note))))))

**** Selection review
Confirming the grid opens this review: a small buffer listing every
toggled property. RET starts the capture with those picks, SPC steps
back into the grid with checkmarks intact, q cancels outright.
C-g at any point unwinds to the quit handler = skip everything.
#+begin_src emacs-lisp
(defun my/manifolding-atlas--selection-review-buffer ()
  "Render the pending selection list; return the buffer."
  (let ((buf (get-buffer-create "*Atlas Selection*"))
        (n 0))
    (with-current-buffer buf
      (erase-buffer)
      (org-mode)
      (insert "* Selected properties\n\n")
      (dolist (k my/manifolding-atlas--selected-keys)
        (setq n (1+ n))
        (insert (format "- %s\n"
                        (or (plist-get
                             (my/manifolding-atlas--update-prompt-spec k)
                             :label)
                            (symbol-name k)))))
      (insert (format "\n%d propert%s will be asked; everything else gets WARNING.\n"
                      n (if (= n 1) "y" "ies"))))
    buf))
#+end_src

**** Review loop
Synchronous and deliberately not a transient: it runs inside the
confirm suffix, so nesting another transient here would fight the
grid's own teardown. read-key keeps it bulletproof.
#+begin_src emacs-lisp
 (defun my/manifolding-atlas--selection-review ()
   "Review the picks. Returns 'start, 'back or 'cancel.
 'start stores the picks in `my/manifolding-atlas--selection-result'."
   (let ((buf (my/manifolding-atlas--selection-review-buffer))
         action)
     (condition-case nil
         (display-buffer buf '((display-buffer-reuse-window display-buffer-at-bottom) (window-height . 0.30)))
       (error (pop-to-buffer buf)))
     (when-let ((win (get-buffer-window buf)))
       (select-window win))
    (while (not action)
      (pcase (read-key "RET start · SPC back to menu · q cancel")
        ((or ?\r ?\n)
         (setq my/manifolding-atlas--selection-result
               (reverse my/manifolding-atlas--selected-keys)
               action 'start))
        (?\s (setq action 'back))
         (?q  (setq action 'cancel))))
    (unless (eq action 'start)
      (my/manifolding-atlas--kill-buffer-reuse-window buf))
    action))
#+end_src

#+begin_src emacs-lisp
(defun my/manifolding-atlas-select-properties-menu (contexts)
  "Open the capture property grid for CONTEXTS; return picked keys."
  (setq my/manifolding-atlas--current-note nil
        my/manifolding-atlas--current-context 'file
        my/manifolding-atlas--current-contexts contexts
        my/manifolding-atlas--selected-keys nil
        my/manifolding-atlas--selection-result nil
        my/manifolding-atlas--capture-selection-mode t)

(defun my/manifolding-atlas--set-file-property (key value strategy)
  "Set KEY to VALUE in the file-level property drawer using STRATEGY."
  (save-excursion
    (goto-char (point-min))
    (if (re-search-forward "^:PROPERTIES:$" nil t)
        (let ((end (save-excursion
                     (or (re-search-forward "^:END:$" nil t) (point-max)))))
          (goto-char (match-end 0))
          (if (re-search-forward (format "^:%s: ?" (regexp-quote key)) end t)
              (if (string-empty-p value)
                  (delete-region (line-beginning-position) (1+ (line-end-position)))
                (when (eq strategy 'replace)
                  (replace-match (format ":%s: %s" key value) t t)))
            (unless (string-empty-p value)
              (re-search-forward "^:END:$" nil t)
              (beginning-of-line)
              (insert (format ":%s: %s\n" key value)))))
      (unless (string-empty-p value)
        (insert ":PROPERTIES:\n")
        (insert (format ":%s: %s\n" key value))
        (insert ":END:\n")))))

(defun my/manifolding-atlas--apply-fragment (frag spec)
  "Apply FRAG plist from SPEC to the current buffer."
  (let ((strategy (or (plist-get spec :merge-strategy) 'replace))
        (at-heading (org-at-heading-p)))
    (unless at-heading
      (setq at-heading (ignore-errors (org-back-to-heading t) t))
      (unless at-heading
        (setq at-heading (ignore-errors (org-next-visible-heading 1) t))))
    (when (plist-member frag :todo)
      (my/manifolding-atlas--apply-todo-fragment (plist-get frag :todo)))
    (when-let ((tags (plist-get frag :tags)))
      (if at-heading
          (pcase strategy
            ('merge (org-set-tags (cl-union (org-get-tags nil t) tags :test #'string=)))
            ('replace (org-set-tags tags)))
        (user-error "Tags on file-level notes not supported yet")))
    (when-let ((props (plist-get frag :properties)))
      (dolist (pair props)
        (if at-heading
            (let ((v (cdr pair)))
              (if (string-empty-p v)
                  (org-entry-delete nil (car pair))
                (pcase strategy
                  ('merge (unless (org-entry-get nil (car pair))
                            (org-entry-put nil (car pair) v)))
                  ('replace (org-entry-put nil (car pair) v)))))
          (my/manifolding-atlas--set-file-property (car pair) (cdr pair) strategy))))
    (when-let ((body (plist-get frag :body)))
      (save-excursion
        (goto-char (point-max))
        (insert "\n" body)))
    (when-let ((fns (plist-get frag :post-apply)))
      (let ((file (buffer-file-name)))
        (dolist (fn fns)
          (funcall fn my/manifolding-atlas--current-note))
        (unless (and file (eq (current-buffer) (get-file-buffer file)))
          (when file (find-file file)))))))
